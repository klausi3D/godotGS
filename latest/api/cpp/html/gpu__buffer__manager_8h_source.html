40 static bool _auto_free_is_valid(RenderingDevice *p_device, const RID &p_rid) {
41 return p_device->uniform_set_is_valid(p_rid) ||
42 p_device->compute_pipeline_is_valid(p_rid) ||
43 p_device->render_pipeline_is_valid(p_rid) ||
44 p_device->framebuffer_is_valid(p_rid);
45 }
46
47 static void _free_pending_delete(const PendingDelete &p_pending_delete) {
48 if (!p_pending_delete.rid.is_valid() || !p_pending_delete.device) {
49 return;
50 }
51
52 if (p_pending_delete.auto_free) {
53 // Auto-free types (PR 103113): only free if still valid.
54 if (_auto_free_is_valid(p_pending_delete.device, p_pending_delete.rid)) {
55 p_pending_delete.device->free(p_pending_delete.rid);
56 }
57 } else {
58 p_pending_delete.device->free(p_pending_delete.rid);
59 }
60 }

62 public:
63 void queue_free(RenderingDevice *p_device, const RID &p_rid, uint32_t p_delay = 2) {
64 if (!p_device || !p_rid.is_valid()) {
65 return;
66 }
67 bool auto_free = _auto_free_is_valid(p_device, p_rid);
68 pending.push_back({p_rid, p_device, current_frame + p_delay, auto_free});
69 }
70
71 void process_frame() {
72 current_frame++;
73 uint32_t write_index = 0;
74 const uint32_t pending_count = pending.size();
75 for (uint32_t read_index = 0; read_index < pending_count; read_index++) {
76 if (pending[read_index].frame_delay <= current_frame) {
77 _free_pending_delete(pending[read_index]);
78 continue;
79 }
80
81 if (write_index != read_index) {
82 pending[write_index] = pending[read_index];
83 }
84 write_index++;
85 }
86 pending.resize(write_index);
87 }
88
89 void flush_all() {
90 for (uint32_t i = 0; i < pending.size(); i++) {
91 _free_pending_delete(pending[i]);
92 }
93 pending.clear();
94 }
95 };
