41template <typename T>
42[[nodiscard]] bool quality_resize_or_fail(Vector<T> &p_vector, uint64_t p_count, const char *p_where)
43{
44 if (!quality_allocation_fits(p_count, sizeof(T), quality_allocation_size_max(),
45 quality_allocation_header_bytes())) {
46 p_vector.clear();
47 ERR_FAIL_V_MSG(
48 false,
49 vformat("%s: rounded HLOD quality allocation (%d elements of %d bytes plus header) exceeds size_t.",
50 String(p_where), p_count, uint64_t(sizeof(T))));
51 }
52 if (!gs_resize_or_fail(p_vector, int64_t(p_count), p_where)) {
53 return false;
54 }
55 if (p_count) {
56 // Same-size resize does not fork an aliased CowData. Acquire/check the
57 // writable storage here, so later ptrw()/write[] use an exclusive owner.
58 T *writable = nullptr;
59#ifdef TESTS_ENABLED
60 if (!quality_consume_writable_failure(p_where))
61#endif
62 {
63 writable = p_vector.ptrw();
64 }
65 if (writable == nullptr) {
66 p_vector.clear();
67 ERR_FAIL_V_MSG(false, vformat("%s: cannot acquire writable HLOD quality storage (%d entries).",
68 String(p_where), p_count));
69 }
70 }
71 return true;
72}

123// Rule 3: a node is drawn whenever its geometric error projects to at most `tau` pixels, and the
124// runtime `tau` is a user budget, so one bake decision serves every `tau` the ADR evaluates (1, 2
125// and 4 px, section 5.2). Each candidate is scored once per `tau` at pixel pitch `eps / tau`, and
126// the per-`tau` errors are combined by their geometric mean: the acceptance gate reports PSNR, a
127// logarithm of the error, so the decision that minimises the expected dB loss over the budgets
128// compares the mean of log errors. `kQualityErrorFloor` is the MSE of 8-bit output rounding
129// alone; errors below it are indistinguishable on screen and must not send a logarithm to -inf.
130static constexpr uint32_t kQualityTauCount = 3u;
131static constexpr double kQualityTau[kQualityTauCount] = {1.0, 2.0, 4.0};
132static constexpr double kQualityErrorFloor = 1.0 / (255.0 * 255.0 * 12.0);
133struct QualityBandResult {
134 QualityResult taus[kQualityTauCount]; // per-`tau` 22-view results, in kQualityTau order
135 float merged_error = 0, selected_error = 0; // geometric means over `tau` (floored)
136 bool use_merged = true;
137};
