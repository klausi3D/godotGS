10// Overlap-record estimate (ADR 6.5): the distance used for a node's footprint term is never smaller than this fraction
11// of the node's bounding radius, so a camera inside a node's sphere (or in the shell up to radius x 1.25 outside it) gets
12// a finite estimate. A heuristic, not a bound (see CutHistory::select()).
13constexpr double kCostInsideDistanceFraction = 0.25;
14// The admission factor of the overlap-record cap (ADR 6.5): a cut is admitted while its record estimate stays within this
15// fraction of the record budget (the allocated overlap storage). The estimate is a heuristic, so the rest is its margin.
16constexpr double kRecordAdmissionFactor = 0.8;
17// Refinement band factor h (ADR 6.4, 6.7): a node with tau < e <= h x tau sits in its band. Without the cross-fade the
18// band is threshold hysteresis; with it (CutRequest::crossfade) the band is the parent -> children blend.
19constexpr double kRefineBandFactor = 1.25;
20// An old-configuration node whose remaining weight would fall below this is dropped rather than drawn.
21constexpr float kFadeMinWeight = 1.0f / 64.0f;
22// Bounded fade bookkeeping per history: groups and old-configuration members beyond these snap (hard switch, counted).
23constexpr uint32_t kMaxFadeGroups = 4096;
24constexpr uint32_t kMaxFadeMembers = 32768;
25// The route's fade length in selections (about 0.27 s at 60 fps).
26constexpr uint32_t kDefaultFadeFrames = 16;
27// Weight of one side of a blend at progress p (0..1): overlapping, so the incoming side reaches full weight at p = 0.5
28// and the outgoing side keeps it until then. Two configurations each cover a leaf once; drawn at fade_cover(p) and
29// fade_cover(1 - p) they cover it between once and twice, never less, so an opaque surface never turns see-through
30// mid-blend (a p / 1 - p blend of two opaque layers is only ~75% opaque at p = 0.5).
31inline float fade_cover(double p_progress) { return float(CLAMP(2.0 * p_progress, 0.0, 1.0)); }

47 // A caller that wants array-order independence passes the rank of each node's stable key. Null keeps
48 // the node index as the tie-break (the pre-existing behaviour).
49 const uint32_t *tie_rank = nullptr;
50 uint32_t tie_rank_count = 0;
51 // Page admission for the cut and residency demand (ADR 6.6): when page_splats != 0, a node occupies
52 // ceil(payload / page) atlas pages. Every transition must fit page_budget together with the cut, every
53 // ancestor of it, and already demanded transitions, leaving the largest non-stale node's run as reserve.
54 // Unprepared snapshots retain option-1 admission without reserve. Parents stay charged during overlap.
55 // The mandatory root uses physical page_budget.
56 uint32_t page_splats = 0;
57 uint64_t page_budget = 0;
58 // S6 cross-fade (ADR 6.7). A node is refined whenever e > tau. Inside its band (tau < e < h x tau) the node and its
59 // complete visible children are drawn together: with t = (e - tau) / ((h - 1) x tau), children at fade_cover(t), the
60 // node at fade_cover(1 - t).
61 // t is a function of the camera only, so a parked camera holds a fixed blend. Fade pairs are admitted after every
62 // full refinement, in refinement priority order, while the pair fits every cap; the first that does not fit stops
63 // them, and its node is drawn alone (a hard switch). Children of a fade pair are not refined further.
64 // false keeps the threshold hysteresis of ADR 6.4 (1) and draws every node at fade 1.
65 bool crossfade = false;
66 // S6b temporal fade. 0: off. Otherwise every change of the drawn configuration (a refinement, a merge, a budget or
67 // residency driven switch, the start or end of a band fade) is blended over this many selections: the changed region
68 // (the highest node whose subtree changed) is drawn as fade_cover(t) x (live configuration) + fade_cover(1 - t) x (the
69 // configuration drawn before the change), t = age / fade_frames. A region over fading ones subsumes them; further
70 // changes inside a fading region join its live configuration. Per leaf the weights cover between once and twice.
71 // The cut is selected exactly as without the fade (full caps); the old configuration is drawn only from the cap
72 // headroom the cut leaves (splats, nodes, records). A region that does not fit is switched at once and counted
73 // (fade_snapped), so fades never feed back into the cut. Depends on the selection sequence, not on wall clock
74 // or atlas placement; a parked camera draws identical frames once fade_frames selections have passed.
75 uint32_t fade_frames = 0;
