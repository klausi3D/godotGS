11// Dedicated rendered-view history: streaming ticks and shadow views must not overwrite it.
12// T0 (probe_path.py / probe_world.gd): far=400m, so far/8=50m. At its maximum
13// 120m/s, steps are 2/4/12m at 60/30/10Hz. S9/C4, S11t (no checkpoint), S12/C7
14// jump ~519/393/202m on the default T0, all >4x the threshold. Moving boundaries
15// turn as much as 139.4 degrees; use >160 degrees, also well above 720deg/s at
16// 10Hz (72 degrees/frame). Angles are dimensionless; translation scales with the
17// frustum. A stalled wall-clock driver can exceed these motion bounds; no pose-only
18// detector can distinguish that from a teleport. Games should hint smaller cuts.

33 bool update(const Transform3D &p_camera_to_world, const Projection &p_projection,
34 const Vector2i &p_viewport, uint64_t p_hint_generation) {
35 pending = pending || p_hint_generation != hint_generation;
36 hint_generation = p_hint_generation;
37 if (have_previous) {
38 const real_t scale = MIN(previous_projection.get_z_far(), p_projection.get_z_far());
39 const bool position_cut = scale > 0 && previous.origin.distance_to(p_camera_to_world.origin) > scale / 8;
40 const Vector3 before = previous.basis.get_column(2).normalized();
41 const Vector3 after = p_camera_to_world.basis.get_column(2).normalized();
42 const bool direction_cut = before.dot(after) < Math::cos(Math::deg_to_rad(real_t(160)));
43 // Use the unjittered lens. Incremental zoom is continuous; a >25% lens
44 // scale/clip change or projection-mode switch invalidates the old demand.
45 const bool lens_cut = previous_projection.is_orthogonal() != p_projection.is_orthogonal() ||
46 scale_jump(Math::abs(previous_projection[0][0]), Math::abs(p_projection[0][0])) ||
47 scale_jump(Math::abs(previous_projection[1][1]), Math::abs(p_projection[1][1])) ||
48 scale_jump(previous_projection.get_z_far(), p_projection.get_z_far()) ||
49 scale_jump(previous_projection.get_z_near(), p_projection.get_z_near());
50 pending = pending || position_cut || direction_cut || lens_cut || previous_viewport != p_viewport;
51 }
52 previous = p_camera_to_world;
53 previous_projection = p_projection;
54 previous_viewport = p_viewport;
55 have_previous = true;
56 return pending;
57 }
