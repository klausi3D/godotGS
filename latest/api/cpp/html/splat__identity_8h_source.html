8// The high word of the sort key remains the full sortable fp32 depth. Only
9// its existing low uint32 is partitioned. Counts are immutable bake/registered
10// topology counts, never the visible set or the resident page allocation.
11namespace gs_splat_identity {
12constexpr uint32_t MIN_LOCAL_BITS = 14;
13inline uint32_t bits_for_count(uint64_t p_count) {
14 uint32_t bits = 0;
15 for (uint64_t maximum = p_count > 0 ? p_count - 1 : 0; maximum; maximum >>= 1) { ++bits; }
16 return bits;
17}
18struct Layout {
19 uint32_t r = 0, c = 0, l = MIN_LOCAL_BITS;

21 String refusal() const {

23 }
24};
25inline Layout instance_layout(uint32_t p_registered_rows, uint32_t p_max_chunks, uint32_t p_max_chunk_splats) {
26 Layout layout;
27 layout.r = bits_for_count(uint64_t(p_registered_rows) + 1u); // ranks 1..rows; 0 is reserved.
28 layout.c = bits_for_count(p_max_chunks);
29 layout.l = MAX(MIN_LOCAL_BITS, bits_for_count(p_max_chunk_splats));

31}
32inline uint32_t world_local_bits(uint32_t p_max_node_splats) {
33 return MAX(MIN_LOCAL_BITS, bits_for_count(p_max_node_splats));
34}
35struct WorldLayout {
36 uint32_t l = MIN_LOCAL_BITS;
37 bool node_id_folded = false;
38 bool fits() const { return l <= 31u; }
39 uint32_t node_mask() const { return fits() ? (0x7fffffffu >> l) : 0u; }
40 // Mirror of the world branch of gs_splat_composite_key. At L=14, node
41 // IDs differing by 2^17 collide; the local field and zero prefix survive.
42 uint32_t sort_key(uint32_t p_node, uint32_t p_local) const {
43 return fits() ? ((p_node & node_mask()) << l) | p_local : 0u;
44 }
45 String refusal() const {
46 return fits() ? String() : vformat("splat_identity_world_local_tie_overflow:L=%d", l);
47 }
48 void append_stats(Dictionary &r_stats) const {
49 r_stats["tie_world_node_folded"] = node_id_folded;
50 }
51};
52inline WorldLayout world_layout(uint32_t p_nodes, uint32_t p_max_node_splats) {
53 WorldLayout layout;
54 layout.l = world_local_bits(p_max_node_splats);
55 layout.node_id_folded = layout.fits() && uint64_t(p_nodes) > uint64_t(layout.node_mask()) + 1u;
56 return layout;
57}
58// Unfolded budget diagnostic retained for existing test vectors. Production

60inline String world_refusal(uint32_t p_nodes, uint32_t p_max_node_splats) {

63}
64// Low six bits retain the instance flags. The local-width/singleton fields fit
65// below bit 14; the ordered prefix starts at C+L >= 14. No ABI stride change.
66inline uint32_t instance_sort_flags(uint32_t p_rank, const Layout &p_layout, bool p_single_chunk) {
67 if (!p_layout.fits()) { return 0; } // Refused rows are omitted before publication.
68 return 0x80000000u | (p_rank << (p_layout.c + p_layout.l)) |
69 (p_layout.l << GS_INSTANCE_SORT_LOCAL_SHIFT) | (p_single_chunk ? GS_INSTANCE_SORT_SINGLE_CHUNK : 0u);
70}
