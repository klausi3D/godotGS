38#ifdef TESTS_ENABLED
39 // #798 review round 3: the allocation-failure contracts on this class (the
40 // populate_from_gaussian_data() reset, and the merge refusal driven by the
41 // structured getters) could not be exercised by any test, because a real
42 // Vector::resize() failure cannot be provoked from a unit test at any size the
43 // test itself can afford to allocate. Their "failure arm" assertions therefore
44 // never ran -- a green test that proves nothing, which is precisely the defect
45 // shape these guards exist to prevent.
46 //
47 // These arm-once hooks reproduce the exact post-failure buffer SHAPES that
48 // cowdata.h leaves behind, so the production failure branches execute for real.
49 // TESTS_ENABLED only: absent from editor and export builds (see #725).
50 enum TestLaneFailure {
51 TEST_LANE_FAILURE_NONE,
52 // A fresh or CoW-forked lane whose _alloc() failed: _ptr stays null, so the
53 // lane reads back EMPTY rather than short.
54 TEST_LANE_FAILURE_EMPTY,
55 // A failed GROW of an already-populated lane: the lane keeps its previous,
56 // shorter size. This is the shape that would silently overflow the heap.
57 TEST_LANE_FAILURE_SHORT,
58 // A failed SHRINK of an already-populated lane: CowData::_fork_allocate()
59 // returns before writing the new size when _realloc() fails, so the lane keeps
60 // its previous, LARGER size. Applied to sh_high_order_coefficients, because that
61 // lane's LENGTH is what _recalculate_sh_component_counts() derives the SH term
62 // count from -- an oversized lane therefore inflates the term count, which the
63 // write loop uses as the stride into the SOURCE GaussianData's (correctly sized,
64 // hence shorter) high-order array. That is an out-of-bounds READ on the way to
65 // sealing a corrupted asset, and unlike the SHORT shape a `size() < required`
66 // test cannot see it.
67 TEST_LANE_FAILURE_OVERSIZED,
68 };
69 enum TestGetterFailure {
70 TEST_GETTER_FAILURE_NONE,
71 // get_normal_vectors() output allocation failed. Deliberately a lane that
72 // _ensure_buffer_sizes() sizes UNCONDITIONALLY to splat_count, so an empty
73 // result can only mean OOM -- the case the merge guard must refuse rather
74 // than paper over with the (0,1,0) default.
75 TEST_GETTER_FAILURE_NORMALS,
76 // get_spherical_harmonics_buffer() output allocation failed. This is the
77 // LARGEST output of the structured-getter family -- splat_count * (1 +
78 // first_order + high_order) * 3 floats -- so it is the one that realistically
79 // fails first, and it is the only member whose expected length is not simply
80 // splat_count. populate_gaussian_data() must therefore size its check off the
81 // SH term counts, not off the splat count, which this shape is what proves.
82 TEST_GETTER_FAILURE_SPHERICAL_HARMONICS,
83 };
