23// Fallible allocation probe for the bake's data-sized allocations (the OOM-probe row of
24// docs/architecture/adr-import-input-hardening.md): LocalVector::reserve() CRASH_CONDs on a failed
25// memrealloc, so each such allocation is first proven available with memalloc(), which returns
26// null instead. False means "do not allocate; report ERR_OUT_OF_MEMORY". p_where names the site
27// for the message and for the TESTS_ENABLED forced-failure seam in gs_vector_alloc.h (a string
28// literal). Defined in gaussian_splat_hlod_bake.cpp.
29bool bake_allocation_probe(uint64_t p_bytes, const char *p_where);
30
31// Resizes r_vec to p_count, proving any growth with bake_allocation_probe() first. A grow makes
32// LocalVector::reserve() allocate max(p_count, 2, 1.5 x capacity) elements; that is what is probed
33// and then reserved explicitly, so the probed and the allocated size are the same. Returns false
34// with r_vec unchanged when the probe fails.
35template <typename T>
36[[nodiscard]] bool fallible_resize(LocalVector<T> &r_vec, uint64_t p_count, const char *p_where) {
37 if (p_count > uint64_t(UINT32_MAX)) {
38 return false;
39 }
40 const uint64_t capacity = r_vec.get_capacity();
41 if (p_count > capacity) {
42 const uint64_t grown = MAX(p_count, MAX<uint64_t>(2u, capacity + ((1u + capacity) >> 1u)));
43 if (grown > uint64_t(UINT32_MAX) || !bake_allocation_probe(grown * uint64_t(sizeof(T)), p_where)) {
44 return false;
45 }
46 r_vec.reserve(uint32_t(grown));
47 }
48 r_vec.resize(uint32_t(p_count));
49 return true;
50}

52// Interior-merge allocation bound (#1149 review). Per interior node the merge holds the children's
53// payloads concatenated twice (bake_world's re-framed copy and the merge reference): at most
54// kMaxChildren x max(leaf_max_splats, node_max_splats) = 8 x 16,384 = 131,072 splats at the
55// defaults, i.e. 18.0 MiB of Gaussian (144 B) plus 131,072 x sh_high_order_count x 12 B of SH
56// (22.5 MiB at 15 coefficients) EACH, before LocalVector's 1.5x growth; the bucketing scratch adds
57// 24 + 16 + 1 B per splat (5.1 MiB), the merged output and the payload up to 16,384 splats each.
58// That is ~100 MiB per node, scaling with the SH count and the interior parameters (neither is
59// capped by bake_world), and every node's payload is retained until the end of the bake, which
60// scales with the world. So these are not "small constant" allocations: each one goes through
61// fallible_resize() and a failure surfaces as ERR_OUT_OF_MEMORY. Excluded: merge_cell_sat()'s
62// SH accumulator, which allocates only above 16 SH coefficients (SH degree > 3) and then
63// sh_high_order_count x 24 B per cell.

120// `sat` merge of p_members (indices into p_rep) into one splat (ADR §6.3):
121// weights w = alpha x sigma_a x sigma_b; mean and covariance by weighted moment matching (H3DGS);
122// SH (DC, first order, high order) as weighted means; M = sum alpha 2 pi sigma_a sigma_b;
123// the two largest axes inflate by k = clamp(sqrt(M / C), 1, 1.25); alpha = min(1 - exp(-M / C), 0.99).
124// A single member is copied unchanged (identity, ADR §6.3 / §11 decision 8d).
125// Non-geometric fields (normal, painterly data, render_meta) come from the heaviest member (the
