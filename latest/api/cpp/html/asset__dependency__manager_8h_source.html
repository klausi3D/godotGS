13// Universal Asset ID with collision-resistant hashing
14struct AssetID {
15 uint64_t id_high;
16 uint64_t id_low;
17
18 AssetID() : id_high(0), id_low(0) {}
19 AssetID(uint64_t high, uint64_t low) : id_high(high), id_low(low) {}
20
21 bool operator==(const AssetID &other) const {
22 return id_high == other.id_high && id_low == other.id_low;
23 }
24
25 bool operator!=(const AssetID &other) const {
26 return !(*this == other);
27 }
28
29 String to_string() const;
30 static AssetID from_string(const String &p_str);
31 static AssetID generate_from_path(const String &p_path);
32 static AssetID generate_unique();
33
34 bool is_valid() const { return id_high != 0 || id_low != 0; }
35};

37static _FORCE_INLINE_ uint32_t hash(const AssetID &p_id) {
38 return hash_murmur3_one_64(p_id.id_high) ^ hash_murmur3_one_64(p_id.id_low);
39}
