8// Helper functions to safely submit/sync RenderingDevice
9// Only the main/local RenderingDevice can call submit() and sync()
10namespace gs_device_utils {
11
12inline bool is_local_device(RenderingDevice *p_device) {
13 if (!p_device) {
14 return false;
15 }
16 // Local devices are NOT the main instance and CAN call submit/sync
17 return !p_device->is_main_rendering_device();
18}
19
20// True when p_device is a local device that has been submit()ed and not yet
21// sync()ed. A local device in that state has already ended its command buffer and
22// its draw graph, so anything that re-enters the frame lifecycle -- a second
23// submit(), or a synchronous readback, which stalls through
24// _flush_and_stall_for_all_frames() -> _end_frame() -- runs against ended state
25// and faults inside the driver. (GS #685)
26inline bool has_outstanding_submit(RenderingDevice *p_device) {
27 return p_device && is_local_device(p_device) && p_device->is_local_device_submission_pending();
28}
29
30// Complete an outstanding submit() so the device is back in a recording state.
31// Returns true when a sync() was actually issued.
32//
33// This is the enforceable half of the local-device submission contract: rather
34// than documenting "do not read back while a submit is in flight" and relying on
35// call ordering to honour it, every helper below settles first, so the illegal
36// state cannot be observed by the operation that would fault on it. Callers pay
37// nothing they were not already paying -- a synchronous readback stalls for all
38// frames regardless, and a second submit() would otherwise have been rejected
39// outright by RenderingDevice::submit()'s "device already submitted" guard.
40inline bool settle_outstanding_submit(RenderingDevice *p_device) {
41 if (!has_outstanding_submit(p_device)) {
42 return false;
43 }
44 p_device->sync();
45 return true;
46}
47
48// Safe submit - only submits on local (non-main) devices.
49//
50// On a local device the submission is COMPLETED here, not left in flight. That is
51// not caution, it is the only representable behaviour: RenderingDevice::submit()
52// runs _end_frame(), so the frame's command buffer and draw graph are ended, and
53// only sync() -> _begin_frame() reopens them. Every caller in this module keeps
54// going afterwards -- more compute lists, more buffer_updates, more readbacks --
55// and all of that would be recorded into an ended command buffer and an ended
56// draw graph, which the next _end_frame() then replays. That is a driver fault,
57// observed as VK_ERROR_DEVICE_LOST on the instance cull -> sort path, and as a
58// direct crash in _capture_instance_count_sync (#685).
59//
60// Nothing in this module ever paired safe_submit with a later sync, so the
61// in-flight state had no owner and no consumer -- it existed only as a window in
62// which every following device call was illegal.
63//
64// This is a no-op on the main RenderingDevice, which is what every shipping
65// configuration uses (GaussianSplatManager::get_primary_rendering_device()), so
66// production behaviour is unchanged; it defines the local-device path the GPU
67// test harness runs on.
68inline void safe_submit(RenderingDevice *p_device) {
69 if (p_device && is_local_device(p_device)) {
70 settle_outstanding_submit(p_device);
71 p_device->submit();
72 p_device->sync();
73 }
74 // Main device: submit is handled automatically by Godot's frame
75}
76
77// Safe sync - completes an outstanding local-device submission, if there is one.
78//
79// Deliberately NOT an unconditional sync(): RenderingDevice::sync() ERR_FAILs with
80// "sync can only be called after a submit" when nothing is in flight, and since
81// safe_submit above completes its own submission, that is now the normal state.
82// Call sites that pair safe_submit with safe_sync therefore still read correctly
83// and the second call is simply a no-op. (GS #685)
84inline void safe_sync(RenderingDevice *p_device) {
85 settle_outstanding_submit(p_device);
86 // Main device: sync is handled automatically by Godot's frame
87}
88
89// Safe submit and sync combo
90inline void safe_submit_and_sync(RenderingDevice *p_device) {
91 if (p_device && is_local_device(p_device)) {
92 settle_outstanding_submit(p_device);
93 p_device->submit();
94 p_device->sync();
95 }
96 // Main device: handled automatically by Godot's frame
97}

99// Synchronous buffer readback that is legal regardless of submission state.
100//
101// RenderingDevice::buffer_get_data() stalls through
102// _flush_and_stall_for_all_frames(), which calls _end_frame() unconditionally. On
103// a local device with an outstanding submit() that re-ends an already-ended
104// command buffer and draw graph, and the driver faults replaying the graph. Every
105// synchronous readback in this module goes through here so the invariant is
106// enforced in one place rather than restated at each call site. On the main
107// device this is exactly buffer_get_data(). (GS #685)
108inline Vector<uint8_t> safe_buffer_get_data(RenderingDevice *p_device, RID p_buffer,
109 uint32_t p_offset = 0, uint32_t p_size = 0) {
110 if (!p_device) {
111 return Vector<uint8_t>();
112 }
113 settle_outstanding_submit(p_device);
114 return p_device->buffer_get_data(p_buffer, p_offset, p_size);
115}
116
117// Same contract as safe_buffer_get_data for the texture readback path:
118// RenderingDevice::texture_get_data() stalls through the identical
119// _flush_and_stall_for_all_frames() call. (GS #685)
120inline Vector<uint8_t> safe_texture_get_data(RenderingDevice *p_device, RID p_texture, uint32_t p_layer) {
121 if (!p_device) {
122 return Vector<uint8_t>();
123 }
124 settle_outstanding_submit(p_device);
125 return p_device->texture_get_data(p_texture, p_layer);
126}

148inline const char *mode_name(ReadbackMode p_mode) {
149 switch (p_mode) {
150 case ReadbackMode::STRICT_ASYNC:
151 return "strict_async";
152 case ReadbackMode::ASYNC_WITH_SYNC_BOOTSTRAP:
153 return "async_with_sync_bootstrap";
154 case ReadbackMode::STRICT_SYNC:
155 return "strict_sync";
156 case ReadbackMode::DEBUG_VALIDATION:
157 return "debug_validation";
158 default:
159 return "strict_async";
160 }
161}
162
163inline ReadbackPolicy make_policy(ReadbackMode p_mode) {
164 ReadbackPolicy policy;
165 policy.mode = p_mode;
166 switch (p_mode) {
167 case ReadbackMode::STRICT_ASYNC: {
168 break;
169 }
170 case ReadbackMode::ASYNC_WITH_SYNC_BOOTSTRAP: {
171 policy.allow_sync_bootstrap = true;
172 // Keep startup correctness with a single bootstrap sample, but
173 // preserve strict async behavior in steady-state to avoid recurring
174 // CPU/GPU stalls from pending/enqueue sync readbacks.
175 policy.allow_sync_pending_readback = false;
176 policy.allow_sync_enqueue_fallback = false;
177 break;
178 }
179 case ReadbackMode::STRICT_SYNC:
180 case ReadbackMode::DEBUG_VALIDATION: {
181 policy.allow_sync_readback = true;
182 policy.allow_sync_sort_fallback = true;
183 policy.allow_sync_bootstrap = true;
184 policy.allow_sync_pending_readback = true;
185 policy.allow_sync_enqueue_fallback = true;
186 break;
187 }
188 default:
189 break;
190 }
191 return policy;
192}
193
194inline ReadbackPolicy resolve_readback_policy(bool p_debug_sync_requested, bool p_preserve_gpu_timestamps) {
195 if (p_debug_sync_requested && !p_preserve_gpu_timestamps) {
196 return make_policy(ReadbackMode::DEBUG_VALIDATION);
197 }
198 if (p_preserve_gpu_timestamps) {
199 return make_policy(ReadbackMode::STRICT_ASYNC);
200 }
201 return make_policy(ReadbackMode::ASYNC_WITH_SYNC_BOOTSTRAP);
202}
