61 void capture(const gs_hlod_runtime::CutHistory &p_history, const gs_hlod_runtime::CutHistory::Result &p_cut,
62 uint32_t p_nodes, const gs_hlod_runtime::DrawRecord *p_records, uint32_t p_count) {
63 // Report-only: sort ids with their record index so each fade stays with its id (a node is drawn once per cut).
64 LocalVector<uint64_t> order;
65 order.resize(p_count);
66 for (uint32_t i = 0; i < p_count; ++i) { order[i] = (uint64_t(p_records[i].node) << 32) | i; }
67 order.sort();
68 drawn_ids.resize(p_count);
69 drawn_fades.resize(p_count);
70 for (uint32_t i = 0; i < p_count; ++i) {
71 drawn_ids.set(i, int32_t(order[i] >> 32));
72 drawn_fades.set(i, p_records[uint32_t(order[i])].fade);
73 }
74 fade_pairs = p_cut.fade_pairs;
75 fade_refused = p_cut.fade_refused;
76 fade_groups = p_cut.fade_groups;
77 fade_snapped = p_cut.fade_snapped;
78 fade_snapped_total += p_cut.fade_snapped;
79 fade_extra_splats = p_cut.fade_extra_splats;
80 // FNV-1a over sorted uint32 IDs encoded little-endian, independent of host byte order. A record that is blending
81 // (S6/S6b fade != 1) also hashes its fade's bits after its ID: the digest names what is drawn, so it keeps
82 // changing while a fade is in flight and a "stable digest" settle waits for the blend to finish. With no blend
83 // every fade is 1 and the digest is exactly the ID-only digest.
84 uint64_t digest = UINT64_C(14695981039346656037);
85 auto mix_u32 = [&digest](uint32_t p_value) {
86 for (uint32_t shift = 0; shift < 32; shift += 8) {
87 digest ^= (p_value >> shift) & 0xffu;
88 digest *= UINT64_C(1099511628211);
89 }
90 };
91 for (int i = 0; i < drawn_ids.size(); ++i) {
92 mix_u32(uint32_t(drawn_ids[i]));
93 const float f = drawn_fades[i];
94 if (f != 1.0f) {
95 uint32_t bits = 0;
96 memcpy(&bits, &f, sizeof(bits));
97 mix_u32(0xFADEFADEu);
98 mix_u32(bits);
99 }
100 }
101 drawn_digest = String::num_uint64(digest, 16).lpad(16, "0"); // pad_zeros() would pad after a leading hex letter.
102 hysteresis_held.resize(p_cut.hysteresis_held_count);
103 residency_missed.resize(p_cut.residency_missed_count);
104 residency_missed_demanded.resize(p_cut.residency_missed_count);
105 uint32_t held = 0, missed = 0;
106 for (uint32_t node = 0; node < p_nodes; ++node) {
107 const uint8_t flags = p_history.diagnostic_flags(node);
108 if (flags & gs_hlod_runtime::CutHistory::HYSTERESIS_HELD) { hysteresis_held.set(held++, int32_t(node)); }
109 if (flags & gs_hlod_runtime::CutHistory::RESIDENCY_MISSED) {
110 residency_missed.set(missed, int32_t(node));
111 residency_missed_demanded[missed++] = bool(flags & gs_hlod_runtime::CutHistory::MISS_DEMANDED);
112 }
113 }
114 budget_bound = p_cut.budget_bound || p_cut.first_cap_refused_node >= 0;
115 first_cap_refused_node = p_cut.first_cap_refused_node;
116 page_report = p_cut;
117 if (p_cut.status != gs_hlod_runtime::CutHistory::INVALID && p_cut.closure_measured) {
118 page_refused_groups_total += p_cut.page_refused_groups;
119 }
120 valid = p_cut.status != gs_hlod_runtime::CutHistory::INVALID;
121 }
122};
