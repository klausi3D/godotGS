49 // FNV-1a over sorted uint32 IDs encoded little-endian, independent of host byte order.
50 uint64_t digest = UINT64_C(14695981039346656037);
51 for (int32_t id : drawn_ids) {
52 for (uint32_t shift = 0; shift < 32; shift += 8) {
53 digest ^= (uint32_t(id) >> shift) & 0xffu;

55 }
56 }
57 drawn_digest = String::num_uint64(digest, 16).lpad(16, "0"); // pad_zeros() would pad after a leading hex letter.
