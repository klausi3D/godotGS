15// BV-04b (ADR adr-beyond-vram-worlds.md D-4 / D-5): GPU timing completeness.
16//
17// Every GPU pass of a Gaussian frame is bracketed by a pair of RenderingDevice timestamps named
18// "<prefix><frame serial>_Begin" / "_End". The tile renderer parses the captured names back into a
19// per-frame-serial table and publishes one duration per pass. Before BV-04b only the tile passes had
20// timestamps; cull, projection (the depth-key pass), the depth pre-sort and the composite were
21// untimed, so the fixed cost F of the frame model could not be measured.
22//
23// This header holds the label vocabulary, the frame-serial source shared by every pass of one frame,
24// and the pure parse / reduce functions (device free, so they are unit tested on the CPU).
25namespace GaussianSplatting {

27// BV-04b: records per splat per HLOD level, from the most recently SAMPLED HLOD-route frame
28// (always-on async readback of HlodLevelRecordStatsGPU; frames between samples are not counted).
29// Slot i < GS_HLOD_LEVEL_STAT_DEEPEST_SLOT = level i; GS_HLOD_LEVEL_STAT_DEEPEST_SLOT = that level
30// and above; GS_HLOD_LEVEL_STAT_OBJECT_SLOT = object (instance) records. frame_serial 0 = no sample.
31struct HlodLevelRecordSample {
32 uint64_t frame_serial = 0;
33 uint64_t sample_count = 0;
34 uint32_t records[GS_HLOD_LEVEL_STAT_SLOTS] = {};
35 uint32_t splats[GS_HLOD_LEVEL_STAT_SLOTS] = {};
36};

40// The order is the order of execution within a frame. PASS_TOTAL is the legacy GaussianSplat_ marker.
41enum Pass : uint32_t {
42 PASS_CULL = 0, // frustum / instance-chunk cull (gpu_culler)
43 PASS_PROJECTION, // depth-key projection: instance chunk dispatch + depth compute + count clamp
44 PASS_PRESORT, // depth pre-sort of the visible splats (GPUSortingPipeline's sorter)
45 PASS_OVERLAP_COUNT, // tile_binning COUNT pass
46 PASS_PREFIX, // tile prefix scan
47 PASS_BINNING, // tile_binning EMIT pass (the "overlap emit" pass)
48 PASS_OVERLAP_SORT, // overlap-record radix sort
49 PASS_RASTER, // tile raster (compute or fragment)
50 PASS_RESOLVE, // tile resolve
51 PASS_COMPOSITE, // output composite into the engine frame (compute path)
52 PASS_TOTAL, // GaussianSplat_ frame bracket (flush-discarded on most frames; not summed)
53 PASS_COUNT,
54};

123// Frame serial. One serial names every pass of one pipeline pass (cull .. composite), so the parser
124// can tell which durations were measured in the same frame. RenderPipelineStages::prepare_frame_context
125// allocates it; the tile renderer takes it through RenderParams::frame_serial. Render thread only;
126// atomic so a stray caller on another thread cannot tear it.
127inline std::atomic<uint64_t> &_frame_serial_counter() {
128 static std::atomic<uint64_t> counter{ 0 };
129 return counter;
130}
131inline std::atomic<uint64_t> &_current_frame_serial() {
132 static std::atomic<uint64_t> current{ 0 };
133 return current;
134}

136// Allocates a fresh serial (never 0) and makes it the current one.
137inline uint64_t allocate_frame_serial() {
138 const uint64_t serial = _frame_serial_counter().fetch_add(1, std::memory_order_relaxed) + 1;
139 _current_frame_serial().store(serial, std::memory_order_relaxed);
140 return serial;
141}

143// The serial of the pipeline pass being recorded (0 = none allocated yet).
144inline uint64_t current_frame_serial() {
145 return _current_frame_serial().load(std::memory_order_relaxed);
146}

148// A caller-supplied serial (e.g. the painterly path's frame counter) was used: never hand it out again.
149inline void note_external_frame_serial(uint64_t p_serial) {
150 uint64_t seen = _frame_serial_counter().load(std::memory_order_relaxed);
151 while (seen < p_serial && !_frame_serial_counter().compare_exchange_weak(seen, p_serial, std::memory_order_relaxed)) {
152 }
153 _current_frame_serial().store(p_serial, std::memory_order_relaxed);
154}

160// Brackets one pass with "<prefix><serial>_Begin" ... "_End" on p_device. The _End marker is written
161// only by finish(), after the pass's last compute list has ended; a pass that bails out early leaves an
162// unpaired _Begin, which the parser treats as "not measured" (never as a 0 ms sample).
163// RenderingDevice refuses capture_timestamp inside a compute list with dispatches, so construct this
164// before compute_list_begin() and call finish() after compute_list_end().
165class ScopedPassTimestamp {
166 RenderingDevice *device = nullptr;
167 String label;
168
169public:
170 ScopedPassTimestamp(RenderingDevice *p_device, Pass p_pass, bool p_enabled, uint64_t p_serial = current_frame_serial()) {
171 if (!p_device || !p_enabled || p_serial == 0) {
172 return;
173 }
174 device = p_device;
175 label = make_label(p_pass, p_serial);
176 device->capture_timestamp(label + String("_Begin"));
177 }
178 void finish() {
179 if (device) {
180 device->capture_timestamp(label + String("_End"));
181 device = nullptr;
182 }
183 }
184 bool is_armed() const { return device != nullptr; }
185
186 ScopedPassTimestamp(const ScopedPassTimestamp &) = delete;
187 ScopedPassTimestamp &operator=(const ScopedPassTimestamp &) = delete;
188};

193struct StageTimes {
194 double begin_ns = -1.0;
195 double end_ns = -1.0;
196 bool is_complete() const { return begin_ns >= 0.0 && end_ns >= 0.0; }
197 // A pass is measured only when both markers exist and are ordered.
198 bool is_ordered() const { return is_complete() && end_ns >= begin_ns; }
199};

201struct FrameStages {
202 StageTimes stages[PASS_COUNT];
203};

205// Parses "<prefix><serial>_Begin|_End". Returns false for any name that is not a pass marker.
206inline bool parse_label(const String &p_name, Pass &r_pass, bool &r_is_begin, uint64_t &r_serial) {
207 bool is_begin = false;
208 int64_t suffix_pos = -1;
209 if (p_name.ends_with("_Begin")) {
210 is_begin = true;
211 suffix_pos = p_name.length() - 6;
212 } else if (p_name.ends_with("_End")) {
213 suffix_pos = p_name.length() - 4;
214 } else {
215 return false;
216 }
217 for (uint32_t i = 0; i < PASS_COUNT; i++) {
218 const Pass pass = Pass(i);
219 const String prefix = label_prefix(pass);
220 if (!p_name.begins_with(prefix)) {
221 continue;
222 }
223 const int64_t prefix_length = prefix.length();
224 if (suffix_pos <= prefix_length) {
225 return false;
226 }
227 const String serial_str = p_name.substr(prefix_length, suffix_pos - prefix_length);
228 if (!serial_str.is_valid_int()) {
229 return false;
230 }
231 r_pass = pass;
232 r_is_begin = is_begin;
233 r_serial = uint64_t(serial_str.to_int());
234 return true;
235 }
236 return false;
237}

239inline void accumulate(HashMap<uint64_t, FrameStages> &r_frames, const String &p_name, double p_gpu_ns) {
240 Pass pass = PASS_COUNT;
241 bool is_begin = false;
242 uint64_t serial = 0;
243 if (!parse_label(p_name, pass, is_begin, serial)) {
244 return;
245 }
246 FrameStages *entry = r_frames.getptr(serial);
247 if (!entry) {
248 r_frames.insert(serial, FrameStages());
249 entry = r_frames.getptr(serial);
250 }
251 StageTimes &stage = entry->stages[pass];
252 if (is_begin) {
253 stage.begin_ns = p_gpu_ns;
254 } else {
255 stage.end_ns = p_gpu_ns;
256 }
257}

259// The newest frame serial that has at least one measured pass, with every pass measured in THAT frame.
260// Serial 0 is never a frame (no allocated serial is 0); markers labelled 0 come from a caller that had
261// no serial and are ignored instead of being read as some frame's timing.
262struct FrameDurations {
263 bool has_data = false;
264 uint64_t serial = 0;
265 double ms[PASS_COUNT] = {};
266 bool valid[PASS_COUNT] = {};
267
268 uint32_t pipeline_mask() const {
269 uint32_t mask = 0;
270 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
271 if (valid[i]) {
272 mask |= 1u << i;
273 }
274 }
275 return mask;
276 }
277 double pipeline_sum_ms() const {
278 double sum = 0.0;
279 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
280 if (valid[i]) {
281 sum += ms[i];
282 }
283 }
284 return sum;
285 }
286};

288// Passes the tile renderer itself brackets (plus its GaussianSplat_ frame marker): the legacy
289// gpu_tile_* fields keep selecting their frame among these only, exactly as before BV-04b.
290static constexpr uint32_t TILE_PASSES_MASK = (1u << PASS_OVERLAP_COUNT) | (1u << PASS_PREFIX) | (1u << PASS_BINNING) |
291 (1u << PASS_OVERLAP_SORT) | (1u << PASS_RASTER) | (1u << PASS_RESOLVE) | (1u << PASS_TOTAL);
292static constexpr uint32_t ANY_PASS_MASK = (1u << PASS_COUNT) - 1u;
293
294// p_select_mask: a frame qualifies only if at least one pass in the mask was measured in it.
295inline FrameDurations latest_frame(const HashMap<uint64_t, FrameStages> &p_frames, uint32_t p_select_mask = ANY_PASS_MASK) {
296 FrameDurations out;
297 for (const KeyValue<uint64_t, FrameStages> &kv : p_frames) {
298 if (kv.key == 0) {
299 continue;
300 }
301 bool any = false;
302 for (uint32_t i = 0; i < PASS_COUNT; i++) {
303 any = any || (((p_select_mask >> i) & 1u) != 0u && kv.value.stages[i].is_ordered());
304 }
305 if (!any || (out.has_data && kv.key <= out.serial)) {
306 continue;
307 }
308 out = FrameDurations();
309 out.has_data = true;
310 out.serial = kv.key;
311 for (uint32_t i = 0; i < PASS_COUNT; i++) {
312 const StageTimes &stage = kv.value.stages[i];
313 out.valid[i] = stage.is_ordered();
314 out.ms[i] = out.valid[i] ? (stage.end_ns - stage.begin_ns) / 1e6 : 0.0;
315 }
316 }
317 return out;
318}

320// ---------------------------------------------------------------------------------------------
321// Published per-pass state.
322//
323// Per-pass values are "sticky": the mid-frame flushes of RenderingDevice discard some markers on most
324// frames, so a pass keeps its last measured value until it has not been measured for STALE_RESOLVES
325// resolves (then it reads invalid and 0, never a frozen value). Because sticky values of different
326// passes can come from different frames, every pass also publishes the frame serial it was measured
327// in, and the snapshot carries the newest COHERENT frame: the set of passes measured in one and the
328// same frame serial, and their sum. A consumer that needs a same-frame breakdown (the fixed cost F)
329// reads the coherent fields, or compares the per-pass serials; it never adds sticky values blindly.
330struct PassTimingSnapshot {
331 float ms[PIPELINE_PASS_COUNT] = {};
332 bool valid[PIPELINE_PASS_COUNT] = {};
333 // Frame serial the published value was measured in. 0 while the pass is invalid.
334 uint64_t frame_serial[PIPELINE_PASS_COUNT] = {};
335 // Newest frame with at least one measured pass. 0 = none (or aged out).
336 uint64_t coherent_frame_serial = 0;
337 // Bit i set = pass i was measured in coherent_frame_serial.
338 uint32_t coherent_pass_mask = 0;
339 // Sum of the passes in coherent_pass_mask, all from coherent_frame_serial.
340 float coherent_total_ms = 0.0f;
341 // Resolves since the coherent frame was last refreshed (0 = refreshed by the latest resolve).
342 uint64_t coherent_age_resolves = 0;
343 // Monotonic resolve counter (every resolve call, data or not).
344 uint64_t resolve_count = 0;
345 // Freshness: OS::get_ticks_usec() at the last resolve, and at the resolve that last refreshed the
346 // coherent frame. Resolves run only while the Gaussian pipeline runs, so when nothing is drawn (e.g.
347 // the node is outside the view) every value above stays as it was; a consumer compares these with
348 // Time.get_ticks_usec() to tell a fresh sample from a held one.
349 uint64_t last_resolve_usec = 0;
350 uint64_t coherent_resolve_usec = 0;
351
352 bool all_bv04b_passes_coherent() const { return (coherent_pass_mask & BV04B_PASSES_MASK) == BV04B_PASSES_MASK; }
353};

355class PassTimingAccumulator {
356public:
357 static constexpr uint64_t STALE_RESOLVES = 120; // ~2 s at 60 fps; same bound as the tile passes
358
359private:
360 PassTimingSnapshot snapshot;
361 uint64_t resolved_at[PIPELINE_PASS_COUNT] = {};
362 uint64_t coherent_resolved_at = 0;
363
364public:
365 void reset() {
366 snapshot = PassTimingSnapshot();
367 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
368 resolved_at[i] = 0;
369 }
370 coherent_resolved_at = 0;
371 }
372
373 // Called once per resolve, with or without fresh data, so stale values age out on an empty view.
374 void ingest(const FrameDurations &p_durations, uint64_t p_now_usec = 0) {
375 const uint64_t now = ++snapshot.resolve_count;
376 snapshot.last_resolve_usec = p_now_usec;
377 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
378 if (p_durations.has_data && p_durations.valid[i]) {
379 snapshot.ms[i] = float(p_durations.ms[i]);
380 snapshot.valid[i] = true;
381 snapshot.frame_serial[i] = p_durations.serial;
382 resolved_at[i] = now;
383 } else if (snapshot.valid[i] && now - resolved_at[i] > STALE_RESOLVES) {
384 snapshot.valid[i] = false;
385 snapshot.ms[i] = 0.0f;
386 snapshot.frame_serial[i] = 0;
387 }
388 }
389 if (p_durations.has_data && p_durations.pipeline_mask() != 0u) {
390 snapshot.coherent_frame_serial = p_durations.serial;
391 snapshot.coherent_pass_mask = p_durations.pipeline_mask();
392 snapshot.coherent_total_ms = float(p_durations.pipeline_sum_ms());
393 snapshot.coherent_resolve_usec = p_now_usec;
394 coherent_resolved_at = now;
395 } else if (snapshot.coherent_pass_mask != 0u && now - coherent_resolved_at > STALE_RESOLVES) {
396 snapshot.coherent_frame_serial = 0;
397 snapshot.coherent_pass_mask = 0;
398 snapshot.coherent_total_ms = 0.0f;
399 snapshot.coherent_resolve_usec = 0;
400 }
401 snapshot.coherent_age_resolves = snapshot.coherent_pass_mask != 0u ? now - coherent_resolved_at : 0u;
402 }
403
404 const PassTimingSnapshot &get_snapshot() const { return snapshot; }
405};

407// Telemetry surface (render stats and streaming_state). Flat keys for the four BV-04b passes, and a
408// nested "gpu_pass_timing" dictionary with every pass:
409// passes.<name> = { time_ms, valid, frame_serial }
410// coherent_frame_serial, coherent_passes (names), coherent_pass_mask, coherent_total_ms,
411// coherent_age_resolves, coherent_has_all_bv04b_passes, resolve_count,
412// last_resolve_usec, coherent_resolve_usec
413// A pass that was not measured reads valid = false and time_ms = 0; nothing is inferred from a 0.
414inline void append_to_dictionary(Dictionary &r_metrics, const PassTimingSnapshot &p_snapshot) {
415 static const Pass flat_passes[] = { PASS_CULL, PASS_PROJECTION, PASS_PRESORT, PASS_COMPOSITE };
416 for (const Pass pass : flat_passes) {
417 const String base = String("gpu_") + pass_name(pass);
418 r_metrics[base + "_time_ms"] = p_snapshot.ms[pass];
419 r_metrics[base + "_time_valid"] = p_snapshot.valid[pass];
420 r_metrics[base + "_frame_serial"] = static_cast<int64_t>(p_snapshot.frame_serial[pass]);
421 }
422 Dictionary timing;
423 Dictionary passes;
424 Array coherent_passes;
425 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
426 Dictionary entry;
427 entry["time_ms"] = p_snapshot.ms[i];
428 entry["valid"] = p_snapshot.valid[i];
429 entry["frame_serial"] = static_cast<int64_t>(p_snapshot.frame_serial[i]);
430 passes[pass_name(Pass(i))] = entry;
431 if ((p_snapshot.coherent_pass_mask >> i) & 1u) {
432 coherent_passes.push_back(pass_name(Pass(i)));
433 }
434 }
435 timing["passes"] = passes;
436 timing["coherent_frame_serial"] = static_cast<int64_t>(p_snapshot.coherent_frame_serial);
437 timing["coherent_passes"] = coherent_passes;
438 timing["coherent_pass_mask"] = static_cast<int64_t>(p_snapshot.coherent_pass_mask);
439 timing["coherent_total_ms"] = p_snapshot.coherent_total_ms;
440 timing["coherent_age_resolves"] = static_cast<int64_t>(p_snapshot.coherent_age_resolves);
441 timing["coherent_has_all_bv04b_passes"] = p_snapshot.all_bv04b_passes_coherent();
442 timing["resolve_count"] = static_cast<int64_t>(p_snapshot.resolve_count);
443 timing["last_resolve_usec"] = static_cast<int64_t>(p_snapshot.last_resolve_usec);
444 timing["coherent_resolve_usec"] = static_cast<int64_t>(p_snapshot.coherent_resolve_usec);
445 r_metrics["gpu_pass_timing"] = timing;
446}
