16// BV-04b (ADR adr-beyond-vram-worlds.md D-4 / D-5): GPU timing completeness.
17//
18// Every GPU pass of a Gaussian frame is bracketed by a pair of RenderingDevice timestamps named
19// "<prefix><frame serial>_Begin" / "_End". The tile renderer parses the captured names back into a
20// per-frame-serial table and publishes one duration per pass. Before BV-04b only the tile passes had
21// timestamps; cull, projection (the depth-key pass), the depth pre-sort and the composite were
22// untimed, so the fixed cost F of the frame model could not be measured.
23//
24// This header holds the label vocabulary, the frame-serial source shared by every pass of one frame,
25// and the pure parse / reduce functions (device free, so they are unit tested on the CPU).
26namespace GaussianSplatting {

41// The order is the order of execution within a frame. PASS_TOTAL is the legacy GaussianSplat_ marker.
42enum Pass : uint32_t {
43 PASS_CULL = 0, // frustum / instance-chunk cull (gpu_culler)
44 PASS_PROJECTION, // depth-key projection: instance chunk dispatch + depth compute + count clamp
45 PASS_PRESORT, // depth pre-sort of the visible splats (GPUSortingPipeline's sorter)
46 PASS_OVERLAP_COUNT, // tile_binning COUNT pass
47 PASS_PREFIX, // tile prefix scan
48 PASS_BINNING, // tile_binning EMIT pass (the "overlap emit" pass)
49 PASS_OVERLAP_SORT, // overlap-record radix sort
50 PASS_RASTER, // tile raster (compute or fragment)
51 PASS_RESOLVE, // tile resolve
52 PASS_COMPOSITE, // output composite into the engine frame (compute path)
53 PASS_TOTAL, // GaussianSplat_ frame bracket (flush-discarded on most frames; not summed)
54 PASS_COUNT,
55};
56
57// Passes that make up one frame's pipeline (everything except PASS_TOTAL).
58static constexpr uint32_t PIPELINE_PASS_COUNT = PASS_TOTAL;
59static constexpr uint32_t ALL_PIPELINE_PASSES_MASK = (1u << PIPELINE_PASS_COUNT) - 1u;
60// The four passes BV-04b adds.
61static constexpr uint32_t BV04B_PASSES_MASK = (1u << PASS_CULL) | (1u << PASS_PROJECTION) | (1u << PASS_PRESORT) |
62 (1u << PASS_COMPOSITE);
63// Fixed minimum for a drawn HLOD frame (TileRenderer's global count/emit/sort/raster
64// route). Prefix may use the CPU emergency fallback; resolve may use raw-input debug

66static constexpr uint32_t HLOD_REQUIRED_PASSES_MASK = (1u << PASS_PROJECTION) | (1u << PASS_PRESORT) |

129// Frame serial. One serial names every pass of one pipeline pass (cull .. composite), so the parser
130// can tell which durations were measured in the same frame. RenderPipelineStages::prepare_frame_context
131// allocates it; the tile renderer takes it through RenderParams::frame_serial. Render thread only;
132// atomic so a stray caller on another thread cannot tear it.
133inline std::atomic<uint64_t> &_frame_serial_counter() {
134 static std::atomic<uint64_t> counter{ 0 };
135 return counter;
136}
137inline std::atomic<uint64_t> &_current_frame_serial() {
138 static std::atomic<uint64_t> current{ 0 };
139 return current;
140}
141
142// Allocates a fresh serial (never 0) and makes it the current one.
143inline uint64_t allocate_frame_serial() {
144 const uint64_t serial = _frame_serial_counter().fetch_add(1, std::memory_order_relaxed) + 1;
145 _current_frame_serial().store(serial, std::memory_order_relaxed);
146 return serial;
147}
148
149// The serial of the pipeline pass being recorded (0 = none allocated yet).
150inline uint64_t current_frame_serial() {
151 return _current_frame_serial().load(std::memory_order_relaxed);
152}

154// A caller-supplied serial (e.g. the painterly path's frame counter) was used: never hand it out again.
155inline void note_external_frame_serial(uint64_t p_serial) {
156 uint64_t seen = _frame_serial_counter().load(std::memory_order_relaxed);
157 while (seen < p_serial && !_frame_serial_counter().compare_exchange_weak(seen, p_serial, std::memory_order_relaxed)) {
158 }
159 _current_frame_serial().store(p_serial, std::memory_order_relaxed);
160}

179struct FrameDispatchRecord {
180 uint64_t serial = 0;
181 FrameRoute route = ROUTE_UNKNOWN;
182 uint32_t dispatched_pass_mask = 0;
183
184 bool is_coherent(uint64_t p_serial, uint32_t p_measured_mask) const {
185 const uint32_t required = required_pass_mask(route);
186 return p_serial != 0 && serial == p_serial && required != 0 &&
187 (dispatched_pass_mask & ~ALL_PIPELINE_PASSES_MASK) == 0 &&
188 (route != ROUTE_HLOD_FUSED || (dispatched_pass_mask & (1u << PASS_CULL)) == 0) &&
189 (dispatched_pass_mask & required) == required && p_measured_mask == dispatched_pass_mask;
190 }
191};

193// CPU submission receipts, independent of timestamp capture/validity. Old receipts
194// are evicted rather than guessed. Device instance IDs distinguish device lifetimes.
195class FrameDispatchRecords {
196public:
197 static constexpr uint32_t CAPACITY = 256;
198
199private:
200 struct Entry {
201 uint64_t device_id = 0;
202 FrameDispatchRecord record;
203 };
204 Entry entries[CAPACITY];
205
206public:
207 void begin_frame(uint64_t p_serial, FrameRoute p_route, uint64_t p_device_id = 0) {
208 if (p_serial == 0) {
209 return;
210 }
211 Entry &entry = entries[p_serial % CAPACITY];
212 if (entry.record.serial == p_serial && entry.device_id == p_device_id) {
213 if (entry.record.route != p_route) {
214 entry.record.route = ROUTE_UNKNOWN; // Conflicting declarations fail closed.
215 }
216 return;
217 }
218 entry.device_id = p_device_id;
219 entry.record = FrameDispatchRecord();
220 entry.record.serial = p_serial;
221 entry.record.route = p_route;
222 }
223 void record_dispatch(uint64_t p_serial, Pass p_pass, uint64_t p_device_id = 0) {
224 if (p_serial == 0 || p_pass >= PASS_TOTAL) {
225 return;
226 }
227 Entry &entry = entries[p_serial % CAPACITY];
228 if (entry.record.serial == p_serial && entry.device_id == p_device_id) {
229 entry.record.dispatched_pass_mask |= 1u << p_pass;
230 }
231 }
232 FrameDispatchRecord get(uint64_t p_serial, uint64_t p_device_id = 0) const {
233 const Entry &entry = entries[p_serial % CAPACITY];
234 return p_serial != 0 && entry.record.serial == p_serial && entry.device_id == p_device_id
235 ? entry.record : FrameDispatchRecord();
236 }
237};

275// Brackets one pass with "<prefix><serial>_Begin" ... "_End" on p_device. The _End marker is written
276// only by finish(), after the pass's last compute list has ended; a pass that bails out early leaves an
277// unpaired _Begin, which the parser treats as "not measured" (never as a 0 ms sample).
278// RenderingDevice refuses capture_timestamp inside a compute list with dispatches, so construct this
279// before compute_list_begin() and call finish() after compute_list_end().
280class ScopedPassTimestamp {
281 RenderingDevice *device = nullptr;
282 String label;
283
284public:
285 ScopedPassTimestamp(RenderingDevice *p_device, Pass p_pass, bool p_enabled, uint64_t p_serial = current_frame_serial()) {
286 if (!p_device || !p_enabled || p_serial == 0) {
287 return;
288 }
289 device = p_device;
290 label = make_label(p_pass, p_serial);
291 device->capture_timestamp(label + String("_Begin"));
292 }
293 void finish() {
294 if (device) {
295 device->capture_timestamp(label + String("_End"));
296 device = nullptr;
297 }
298 }
299 bool is_armed() const { return device != nullptr; }
300
301 ScopedPassTimestamp(const ScopedPassTimestamp &) = delete;
302 ScopedPassTimestamp &operator=(const ScopedPassTimestamp &) = delete;
303};

308struct StageTimes {
309 double begin_ns = -1.0;
310 double end_ns = -1.0;
311 bool is_complete() const { return begin_ns >= 0.0 && end_ns >= 0.0; }
312 // A pass is measured only when both markers exist and are ordered.
313 bool is_ordered() const { return is_complete() && end_ns >= begin_ns; }
314};

316struct FrameStages {
317 StageTimes stages[PASS_COUNT];
318};

320// Parses "<prefix><serial>_Begin|_End". Returns false for any name that is not a pass marker.
321inline bool parse_label(const String &p_name, Pass &r_pass, bool &r_is_begin, uint64_t &r_serial) {
322 bool is_begin = false;
323 int64_t suffix_pos = -1;
324 if (p_name.ends_with("_Begin")) {
325 is_begin = true;
326 suffix_pos = p_name.length() - 6;
327 } else if (p_name.ends_with("_End")) {
328 suffix_pos = p_name.length() - 4;
329 } else {
330 return false;
331 }
332 for (uint32_t i = 0; i < PASS_COUNT; i++) {
333 const Pass pass = Pass(i);
334 const String prefix = label_prefix(pass);
335 if (!p_name.begins_with(prefix)) {
336 continue;
337 }
338 const int64_t prefix_length = prefix.length();
339 if (suffix_pos <= prefix_length) {
340 return false;
341 }
342 const String serial_str = p_name.substr(prefix_length, suffix_pos - prefix_length);
343 if (!serial_str.is_valid_int()) {
344 return false;
345 }
346 r_pass = pass;
347 r_is_begin = is_begin;
348 r_serial = uint64_t(serial_str.to_int());
349 return true;
350 }
351 return false;
352}

354inline void accumulate(HashMap<uint64_t, FrameStages> &r_frames, const String &p_name, double p_gpu_ns) {
355 Pass pass = PASS_COUNT;
356 bool is_begin = false;
357 uint64_t serial = 0;
358 if (!parse_label(p_name, pass, is_begin, serial)) {
359 return;
360 }
361 FrameStages *entry = r_frames.getptr(serial);
362 if (!entry) {
363 r_frames.insert(serial, FrameStages());
364 entry = r_frames.getptr(serial);
365 }
366 StageTimes &stage = entry->stages[pass];
367 if (is_begin) {
368 stage.begin_ns = p_gpu_ns;
369 } else {
370 stage.end_ns = p_gpu_ns;
371 }
372}

374// The newest frame serial that has at least one measured pass, with every pass measured in THAT frame.
375// Serial 0 is never a frame (no allocated serial is 0); markers labelled 0 come from a caller that had
376// no serial and are ignored instead of being read as some frame's timing.
377struct FrameDurations {
378 bool has_data = false;
379 uint64_t serial = 0;
380 double ms[PASS_COUNT] = {};
381 bool valid[PASS_COUNT] = {};
382
383 uint32_t pipeline_mask() const {
384 uint32_t mask = 0;
385 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
386 if (valid[i]) {
387 mask |= 1u << i;
388 }
389 }
390 return mask;
391 }
392 double pipeline_sum_ms() const {
393 double sum = 0.0;
394 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
395 if (valid[i]) {
396 sum += ms[i];
397 }
398 }
399 return sum;
400 }
401};
402
403// Passes the tile renderer itself brackets (plus its GaussianSplat_ frame marker): the legacy
404// gpu_tile_* fields keep selecting their frame among these only, exactly as before BV-04b.
405static constexpr uint32_t TILE_PASSES_MASK = (1u << PASS_OVERLAP_COUNT) | (1u << PASS_PREFIX) | (1u << PASS_BINNING) |
406 (1u << PASS_OVERLAP_SORT) | (1u << PASS_RASTER) | (1u << PASS_RESOLVE) | (1u << PASS_TOTAL);
407static constexpr uint32_t ANY_PASS_MASK = (1u << PASS_COUNT) - 1u;
408
409// p_select_mask: a frame qualifies only if at least one pass in the mask was measured in it.
410inline FrameDurations latest_frame(const HashMap<uint64_t, FrameStages> &p_frames, uint32_t p_select_mask = ANY_PASS_MASK) {
411 FrameDurations out;
412 for (const KeyValue<uint64_t, FrameStages> &kv : p_frames) {
413 if (kv.key == 0) {
414 continue;
415 }
416 bool any = false;
417 for (uint32_t i = 0; i < PASS_COUNT; i++) {
418 any = any || (((p_select_mask >> i) & 1u) != 0u && kv.value.stages[i].is_ordered());
419 }
420 if (!any || (out.has_data && kv.key <= out.serial)) {
421 continue;
422 }
423 out = FrameDurations();
424 out.has_data = true;
425 out.serial = kv.key;
426 for (uint32_t i = 0; i < PASS_COUNT; i++) {
427 const StageTimes &stage = kv.value.stages[i];
428 out.valid[i] = stage.is_ordered();
429 out.ms[i] = out.valid[i] ? (stage.end_ns - stage.begin_ns) / 1e6 : 0.0;
430 }
431 }
432 return out;
433}

435// ---------------------------------------------------------------------------------------------
436// Published per-pass state.
437//
438// Per-pass values are "sticky": the mid-frame flushes of RenderingDevice discard some markers on most
439// frames, so a pass keeps its last measured value until it has not been measured for STALE_RESOLVES
440// resolves (then it reads invalid and 0, never a frozen value). Because sticky values of different
441// passes can come from different frames, every pass also publishes the frame serial it was measured
442// in, and the snapshot carries the newest COHERENT frame: the set of passes measured in one and the
443// same frame serial, and their sum. A consumer that needs a same-frame breakdown (the fixed cost F)
444// reads the coherent fields, or compares the per-pass serials; it never adds sticky values blindly.
445struct PassTimingSnapshot {
446 float ms[PIPELINE_PASS_COUNT] = {};
447 bool valid[PIPELINE_PASS_COUNT] = {};
448 // Frame serial the published value was measured in. 0 while the pass is invalid.
449 uint64_t frame_serial[PIPELINE_PASS_COUNT] = {};
450 // Newest frame with at least one measured pass. 0 = none (or aged out).
451 uint64_t coherent_frame_serial = 0;
452 // Bit i set = pass i was measured in coherent_frame_serial.
453 uint32_t coherent_pass_mask = 0;
454 FrameDispatchRecord coherent_dispatch;
455 // Sum of the passes in coherent_pass_mask, all from coherent_frame_serial.
456 float coherent_total_ms = 0.0f;
457 // Resolves since the coherent frame was last refreshed (0 = refreshed by the latest resolve).
458 uint64_t coherent_age_resolves = 0;
459 // Monotonic resolve counter (every resolve call, data or not).
460 uint64_t resolve_count = 0;
461 // Freshness: OS::get_ticks_usec() at the last resolve, and at the resolve that last refreshed the
462 // coherent frame. Resolves run only while the Gaussian pipeline runs, so when nothing is drawn (e.g.
463 // the node is outside the view) every value above stays as it was; a consumer compares these with
464 // Time.get_ticks_usec() to tell a fresh sample from a held one.
465 uint64_t last_resolve_usec = 0;
466 uint64_t coherent_resolve_usec = 0;
467
468 bool all_bv04b_passes_coherent() const { return (coherent_pass_mask & BV04B_PASSES_MASK) == BV04B_PASSES_MASK; }
469 bool all_route_passes_coherent() const { return coherent_dispatch.is_coherent(coherent_frame_serial, coherent_pass_mask); }
470};

472class PassTimingAccumulator {
473public:
474 static constexpr uint64_t STALE_RESOLVES = 120; // ~2 s at 60 fps; same bound as the tile passes
475
476private:
477 PassTimingSnapshot snapshot;
478 uint64_t resolved_at[PIPELINE_PASS_COUNT] = {};
479 uint64_t coherent_resolved_at = 0;
480
481public:
482 void reset() {
483 snapshot = PassTimingSnapshot();
484 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
485 resolved_at[i] = 0;
486 }
487 coherent_resolved_at = 0;
488 }
489
490 // Called once per resolve, with or without fresh data, so stale values age out on an empty view.
491 void ingest(const FrameDurations &p_durations, uint64_t p_now_usec = 0,
492 const FrameDispatchRecord &p_dispatch = FrameDispatchRecord()) {
493 const uint64_t now = ++snapshot.resolve_count;
494 snapshot.last_resolve_usec = p_now_usec;
495 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
496 if (p_durations.has_data && p_durations.valid[i]) {
497 snapshot.ms[i] = float(p_durations.ms[i]);
498 snapshot.valid[i] = true;
499 snapshot.frame_serial[i] = p_durations.serial;
500 resolved_at[i] = now;
501 } else if (snapshot.valid[i] && now - resolved_at[i] > STALE_RESOLVES) {
502 snapshot.valid[i] = false;
503 snapshot.ms[i] = 0.0f;
504 snapshot.frame_serial[i] = 0;
505 }
506 }
507 if (p_durations.has_data && p_durations.pipeline_mask() != 0u) {
508 snapshot.coherent_frame_serial = p_durations.serial;
509 snapshot.coherent_pass_mask = p_durations.pipeline_mask();
510 snapshot.coherent_dispatch = p_dispatch.serial == p_durations.serial ? p_dispatch : FrameDispatchRecord();
511 snapshot.coherent_total_ms = float(p_durations.pipeline_sum_ms());
512 snapshot.coherent_resolve_usec = p_now_usec;
513 coherent_resolved_at = now;
514 } else if (snapshot.coherent_pass_mask != 0u && now - coherent_resolved_at > STALE_RESOLVES) {
515 snapshot.coherent_frame_serial = 0;
516 snapshot.coherent_pass_mask = 0;
517 snapshot.coherent_dispatch = FrameDispatchRecord();
518 snapshot.coherent_total_ms = 0.0f;
519 snapshot.coherent_resolve_usec = 0;
520 }
521 snapshot.coherent_age_resolves = snapshot.coherent_pass_mask != 0u ? now - coherent_resolved_at : 0u;
522 }
523
524 const PassTimingSnapshot &get_snapshot() const { return snapshot; }
525};

527// Telemetry surface (render stats and streaming_state). Flat keys for the four BV-04b passes, and a
528// nested "gpu_pass_timing" dictionary with every pass:
529// passes.<name> = { time_ms, valid, frame_serial }
530// coherent_frame_serial, coherent_passes (names), coherent_pass_mask, coherent_total_ms,
531// coherent_age_resolves, coherent_has_all_bv04b_passes, resolve_count,
532// last_resolve_usec, coherent_resolve_usec
533// A pass that was not measured reads valid = false and time_ms = 0; nothing is inferred from a 0.
534inline void append_to_dictionary(Dictionary &r_metrics, const PassTimingSnapshot &p_snapshot) {
535 static const Pass flat_passes[] = { PASS_CULL, PASS_PROJECTION, PASS_PRESORT, PASS_COMPOSITE };
536 for (const Pass pass : flat_passes) {
537 const String base = String("gpu_") + pass_name(pass);
538 r_metrics[base + "_time_ms"] = p_snapshot.ms[pass];
539 r_metrics[base + "_time_valid"] = p_snapshot.valid[pass];
540 r_metrics[base + "_frame_serial"] = static_cast<int64_t>(p_snapshot.frame_serial[pass]);
541 }
542 Dictionary timing;
543 Dictionary passes;
544 Array coherent_passes;
545 for (uint32_t i = 0; i < PIPELINE_PASS_COUNT; i++) {
546 Dictionary entry;
547 entry["time_ms"] = p_snapshot.ms[i];
548 entry["valid"] = p_snapshot.valid[i];
549 entry["frame_serial"] = static_cast<int64_t>(p_snapshot.frame_serial[i]);
550 passes[pass_name(Pass(i))] = entry;
551 if ((p_snapshot.coherent_pass_mask >> i) & 1u) {
552 coherent_passes.push_back(pass_name(Pass(i)));
553 }
554 }
555 timing["passes"] = passes;
556 timing["coherent_frame_serial"] = static_cast<int64_t>(p_snapshot.coherent_frame_serial);
557 timing["coherent_passes"] = coherent_passes;
558 timing["coherent_pass_mask"] = static_cast<int64_t>(p_snapshot.coherent_pass_mask);
