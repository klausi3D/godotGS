11// W-S global bound, in scene units. Effectors have independent additive displacement;
12// their push is the SUM of absolute enabled positional strengths, including overlapping
13// spheres. The effective base is at least one for effectors/forced rows, preserving the
14// existing zero-project-strength override while bounding the additive term.

16// RenderingServer grows the LOCAL AABB before transforming it. The transformed cube's support on world axis i is
17// local_margin * sum(abs(basis[i])); choose the smallest support so every world axis gains at least D_max.
18inline real_t local_visibility_margin(double p_scene_margin, const Basis &p_basis) {
19 if (p_scene_margin <= 0.0) { return 0.0; }
20 double support = std::numeric_limits<double>::infinity();
21 for (int axis = 0; axis < 3; ++axis) {
22 const Vector3 &row = p_basis[axis];
23 support = MIN(support, std::abs(double(row.x)) + std::abs(double(row.y)) + std::abs(double(row.z)));
24 }
25 ERR_FAIL_COND_V_MSG(!std::isfinite(p_scene_margin) || !std::isfinite(support) || support <= 0.0, 0.0,
26 "Wind visibility bounds require finite displacement and nonzero support on every instance axis.");
27 const double local = p_scene_margin / support;
28 ERR_FAIL_COND_V_MSG(local > std::numeric_limits<real_t>::max(), 0.0, "Wind visibility margin exceeds coordinate precision.");
29 real_t rounded = real_t(local);
30 if (double(rounded) < local) { rounded = std::nextafter(rounded, std::numeric_limits<real_t>::infinity()); }
31 return rounded;
32}

34inline double displacement_bound(double p_strength, double p_intensity, double p_push) {
35 return MAX(0.0, p_strength) * MAX(0.0, p_intensity) * (1.0 + MAX(0.0, p_push));

44// Reusable shadow-refresh predicate. Pass the corresponding FrameRecord::row_casts bit.
45// This means nonzero wind amplitude on a casting row; it does not assert changing time
46// (time_scale/frequency may be zero), evaluate distance/level fade, or include effectors.
47inline bool wind_active_on_casting_row(const gs::settings::GSWindSettings &p_settings,
48 const InstanceDataGPU &p_row, bool p_casts) {
49 if (!p_casts || !row_wind_enabled(p_settings, p_row) || !(p_row.params[2] > 0.0f)) { return false; }
50 const bool forced = uint32_t(p_row.params[3]) == GS_INSTANCE_WIND_MODE_FORCE_ENABLED;
51 if (!(p_settings.strength > 0.0f) && !forced) { return false; }
52 const Vector3 override_direction(p_row.wind_params[0], p_row.wind_params[1], p_row.wind_params[2]);
53 const Vector3 direction = override_direction.length_squared() > 1.0e-8f ? override_direction : p_settings.direction;
54 return direction.length_squared() > 1.0e-12f;
55}

63inline double frame_displacement_bound(const gs::settings::GSWindSettings &p_settings,
64 const InstanceDataGPU *p_rows, uint32_t p_count, double p_effector_push, bool p_project_effector = false) {
65 double strength = MAX(0.0, double(p_settings.strength));
66 double intensity = 0.0;
67 for (uint32_t i = 0; i < p_count; ++i) {
68 const InstanceDataGPU &row = p_rows[i];
69 if (row_wind_enabled(p_settings, row)) {
70 intensity = MAX(intensity, MAX(0.0, double(row.params[2])));
71 if (uint32_t(row.params[3]) == GS_INSTANCE_WIND_MODE_FORCE_ENABLED) { strength = MAX(strength, 1.0); }
72 }
73 if (p_effector_push > 0.0 && (row_has_effectors(row) || (p_project_effector && row.effect_params[0] > 0.0f))) {
74 strength = MAX(strength, 1.0);
75 intensity = MAX(intensity, double(row.effect_params[0]));
76 }
77 }
78 return displacement_bound(strength, intensity, p_effector_push);
79}
