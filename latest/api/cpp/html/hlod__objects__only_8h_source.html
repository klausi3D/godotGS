8// The objects-only HLOD bundle: the P0-absent mode of the one route (docs/architecture/adr-one-route-many-producers.md
9// section 12.7, PR-1). A frame whose renderer has node instances (rows 1..N) but no world cut (P0) - no world at all, or
10// a world without an HLOD snapshot - is drawn on the HLOD route with an empty row 0: no snapshot, no world node, no
11// ViewSet, and no world floor. Pure, so the routing matrix is a CPU doctest; the publisher reads the inputs and applies
12// the plan (GaussianSplatRenderer::try_publish_hlod_resident_contract()).
13//
14// - Owner decision OD-2 (2026-11-27): the opt-out rendering/gaussian_splatting/lod/hlod_runtime_cut_enabled = false does
15// not keep an objects-only bundle off the route. It only names the cause of a dropped world.
16// - A world submission without a snapshot next to node instances is dropped explicitly: the canonical routes never drew
17// it next to a node either (the world row is emitted only without node instances, director build_instance_buffer_for_renderer;
18// measured on a GPU before PR-1). The plan names why (`world_reason`); the renderer reports it as the render stat
19// hlod_world_producer_reason and counts the frames in hlod_world_producer_dropped_frames.
20// - Without node instances nothing changes: a world without a snapshot keeps its route and draws.

53inline Plan plan(const Inputs &p_in) {
54 Plan out;
55 if (p_in.world_route || p_in.node_rows == 0) {
56 return out;
57 }
58 out.objects_only = true;
59 if (!p_in.world_submission) {
60 return out;
61 }
62 String cause = p_in.submission_reason;
63 if (cause.is_empty()) {
64 if (!p_in.world_renderable) {
65 cause = CAUSE_NO_RENDERABLE_PAYLOAD;
66 } else if (!p_in.cut_enabled) {
67 cause = CAUSE_RUNTIME_CUT_DISABLED;
68 } else if (p_in.snapshot) {
69 cause = CAUSE_SUBMISSION_NOT_OWNED;
70 } else {
71 cause = CAUSE_NO_RUNTIME_TREE;
72 }
73 }
74 out.world_reason = String(WORLD_WITHOUT_SNAPSHOT_PREFIX) + cause;
75 return out;
76}
