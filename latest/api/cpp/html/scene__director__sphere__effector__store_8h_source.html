33// #610 S4: SphereEffectorStore owns the per-scenario sphere-effector
34// records, the effector->slot lookup, the generation counter and the
35// monotonic registration serial. It is the boundary the prune predicate
36// (_world_has_no_sphere_effectors) and the sibling decomposition slices
37// see: callers touch only these public methods, never the fields -- that
38// is what lets S4/S5/S8 be extracted independently. Mirrors InstanceStore
39// (#610 S3).
40//
41// It takes NO lock of its own: every method is called under the director's
42// single world_mutex, preserving the one ThreadOwnedMutex ownership record
43// the renderer-contract-boundary guard depends on.
44//
45// D5 (OPEN owner decision, NOT resolved by this slice): the render-thread
46// payload builder (_build_sorted_sphere_effector_payload) revalidates each
47// record's cached scope_root_valid against ObjectDB and, when it flips,
48// writes the flag back through a const view (const_cast) and bumps the
49// generation. That write-through-const behavior is preserved verbatim here;
50// S4 is a pure move. records() hands out a const view and the builder keeps
51// its existing const_cast to mutate scope_root_valid in place.
52class GaussianSplatSceneDirector::SphereEffectorStore {
53public:
54 // --- queries ---
55 bool is_empty() const { return sphere_effectors.is_empty(); }
56 uint32_t count() const { return sphere_effectors.size(); }
57 bool has_effector(ObjectID p_effector_id) const { return sphere_effector_lookup.has(p_effector_id); }
58 uint64_t generation() const { return sphere_effector_generation; }
59
60 // Read-only view for the render-thread payload builder and the
61 // main-thread debug-state query. The builder mutates each record's
62 // cached scope_root_valid in place via const_cast (D5, above);
63 // membership never changes through this handle -- adds/removes go
64 // through append()/remove() so sphere_effector_lookup stays consistent.
65 const LocalVector<SphereEffectorRecord> &records() const { return sphere_effectors; }
66
67 // Mutable lookup of a single record for the in-place update path.
68 // Returns nullptr when the effector is absent.
69 SphereEffectorRecord *find_mutable(ObjectID p_effector_id) {
70 uint32_t *index_ptr = sphere_effector_lookup.getptr(p_effector_id);
71 if (!index_ptr || *index_ptr >= sphere_effectors.size()) {
72 return nullptr;
73 }
74 return &sphere_effectors[*index_ptr];
75 }
76
77 // --- mutations (keep sphere_effector_lookup consistent) ---
78 // Stamp the record with the next registration serial, record its lookup
79 // slot and append it. Mirrors the inline register path verbatim: the
80 // serial is assigned BEFORE the copy is stored.
81 void append(SphereEffectorRecord &p_record) {
82 p_record.registration_serial = ++sphere_effector_registration_serial;
83 sphere_effector_lookup[p_record.effector_id] = sphere_effectors.size();
84 sphere_effectors.push_back(p_record);
85 }
86 // Swap-remove the record for p_effector_id, fixing up the moved
87 // record's lookup slot. Returns false when the effector is absent
88 // (leaving the store untouched), matching the prior inline guard.
89 bool remove(ObjectID p_effector_id) {
90 uint32_t *index_ptr = sphere_effector_lookup.getptr(p_effector_id);
91 if (!index_ptr || *index_ptr >= sphere_effectors.size()) {
92 return false;
93 }
94 const uint32_t index = *index_ptr;
95 const uint32_t last_index = sphere_effectors.size() - 1;
96 if (index != last_index) {
97 sphere_effectors[index] = sphere_effectors[last_index];
98 sphere_effector_lookup[sphere_effectors[index].effector_id] = index;
99 }
100 sphere_effectors.remove_at(last_index);
101 sphere_effector_lookup.erase(p_effector_id);
102 return true;
103 }
104
105 // Saturating generation bump (skip 0, the "unset"/"no world" sentinel
106 // that get_sphere_effector_generation_for_renderer returns for a
107 // missing world). Defined in the .cpp so it reuses the file-local
108 // saturating-increment helper. #610 S4 (D2): the register/update/remove
109 // paths previously bumped through that saturating helper while the two
110 // render-thread D5 revalidation sites used a raw `++` (wrap-to-0 on
111 // uint64 overflow). All six now route through this one saturating
112 // method, so the two D5 sites wrap to 1 instead of 0. Because 0 is the
113 // reserved sentinel above, that is a deliberate consistency fix that
114 // closes a spurious-0-on-overflow hazard on those two sites -- NOT a
115 // 1:1 move -- and it is unreachable in practice (2^64 bumps). Firing
116 // conditions and order are otherwise unchanged. See the D2 note at the
117 // two sites in scene_director_sphere_effectors.cpp.
118 void bump_generation();
119
120 // Drop every effector and lookup slot. Used by
121 // teardown_world_for_scenario, which erases the whole world entry right
122 // after; the generation/serial counters are intentionally left
123 // untouched, matching the prior inline teardown.
124 void clear() {
125 sphere_effectors.clear();
126 sphere_effector_lookup.clear();
127 }
128
129private:
130 LocalVector<SphereEffectorRecord> sphere_effectors;
131 HashMap<ObjectID, uint32_t> sphere_effector_lookup;
132 uint64_t sphere_effector_generation = 1;
133 uint64_t sphere_effector_registration_serial = 0;
134};
