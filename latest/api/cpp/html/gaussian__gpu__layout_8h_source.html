30// C++ mirror of the GLSL decoder (host round-trip tests; the layout-sync guard pins both).
31inline Vector3 gs_decode_sh_snorm10(uint32_t p_packed, float p_scale) {
32 const auto extract = [](uint32_t p_word, uint32_t p_offset) -> int32_t {
33 const int32_t q = int32_t((p_word >> p_offset) & 0x3FFu);
34 return (q & 0x200) ? q - 0x400 : q; // sign-extend 10 bits
35 };
36 const float step = p_scale * (1.0f / float(GS_SH_SNORM10_MAX));
37 return Vector3(float(extract(p_packed, 0u)) * step,
38 float(extract(p_packed, 10u)) * step,
39 float(extract(p_packed, 20u)) * step);
40}

44inline uint32_t gs_pack_sh_metadata(uint32_t p_stored_first, uint32_t p_stored_high, uint32_t p_encoded_total,
45 GaussianDCEncoding p_dc_encoding, uint32_t p_sh_encoding) {
46 p_stored_first = MIN(p_stored_first, 0xFFu);
47 p_stored_high = MIN(p_stored_high, 0xFFu);
48 p_encoded_total = MIN(p_encoded_total, 0xFFu);
49 p_sh_encoding &= 0x7Fu;
50 const uint32_t dc_encoding_bit = p_dc_encoding == GAUSSIAN_DC_ENCODING_LINEAR_RGB ? GS_SH_METADATA_DC_ENCODING_MASK : 0u;
51 return (p_stored_first & GS_SH_METADATA_FIRST_ORDER_MASK) |
52 ((p_stored_high << 8u) & GS_SH_METADATA_HIGH_ORDER_MASK) |
53 ((p_encoded_total << 16u) & GS_SH_METADATA_ENCODED_COUNT_MASK) |
54 ((p_sh_encoding << 24u) & GS_SH_METADATA_ENCODING_MASK) |
55 dc_encoding_bit;
56}
