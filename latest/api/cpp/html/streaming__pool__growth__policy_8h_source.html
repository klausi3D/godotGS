4// BV-POOL: how far the streaming persistent buffer (the chunk-route atlas pool) grows, and
5// whether an allocation may be kept. Pure (no RenderingDevice), so the policy is testable on
6// the host; GaussianStreamingSystem::_try_grow_persistent_buffer_for_atlas_pressure() and
7// _grow_persistent_buffer() feed it.
8//
9// Why: a grow allocates the new buffer while the old one is still alive (the contents are
10// copied, and RenderingDevice::free() defers the release). Doubling to the 4 GiB cap therefore
11// peaked at old + new = 2.8 + 4.0 GB on a 10 GB card, and the allocator, which only *prefers*
12// device-local memory, put the new pool in host memory without an error: every atlas-reading
13// pass then ran over PCIe (run 42644237192: 9.4 -> 100 ms at the same pose).
14//
15// The policy:
16// 1. Grow straight to what the registered assets need (the same "asset pages + headroom"
17// sizing initialize() uses), never less than doubling. A world larger than the pool
18// reaches the ceiling in its first grow, while the old buffer is still small.
19// 2. Clamp to the growth ceiling and the VRAM budget, as before.
20// 3. Clamp to the device-local memory that is actually free (old + new must fit), when the
21// device reports it. Unknown free memory leaves the target unclamped.
22// 4. A known non-device-local allocation is never kept by a grow (the old pool stays and
23// growth stops). At initialization there is no older buffer to keep: it is reported loudly.

29// Safety margin left free on the device-local heap(s) after a grow's allocation.
30static constexpr uint64_t kDeviceFreeMarginBytes = uint64_t(256) * 1024 * 1024;
31// Without VK_EXT_memory_budget the usable share of the device-local heap is estimated like
32// the allocator does (VMA's default budget heuristic, 80 % of the heap).
33static constexpr uint64_t kEstimatedUsableHeapPercent = 80;

35struct GrowInputs {

45// Headroom on top of the registered asset pages, as initialize() sizes it.
46inline uint64_t needed_pages_with_headroom(uint64_t p_asset_pages, uint64_t p_pages_per_max_chunk) {
47 const uint64_t headroom = p_asset_pages / 4u > 2u * p_pages_per_max_chunk ? p_asset_pages / 4u : 2u * p_pages_per_max_chunk;
48 return p_asset_pages + headroom;
49}

51// Free device-local bytes. Prefers VK_EXT_memory_budget (budget - usage, which accounts for
52// other processes); otherwise estimates from the heap size and this process's allocator blocks.
53// Returns false when neither is available.
54inline bool device_free_bytes(bool p_heap_known, uint64_t p_heap_bytes, uint64_t p_allocator_block_bytes,
55 bool p_ext_known, uint64_t p_ext_budget_bytes, uint64_t p_ext_usage_bytes, uint64_t &r_free_bytes) {
56 r_free_bytes = 0;
57 if (p_ext_known) {
58 r_free_bytes = p_ext_budget_bytes > p_ext_usage_bytes ? p_ext_budget_bytes - p_ext_usage_bytes : 0;
59 return true;
60 }
61 if (p_heap_known && p_heap_bytes > 0) {
62 const uint64_t usable = p_heap_bytes / 100u * kEstimatedUsableHeapPercent;
63 r_free_bytes = usable > p_allocator_block_bytes ? usable - p_allocator_block_bytes : 0;
64 return true;
65 }
66 return false;
67}

69// The page count to grow to, or 0 when the pool must not grow.
70inline uint32_t compute_grow_target(const GrowInputs &p) {
71 if (p.current_pages == 0 || p.ceiling_pages <= p.current_pages) {
72 return 0;
73 }
74 uint64_t target = uint64_t(p.current_pages) * 2u;
75 if (p.needed_pages > target) {
76 target = p.needed_pages;
77 }
78 if (target > p.ceiling_pages) {
79 target = p.ceiling_pages;
80 }
81 if (target > p.budget_max_pages) {
82 target = p.budget_max_pages;
83 }
84 if (p.device_free_known && p.page_bytes > 0) {
85 const uint64_t room = p.device_free_bytes > kDeviceFreeMarginBytes ? p.device_free_bytes - kDeviceFreeMarginBytes : 0;
86 const uint64_t device_max_pages = room / p.page_bytes;
87 if (target > device_max_pages) {
88 target = device_max_pages;
89 }
90 }
91 return target > p.current_pages ? uint32_t(target) : 0u;
92}

94// Whether a grow may keep a new allocation. Unknown placement is kept (the backend cannot tell).
95inline bool grow_keeps_allocation(bool p_placement_known, bool p_device_local) {
96 return !p_placement_known || p_device_local;
97}
