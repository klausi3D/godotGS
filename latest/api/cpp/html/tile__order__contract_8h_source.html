4// #1544: total order of the overlap (tile) sort.
5//
6// The overlap sort key is (tile, depth, tie). The tie holds only the low GS_SORT_TIE_BITS (<= 16) bits
7// of global_idx (tile_binning.glsl gs_pack_sort_key), and EMIT takes each record's slot from
8// atomicAdd, so records of one tile with equal full keys leave the sort in emit order, which varies
9// from frame to frame. Pass 4 of tile_prefix_scan.glsl runs on the sorted arrays inside the overlap
10// sort's command list and orders every run of equal keys by global_idx. The raster order is then
11// (key, global_idx): total, and independent of the emit slots and of the sort's stability.

23// Longest run of equal keys pass 4 orders. With 64-bit keys and 16 tie bits a run needs 2^16
24// equal-depth splats per extra member, so 256 covers 2^24 visible splats. Longer runs are left as
25// the sort produced them and counted in key_overflow (tile_order_key_overflow), never truncated.
26static constexpr uint32_t kTileOrderRunCap = 256u;

28// Host mirror of OrderStats (tile_prefix_scan.glsl pass 4, set 0 binding 3). Cumulative counters;
29// the renderer never clears them.
30struct TileOrderStatsGPU {
31 uint32_t runs_resolved = 0; // runs of equal keys ordered by value
32 uint32_t runs_reordered = 0; // runs the sort left out of global_idx order (counted before ordering)
33 uint32_t key_overflow = 0; // tile_order_key_overflow: runs longer than the run cap, left unresolved
34 uint32_t equal_key_pairs = 0; // sort_equal_key_adjacent_pairs: adjacent records with equal full keys
35 uint32_t equal_depth_pairs = 0; // tile_equal_depth_adjacent_pairs: adjacent records with equal (tile, depth)
36 uint32_t truncated_tie_equal = 0; // test mode TIE_TRUNCATED_16 only
37 uint32_t passes = 0; // dispatches of pass 4
38 uint32_t reserved = 0;
39};
40static_assert(sizeof(TileOrderStatsGPU) == 32, "TileOrderStatsGPU must match the 32-byte OrderStats SSBO");

42static_assert(offsetof(TileOrderStatsGPU, passes) == 24, "passes is the seventh OrderStats word");

59// Bits of the overlap sort key that hold the tie (the low bits of global_idx), per layout. Single source
60// for the binning define GS_SORT_TIE_BITS and for pass 4. 32-bit keys carry no tie bits here: the
61// binning's 8-bit quantized-path tie is compiled only when GS_SORT_TIE_BITS > 0, which this returns
62// only for keys wider than 32 bits.
63inline uint32_t tile_order_tie_bits(uint32_t p_key_bits, uint32_t p_required_tile_bits) {
64 if (p_key_bits <= 32u) {
65 return 0u;
66 }
67 const uint32_t tile_bits = p_required_tile_bits < 32u ? p_required_tile_bits : 32u;
68 const uint32_t spare_bits = tile_bits < 32u ? 32u - tile_bits : 0u;
69 return spare_bits < 16u ? spare_bits : 16u;
70}
