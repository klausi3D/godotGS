11// The instance-row region of an HLOD bundle, and the bundle's byte ledger, as pure device-free host code (PR1a of
12// "world + movable instances in one frame"). A bundle holds N instance rows and N grading rows: row 0 is the world.
13// Today exactly one row is produced (the world identity row; the route still requires an exclusive world
14// submission), and for N = 1 every function here returns exactly what the single-row bundle used before.
15//
16// Two generations govern a prepared bundle (the director's InstanceStore counters):
17// - the ASSET-SET generation (get_instance_asset_generation_for_renderer) changes what the bundle packs, so it
18// forces a rebuild;
19// - the ROW generation (get_instance_generation_for_renderer) changes row content only (transform, params,
20// grading), so with an unchanged row count the rows are re-uploaded in place.
21namespace gs_hlod_bundle {

23// The bundle packs one asset, the world: asset id 0 at dense id 0. Rows naming any other asset are refused here; packing
24// node assets (dense ids 1..M) is the next step (PR1b).
25constexpr uint64_t WORLD_ASSET_ID = 0u;
26constexpr uint32_t WORLD_DENSE_ID = 0u;
27// The world is always row 0 (DrawnNodeGPU::instance_id of every world draw record).
28constexpr uint32_t WORLD_ROW = 0u;
29
30struct Rows {
31 LocalVector<InstanceDataGPU> instances;
32 LocalVector<InstanceGradingGPU> grading; // 1:1 with instances.
33 LocalVector<uint64_t> asset_ids; // 1:1 with instances: the director asset id each row names.
34 uint32_t count() const { return instances.size(); }
35};

37// Dense id of a packed asset, or UINT32_MAX when the bundle does not pack it.
38inline uint32_t dense_id_for_asset(uint64_t p_asset_id) {
39 return p_asset_id == WORLD_ASSET_ID ? WORLD_DENSE_ID : UINT32_MAX;
40}

42// GPU-culled (canonical) bundles cull every (row, chunk) pair, so their visible-chunk list holds a slot per pair. The
43// relative bundle's draw list is CPU-written and node-indexed: one entry per world node.
44inline uint64_t visible_chunk_capacity(uint64_t p_node_count, bool p_relative, uint32_t p_rows) {
45 return p_relative ? p_node_count : p_node_count * uint64_t(p_rows);
46}

48// nullptr when the rows can be packed into one bundle of `p_node_count` nodes; otherwise the reason key.
49inline const char *rows_refusal(const Rows &p_rows, uint64_t p_node_count) {
50 const uint32_t n = p_rows.count();
51 if (n == 0) { return "hlod_bundle_rows_empty"; }
52 if (p_rows.grading.size() != n || p_rows.asset_ids.size() != n) { return "hlod_bundle_rows_mismatched"; }
53 // Every buffer of the bundle is created with a 32-bit size (hlod_create_buffer).
54 if (uint64_t(n) * sizeof(InstanceDataGPU) > UINT32_MAX ||
55 visible_chunk_capacity(p_node_count, false, n) * sizeof(VisibleChunkRefGPU) > UINT32_MAX) {
56 return "hlod_bundle_rows_exceed_capacity";
57 }
58 for (uint32_t i = 0; i < n; ++i) {
59 if (dense_id_for_asset(p_rows.asset_ids[i]) == UINT32_MAX) { return "hlod_bundle_row_asset_not_packed"; }
60 }
61 return nullptr;
62}

64// The instance rows exactly as the bundle uploads them: each row names its asset's dense id (ids[0]); every other byte
65// is the director's row. Only valid for rows that passed rows_refusal().
66inline void pack_instance_rows(const Rows &p_rows, LocalVector<InstanceDataGPU> &r_packed) {
67 r_packed = p_rows.instances;
68 for (uint32_t i = 0; i < r_packed.size(); ++i) {
69 r_packed[i].ids[0] = dense_id_for_asset(p_rows.asset_ids[i]);
70 }
71}

73// Bytes of one HLOD bundle: atlas records, per-node tables, the visible/draw list, one AssetMetaGPU, N instance and N
74// grading rows, the splat-ref (and relative draw-entry) capacity, counters and two indirect headers. `p_records` is the
75// atlas record count (whole tree, or the streamed atlas). With p_rows == 1 this is the single-row ledger unchanged.
76inline uint64_t bundle_bytes(uint64_t p_node_count, uint64_t p_records, bool p_relative, bool p_quantized, uint32_t p_capacity,
77 uint32_t p_rows) {
78 return p_records * (p_quantized ? sizeof(PackedGaussianQuantized) : sizeof(PackedGaussian)) +
79 p_node_count * (sizeof(ChunkMetaGPU) + sizeof(AssetChunkIndexGPU) + (p_quantized ? sizeof(ChunkQuantizationGPU) : 0)) +
80 visible_chunk_capacity(p_node_count, p_relative, p_rows) * (p_relative ? sizeof(DrawnNodeGPU) : sizeof(VisibleChunkRefGPU)) +
81 sizeof(AssetMetaGPU) + uint64_t(p_rows) * (sizeof(InstanceDataGPU) + sizeof(InstanceGradingGPU)) +
82 uint64_t(p_capacity) * (sizeof(SplatRefGPU) + (p_relative ? sizeof(uint32_t) : 0)) +
83 6 * sizeof(uint32_t) + 2 * sizeof(GaussianSplatting::IndirectDispatchLayout);
84}

86enum class PreparedRows {
87 CURRENT, // Nothing changed: publish as is.
88 REUPLOAD, // Row content changed, asset set and row count did not: re-upload the rows in place.
89 REBUILD, // The asset set or the row count changed: the bundle no longer describes the scene.
90};

92// `p_row_count` is the current row count; it is only read when the row generation moved.
93inline PreparedRows prepared_rows_action(uint64_t p_prepared_asset_generation, uint64_t p_asset_generation,
94 uint64_t p_prepared_row_generation, uint64_t p_row_generation, uint32_t p_prepared_rows, uint32_t p_row_count) {
95 if (p_prepared_asset_generation != p_asset_generation) { return PreparedRows::REBUILD; }
96 if (p_prepared_row_generation == p_row_generation) { return PreparedRows::CURRENT; }
97 return p_row_count == p_prepared_rows ? PreparedRows::REUPLOAD : PreparedRows::REBUILD;
98}
