24// The project's own per-splat importance metric (interfaces/gpu_culler.cpp:291-293):
25// opacity (clamped) * (max scale axis + epsilon), floored to epsilon. Reused so the subset
26// is selected by the same definition the culler uses, not a new heuristic -- with one
27// deliberate refinement for this DESTRUCTIVE selection: use the scale MAGNITUDE
28// (max(abs(scale))) rather than the signed axis. The covariance build squares scale and
29// GaussianData::compute_radius() treats scale by magnitude, so a large negative-scale splat
30// (e.g. (-5,-4,-3)) is visibly large and must not be forced to the minimum importance and
31// preferentially dropped (Codex #420).
32inline float gaussian_importance(const Gaussian &p_g) {
33 const float opacity = CLAMP(p_g.opacity, 0.0f, 1.0f);
34 const float size_factor = MAX(MAX(Math::abs(p_g.scale.x), Math::abs(p_g.scale.y)), Math::abs(p_g.scale.z));
35 const float importance = opacity * (size_factor + 0.0001f);
36 // Scan/training PLY data can carry NaN/Inf scale or opacity. A non-finite importance would
37 // make the strict-weak-ordering comparator in select_top_k_indices non-transitive, which is
38 // undefined behavior for std::nth_element / std::sort in this DESTRUCTIVE selection. Floor
39 // any non-finite value so the metric is always finite and orderable (deterministic), and
40 // such degenerate splats sort to the bottom rather than corrupting the kept set.
41 if (!Math::is_finite(importance)) {
42 return 0.0001f;
43 }
44 return MAX(0.0001f, importance);
45}

47// Select the `keep` highest-importance indices in [0, count), ties broken by
48// ascending index (a strict total order => a unique, deterministic kept set), and
49// return them SORTED ASCENDING so the caller can compact in place forward-only.
50inline void select_top_k_indices(const float *p_importance, uint32_t p_count, uint32_t p_keep,
51 LocalVector<uint32_t> &r_indices) {
52 r_indices.clear();
53 if (p_count == 0u || p_keep == 0u) {
54 return;
55 }
56 if (p_keep >= p_count) {
57 r_indices.resize(p_count);
58 for (uint32_t i = 0; i < p_count; i++) {
59 r_indices[i] = i;
60 }
61 return;
62 }
63 LocalVector<uint32_t> order;
64 order.resize(p_count);
65 for (uint32_t i = 0; i < p_count; i++) {
66 order[i] = i;
67 }
68 const float *importance = p_importance;
69 const auto better = [importance](uint32_t a, uint32_t b) {
70 if (importance[a] != importance[b]) {
71 return importance[a] > importance[b];
72 }
73 return a < b; // deterministic tie-break
74 };
75 std::nth_element(order.ptr(), order.ptr() + p_keep, order.ptr() + p_count, better);
76 r_indices.resize(p_keep);
77 for (uint32_t i = 0; i < p_keep; i++) {
78 r_indices[i] = order[i];
79 }
80 std::sort(r_indices.ptr(), r_indices.ptr() + p_keep);
81}
