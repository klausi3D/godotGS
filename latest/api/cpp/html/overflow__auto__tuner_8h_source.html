9// Configuration for overflow auto-tuning
10// Close-up safety: Uses EMA smoothing and hysteresis to prevent aggressive culling
11// when the camera moves close to surfaces (which naturally increases overlap).
12struct OverflowAutoTuneConfig {
13 bool enabled = true;
14 float trigger_ratio = 0.01f; // Overflow ratio that triggers adjustment
15 float importance_step = 0.005f; // How much to increase importance threshold
16 float importance_max = 0.1f; // Maximum increase from baseline
17 float importance_decay = 0.001f; // Recovery rate when overflow subsides
18 float tiny_step = 0.25f; // How much to increase tiny splat radius
19 float tiny_max = 2.0f; // Maximum increase from baseline
20 float tiny_decay = 0.05f; // Recovery rate for tiny splat radius
21
22 // Close-up safety parameters (added December 2024)
23 float ema_alpha = 0.15f; // EMA smoothing factor (0.1-0.3 typical, lower = more smoothing)
24 float hysteresis_band = 0.3f; // Must drop to (trigger * (1 - hysteresis_band)) before releasing
25 float max_step_multiplier = 1.5f; // Max amplification of step size (was 4.0, reduced for stability)
26 uint32_t warmup_frames = 10; // Frames before autotune activates (ignore startup spikes)
27 uint32_t cooldown_frames = 30; // Frames to wait before relaxing after overflow event
28 float close_up_threshold = 0.1f; // Average splat screen coverage above this indicates close-up
29 float close_up_dampen = 0.5f; // Reduce aggressiveness by this factor when close-up detected
30};
