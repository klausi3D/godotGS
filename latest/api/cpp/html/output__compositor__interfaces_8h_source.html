23 // GPU-001 Option B source-encoding contract: the GS raster output (RGBA8,
24 // see _resolve_compute_friendly_raster_format) is treated as premultiplied,
25 // display-referred, sRGB-encoded LDR. Ground truth for that definition is
26 // the shipped legacy composite: it wrote these bytes UNCONVERTED into the
27 // presented target, whose content is sRGB-encoded — so "as presented" and
28 // "sRGB-encoded" are the same claim. The raster/resolve shaders perform no
29 // color-space conversion, so this holds for every asset source (PLY SH-DC
30 // and SPZ direct color alike): whatever looked right under the legacy
31 // present-target composite keeps exactly that meaning under the decode.
32 // When the composite destination is the LINEAR pre-tonemap scene buffer
33 // (pre-upscale phase), the compute blit must decode the source to linear
34 // before blending; set this flag to request that decode. Leave false for
35 // legacy post-tonemap destinations, whose content is sRGB-encoded like the
36 // source.
37 bool source_decode_srgb = false;
38 // Destination-alpha contract (#928). The LEGACY composite destination is
39 // Godot's post-tonemap framebuffer, whose alpha channel carries no meaning
40 // — sky, meshes and background are all opaque content — so the compute blit
41 // writes a fully opaque result there, as it always has. Under the Option B
42 // pre-upscale contract the destination is the INTERNAL scene buffer, whose
43 // alpha IS meaningful: a transparent_bg viewport clears it to 0 and Godot's
44 // tonemapper passes the sampled alpha through unchanged. Set this flag when
45 // the destination's alpha must survive the composite, or splat coverage is
46 // flattened to fully opaque and partially-covered silhouette texels are lost.
47 bool destination_has_alpha = false;
48 bool depth_test_enabled = false;
49 bool depth_is_orthogonal = false;
50 float z_near = 0.0f;
51 float z_far = 1.0f;
52 float depth_linearize_mul = 0.0f;
53 float depth_linearize_add = 1.0f;
54 float depth_epsilon = GS_COMPOSITE_DEPTH_EPSILON_VIEW;
55};

58struct OutputCopyResult {

63 // True iff the caller's depth-test request (params.depth_test_enabled) was
64 // actually honored by the executed path. The graphics-blit fallback used
65 // when destination lacks TEXTURE_USAGE_CAN_COPY_FROM_BIT (which would
66 // otherwise require the scratch-copy compute path) does NOT carry depth
67 // information through copy_to_fb_rect, so callers that requested
68 // depth-tested compositing on such targets will see success=true here but
69 // depth_test_honored=false. Set to true when depth was not requested.
70 bool depth_test_honored = true;
71 // True iff no source sRGB->linear decode was requested
72 // (params.source_decode_srgb == false) OR the executed path actually
73 // performed it — which only the compute composite does. Fail-closed: the
74 // dispatcher initializes this to false whenever a decode was requested, so
75 // every other exit (early validation failure, CopyEffects unavailable,
76 // graphics fallback) reports false, including outright failures. The
77 // CopyEffects graphics fallback composites WITHOUT the decode, so a
78 // pre-upscale caller can see success=true (splats still composited —
79 // presence beats absence) with source_decode_honored=false; the caller must
80 // surface that as a degraded copy, never as a clean success.
81 bool source_decode_honored = true;
82};
