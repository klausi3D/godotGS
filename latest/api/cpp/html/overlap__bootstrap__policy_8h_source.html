3// Overlap-sort sizing policy (BV-10, ADR beyond-VRAM worlds, section 5.2 and invariant I-6). Pure so
4// every boundary is testable without a device.
5//
6// E3: the overlap sort used to bootstrap at visible * 50, which reaches the 100M hard cap at 2M
7// visible splats and costs 2,670 MiB before any record exists. Capacity now follows demand:
8// visible * measured records/splat. Before the first measurement a small seed is allocated and the
9// bootstrap frame takes one synchronous overlap sample, so a demand above the seed grows in that
10// same frame and no record is dropped. max_overlap_records is still the only cap: this sizes the
11// allocation, it never limits the records of a frame.
12//
13// I-6: a replacement is built while the old sorter and buffers are live, and RenderingDevice frees
14// them only after the frames in flight, so a resize peaks at old + new.

22// Seed before any measurement. 8 records/splat is the steady planning figure of the ADR (32M records
23// at 4M visible, 855 MiB). It is a seed, not a limit: the bootstrap sample grows past it.
24static constexpr uint32_t OVERLAP_BOOTSTRAP_RECORDS_PER_SPLAT = 8u;
25// Measured records/splat are kept in 8.8 fixed point: far views measure well below one.
26static constexpr uint32_t OVERLAP_RECORDS_PER_SPLAT_ONE = 256u;
27// Same +20% as the adaptive budget, so both estimates read the same demand on the same scale.
28static constexpr uint32_t OVERLAP_ESTIMATE_HEADROOM_PERCENT = 120u;
29// A grow pads to 1.5 * capacity. Every grow recompiles the radix kernels and reallocates hundreds of
30// MB, and capacity no longer starts at the hard cap, so a ramp of slowly rising demand must not
31// replace the sorter on every step.
32static constexpr uint32_t OVERLAP_GROW_FACTOR_PERCENT = 150u;
33// RenderingDevice usage does not see the driver. ADR section 5.2 puts driver and others at up to
34// 0.6 GiB on the 3080.
35static constexpr uint64_t OVERLAP_RESIZE_DRIVER_RESERVE_BYTES = uint64_t(600) * 1024 * 1024;
36
37// 0 means "no sample": an empty view says nothing about the density of the next one.
38inline uint32_t overlap_records_per_splat_q8(uint32_t p_raw_records, uint32_t p_visible_splats) {
39 if (p_raw_records == 0u || p_visible_splats == 0u) {
40 return 0u;
41 }
42 const uint64_t rate = (uint64_t(p_raw_records) * OVERLAP_RECORDS_PER_SPLAT_ONE + p_visible_splats - 1u) / p_visible_splats;
43 return rate > UINT32_MAX ? UINT32_MAX : uint32_t(rate);
44}
45
46// Records to reserve at the start of a frame: visible * measured records/splat + headroom, or the
47// seed before any measurement. The hard cap is applied by the caller (ensure_resources()).
48inline uint32_t overlap_demand_estimate(uint32_t p_visible_splats, uint32_t p_measured_records_per_splat_q8) {
49 const uint64_t rate_q8 = p_measured_records_per_splat_q8 > 0u
50 ? uint64_t(p_measured_records_per_splat_q8) * OVERLAP_ESTIMATE_HEADROOM_PERCENT / 100u + 1u
51 : uint64_t(OVERLAP_BOOTSTRAP_RECORDS_PER_SPLAT) * OVERLAP_RECORDS_PER_SPLAT_ONE;
52 const uint64_t clamped_rate_q8 = rate_q8 > UINT32_MAX ? uint64_t(UINT32_MAX) : rate_q8;
53 const uint64_t records = (uint64_t(p_visible_splats) * clamped_rate_q8 + OVERLAP_RECORDS_PER_SPLAT_ONE - 1u) / OVERLAP_RECORDS_PER_SPLAT_ONE;
54 return records > UINT32_MAX ? UINT32_MAX : uint32_t(records);
55}

57// Per-splat storage must cover the reservation on the indirect route, where the CPU visibility
58// readback can lag the GPU. Keep this selection shared with frame-start overlap sizing.
59inline uint32_t overlap_effective_visible_splats(uint32_t p_splat_count, uint32_t p_max_visible_splats, bool p_indirect_dispatch) {
60 return p_indirect_dispatch && p_max_visible_splats > p_splat_count ? p_max_visible_splats : p_splat_count;
61}

79struct OverlapFrameSizingInput {
80 uint32_t splat_count = 0u;
81 uint32_t effective_visible_splats = 0u;
82 uint32_t current_cpu_draw_domain_count = 0u;

94struct OverlapFrameSizingPlan {
95 uint32_t overlap_base_count = 0u;
96 bool bootstrap_floor_applied = false;
97 uint32_t bootstrap_estimate = 0u;
98 uint32_t requested_records = 0u;
99 // Capacity needed if the request is served, retaining a larger warmed allocation. The resource
100 // owner still applies caps, optional growth padding, shrink hysteresis and resize preflight.
101 uint32_t capacity = 0u;
102};

104// Pure caller policy. Allocation and its failure paths remain with the resource owner.
105inline OverlapFrameSizingPlan overlap_frame_sizing_plan(const OverlapFrameSizingInput &p_input) {
106 OverlapFrameSizingPlan plan;
107 plan.overlap_base_count = p_input.splat_count > p_input.effective_visible_splats ? p_input.splat_count : p_input.effective_visible_splats;
108 uint32_t estimate = overlap_demand_estimate(plan.overlap_base_count, p_input.measured_records_per_splat_q8);
109 plan.bootstrap_floor_applied = p_input.capacity == 0u && p_input.total_gaussians > 0u && plan.overlap_base_count < 1024u;
110 if (plan.bootstrap_floor_applied) {
111 // Streaming warm-up can expose a tiny visible set while the atlas holds a larger world.
112 const uint32_t bootstrap_cap = 8000000u;
113 const uint32_t bootstrap_floor = p_input.total_gaussians < bootstrap_cap ? p_input.total_gaussians : bootstrap_cap;
114 estimate = estimate > bootstrap_floor ? estimate : bootstrap_floor;
115 }
116 plan.bootstrap_estimate = estimate;
117 if (p_input.adaptive_overlap_floor > 0u) {
118 // With history but no measured rate, use the adaptive budget alone. Otherwise the demand
119 // estimate can lead the budget, which only sees a rise in splats after readback completes.
120 if (p_input.adaptive_has_history && p_input.measured_records_per_splat_q8 == 0u) {
121 estimate = p_input.adaptive_overlap_floor;
122 } else {
123 estimate = estimate > p_input.adaptive_overlap_floor ? estimate : p_input.adaptive_overlap_floor;
124 }
125 }
126 if (p_input.sampled_records > 0u && p_input.sampled_cpu_draw_domain_count > 0u &&
127 p_input.current_cpu_draw_domain_count > p_input.sampled_cpu_draw_domain_count) {
128 // Predict only domain growth, using the existing Q8 rate and headroom convention.
129 // Keeping today's estimate as a floor leaves unchanged/shrinking domains on their
130 // existing sizing and shrink policy. The resource owner still enforces the hard cap.
131 const uint32_t sampled_rate_q8 = overlap_records_per_splat_q8(p_input.sampled_records, p_input.sampled_cpu_draw_domain_count);
132 const uint32_t predicted = overlap_demand_estimate(p_input.current_cpu_draw_domain_count, sampled_rate_q8);
133 estimate = estimate > predicted ? estimate : predicted;
134 }
135 plan.requested_records = estimate;
136 plan.capacity = p_input.capacity > estimate ? p_input.capacity : estimate;
137 return plan;
138}

140// Capacity a grow should aim for. Never below the request, and the padding never crosses the hard
141// cap (0 = no cap).
142inline uint32_t overlap_grow_target(uint32_t p_requested_records, uint32_t p_capacity, uint32_t p_hard_cap) {
143 const uint64_t padded = uint64_t(p_capacity) * OVERLAP_GROW_FACTOR_PERCENT / 100u;
144 uint64_t target = padded > p_requested_records ? padded : uint64_t(p_requested_records);
145 if (p_hard_cap > 0u && target > p_hard_cap) {
146 target = p_hard_cap > p_requested_records ? p_hard_cap : p_requested_records;
147 }
148 return target > UINT32_MAX ? UINT32_MAX : uint32_t(target);
149}

157// Pre-flight of a live replacement. p_device_used_bytes is what is allocated now, so it already holds
158// the old sorter and buffers; p_new_bytes is the replacement's full size. Refused when old + new does
159// not fit, even if new alone would. Until the device ledger (BV-11) owns this, the bound is the device
160// heap minus the driver reserve, not the 0.60 GS share, which ADR D-11 keeps report-only on this route.
161inline OverlapResizePreflight overlap_resize_preflight(uint64_t p_device_used_bytes, uint64_t p_new_bytes,
162 uint64_t p_device_capacity_bytes) {
163 if (p_device_capacity_bytes < ResidentAtlasBudget::kMinTrustedDeviceBudgetBytes || p_device_used_bytes == 0u) {
164 return OverlapResizePreflight::UNVERIFIED;
165 }
166 const uint64_t limit = p_device_capacity_bytes > OVERLAP_RESIZE_DRIVER_RESERVE_BYTES
167 ? p_device_capacity_bytes - OVERLAP_RESIZE_DRIVER_RESERVE_BYTES
168 : 0u;
169 if (p_device_used_bytes > limit || p_new_bytes > limit - p_device_used_bytes) {
170 return OverlapResizePreflight::REFUSED;
171 }
172 return OverlapResizePreflight::FITS;
173}
