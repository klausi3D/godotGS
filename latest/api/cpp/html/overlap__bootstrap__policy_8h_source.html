3// Overlap-sort sizing policy (BV-10, ADR beyond-VRAM worlds, section 5.2 and invariant I-6). Pure so
4// every boundary is testable without a device.
5//
6// E3: the overlap sort used to bootstrap at visible * 50, which reaches the 100M hard cap at 2M
7// visible splats and costs 2,670 MiB before any record exists. Capacity now follows demand:
8// visible * measured records/splat. Before the first measurement a small seed is allocated and the
9// bootstrap frame takes one synchronous overlap sample, so a demand above the seed grows in that
10// same frame and no record is dropped. max_overlap_records is still the only cap: this sizes the
11// allocation, it never limits the records of a frame.
12//
13// I-6: a replacement is built while the old sorter and buffers are live, and RenderingDevice frees
14// them only after the frames in flight, so a resize peaks at old + new.

22// Seed before any measurement. 8 records/splat is the steady planning figure of the ADR (32M records
23// at 4M visible, 855 MiB). It is a seed, not a limit: the bootstrap sample grows past it.
24static constexpr uint32_t OVERLAP_BOOTSTRAP_RECORDS_PER_SPLAT = 8u;
25// Measured records/splat are kept in 8.8 fixed point: far views measure well below one.
26static constexpr uint32_t OVERLAP_RECORDS_PER_SPLAT_ONE = 256u;
27// Same +20% as the adaptive budget, so both estimates read the same demand on the same scale.
28static constexpr uint32_t OVERLAP_ESTIMATE_HEADROOM_PERCENT = 120u;
29// A grow pads to 1.5 * capacity. Every grow recompiles the radix kernels and reallocates hundreds of
30// MB, and capacity no longer starts at the hard cap, so a ramp of slowly rising demand must not
31// replace the sorter on every step.
32static constexpr uint32_t OVERLAP_GROW_FACTOR_PERCENT = 150u;
33// RenderingDevice usage does not see the driver. ADR section 5.2 puts driver and others at up to
34// 0.6 GiB on the 3080.
35static constexpr uint64_t OVERLAP_RESIZE_DRIVER_RESERVE_BYTES = uint64_t(600) * 1024 * 1024;
36
37// 0 means "no sample": an empty view says nothing about the density of the next one.
38inline uint32_t overlap_records_per_splat_q8(uint32_t p_raw_records, uint32_t p_visible_splats) {
39 if (p_raw_records == 0u || p_visible_splats == 0u) {
40 return 0u;
41 }
42 const uint64_t rate = (uint64_t(p_raw_records) * OVERLAP_RECORDS_PER_SPLAT_ONE + p_visible_splats - 1u) / p_visible_splats;
43 return rate > UINT32_MAX ? UINT32_MAX : uint32_t(rate);
44}
45
46// Records to reserve at the start of a frame: visible * measured records/splat + headroom, or the
47// seed before any measurement. The hard cap is applied by the caller (ensure_resources()).
48inline uint32_t overlap_demand_estimate(uint32_t p_visible_splats, uint32_t p_measured_records_per_splat_q8) {
49 const uint64_t rate_q8 = p_measured_records_per_splat_q8 > 0u
50 ? uint64_t(p_measured_records_per_splat_q8) * OVERLAP_ESTIMATE_HEADROOM_PERCENT / 100u + 1u
51 : uint64_t(OVERLAP_BOOTSTRAP_RECORDS_PER_SPLAT) * OVERLAP_RECORDS_PER_SPLAT_ONE;
52 const uint64_t clamped_rate_q8 = rate_q8 > UINT32_MAX ? uint64_t(UINT32_MAX) : rate_q8;
53 const uint64_t records = (uint64_t(p_visible_splats) * clamped_rate_q8 + OVERLAP_RECORDS_PER_SPLAT_ONE - 1u) / OVERLAP_RECORDS_PER_SPLAT_ONE;
54 return records > UINT32_MAX ? UINT32_MAX : uint32_t(records);
55}

57// Capacity a grow should aim for. Never below the request, and the padding never crosses the hard
58// cap (0 = no cap).
59inline uint32_t overlap_grow_target(uint32_t p_requested_records, uint32_t p_capacity, uint32_t p_hard_cap) {
60 const uint64_t padded = uint64_t(p_capacity) * OVERLAP_GROW_FACTOR_PERCENT / 100u;
61 uint64_t target = padded > p_requested_records ? padded : uint64_t(p_requested_records);
62 if (p_hard_cap > 0u && target > p_hard_cap) {
63 target = p_hard_cap > p_requested_records ? p_hard_cap : p_requested_records;
64 }
65 return target > UINT32_MAX ? UINT32_MAX : uint32_t(target);
66}

74// Pre-flight of a live replacement. p_device_used_bytes is what is allocated now, so it already holds
75// the old sorter and buffers; p_new_bytes is the replacement's full size. Refused when old + new does
76// not fit, even if new alone would. Until the device ledger (BV-11) owns this, the bound is the device
77// heap minus the driver reserve, not the 0.60 GS share, which ADR D-11 keeps report-only on this route.
78inline OverlapResizePreflight overlap_resize_preflight(uint64_t p_device_used_bytes, uint64_t p_new_bytes,
79 uint64_t p_device_capacity_bytes) {
80 if (p_device_capacity_bytes < ResidentAtlasBudget::kMinTrustedDeviceBudgetBytes || p_device_used_bytes == 0u) {
81 return OverlapResizePreflight::UNVERIFIED;
82 }
83 const uint64_t limit = p_device_capacity_bytes > OVERLAP_RESIZE_DRIVER_RESERVE_BYTES
84 ? p_device_capacity_bytes - OVERLAP_RESIZE_DRIVER_RESERVE_BYTES
85 : 0u;
86 if (p_device_used_bytes > limit || p_new_bytes > limit - p_device_used_bytes) {
87 return OverlapResizePreflight::REFUSED;
88 }
89 return OverlapResizePreflight::FITS;
90}
