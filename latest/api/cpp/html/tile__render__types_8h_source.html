21// ProjectedGaussian payload layout (must match tile_projection_common.glsl).
22// Full layout (36 bytes):
23// data[0]: screen XY as packHalf2x16 (4 bytes)
24// data[1]: depth as raw float32 (was float16 -- f16 far-field quantization exceeded the
25// composite depth_epsilon and caused silhouette shimmer against mesh depth)
26// data[2]: color as R11G11B10F packed format
27// data[3]: conic.x as float32
28// data[4]: conic.z as float32
29// data[5]: conic.y as float32
30// data[6]: global_idx as uint32
31// data[7]: normal.xy as half2
32// data[8]: normal.z as half (low 16) + opacity (unorm8) + flags (uint8) (high 16)
33//
34// Packed layout (32 bytes, optional): conic.y packed as float16 + 16-bit global_idx;
35// normal.z+opacity+flags share data[7]. Lower precision and limited to 16-bit indices,
36// so it is gated at runtime.
37struct TileProjectionLayout {
38 struct alignas(4) Payload {
39 uint32_t data[9];
40 };
41
42 struct alignas(4) PackedPayload {
43 uint32_t data[8];
44 };
45
46 static constexpr uint32_t STRIDE_FULL = sizeof(Payload);
47 static constexpr uint32_t STRIDE_PACKED = sizeof(PackedPayload);
48 static constexpr uint32_t STRIDE = STRIDE_FULL;
49 static_assert(sizeof(Payload) == 36, "TileProjectionLayout::Payload must be 36 bytes");
50 static_assert(sizeof(PackedPayload) == 32, "TileProjectionLayout::PackedPayload must be 32 bytes");
51};
