41 // #611 PR B2: a restore that must run BEFORE anything already queued.
42 //
43 // This exists for exactly one caller: `submit_world_submission`'s
44 // rollback, and it is a behavioural requirement rather than a
45 // preference.
46 //
47 // When phase 1 lazily creates a renderer for a world that already has an
48 // active record P, `_get_or_create_world_for_scenario` queues an apply of
49 // P, and `target_previous_renderer_state` is snapshotted from the
50 // still-blank renderer BEFORE that apply runs. The pre-#611 code then
51 // restored INLINE and let the queued apply(P) flush afterwards, so P was
52 // the last writer and the renderer ended up holding P's contract --
53 // matching the director's record, which a rejection leaves at P.
54 //
55 // Appending the rollback instead inverts that: the flush order becomes
56 // (apply(P), restore(blank)) and the blank snapshot wins, leaving the
57 // renderer cleared while the director still considers P active. That is a
58 // real divergence, not a cosmetic one, and it is the third ordering
59 // defect this restructure has produced -- the queue and an inline call
60 // racing to be last writer.
61 //
62 // Front-insertion restores the original ordering in every case: with a
63 // queued apply present the sequence is (restore, apply(P)) as before, and
64 // with an empty queue it degenerates to the plain append.
65 //
66 // NOT safe to front-insert ahead of an INITIALIZE entry (GPU bring-up must
67 // precede any contract work). That cannot arise here: INITIALIZE is queued
68 // only by `_initialize_world_renderer`, whose sole caller is
69 // `register_instance`, never `submit_world_submission`. If that ever
70 // changes, this must insert after the leading INITIALIZE entries instead.
71 void queue_restore_first(const Ref<GaussianSplatRenderer> &p_renderer,
72 const GaussianSplatWorldSubmissionRuntimeStateSnapshot &p_snapshot);
73 // #611 PR B1: defer `GaussianSplatRenderer::initialize()`.
74 //
75 // ORDERING — this inserts at the FRONT of the queue, and that is
76 // load-bearing. The one caller (`_initialize_world_renderer`, from
77 // `register_instance`) decides to initialize *after*
78 // `_get_or_create_world_for_scenario` may already have queued an apply
79 // for the same lazily-created renderer, but the code this replaces ran
80 // `initialize()` inline under the lock — i.e. BEFORE that queued apply
81 // dispatched. Appending would invert that order and run the apply
82 // against a renderer whose GPU resources are not up yet.
83 //
84 // The guard (`!gpu_resources_initialized && !gpu_initialization_pending`)
85 // is re-evaluated at flush time, not captured here. Under the lock the
86 // check and the call were atomic; deferring separates them, so the check
87 // has to move with the call. It is never weaker than the inline form: the
88 // only way the re-check can skip is if the renderer became initialized or
89 // had an initialization queued in the gap, which is exactly what the
90 // guard exists to detect.
91 void queue_initialize(const Ref<GaussianSplatRenderer> &p_renderer);
92 // Drop queued work that a later decision has superseded. Used where the
93 // caller re-applies a newer contract to the same renderer before
94 // returning; running the stale entry afterwards would clobber it.
95 void cancel();
96 bool is_empty() const { return entries.is_empty(); }
97 uint32_t size() const { return entries.size(); }
98 uint32_t get_dispatched_entry_count() const { return dispatched_entry_count; }
99 // Dispatch-order inspection. Exists so the front-insertion rule above is

114// #610 S6: RendererContractWorkQueue bundles the two deferred-renderer
115// mechanisms every world_mutex critical section in this director must carry
116// into ONE owned local, so the fragile relative destruction order they require
117// stops being a per-call-site convention — a `deferred_renderer_release`
118// vector hand-declared before a DeferredRendererWork, before the lock — and
119// becomes an invariant of this type:
120//
121// * a DeferredRendererWork queue — apply/restore/initialize work whose
122// blocking render-thread dispatch must run AFTER world_mutex is released
123// (#611); and
124// * the deferred-RELEASE vector — renderer Refs a prune
125// (_prune_world_if_unused) moved out of the `worlds` map under the lock,
126// whose ~GaussianSplatRenderer must likewise run unlocked or it is the
127// #628 indefinite editor hang.
128//
129// ORDER — LOAD-BEARING. The queued work must flush BEFORE the released
130// renderer Refs drop: that is the order the pre-S6 inline code had (restore,
131// then free), and release_world_submission still depends on it — it queues a
132// restore against the very Ref the prune moved into release_vector(). Members
133// are destroyed in reverse declaration order, so `deferred_release` is declared
134// FIRST (destroyed LAST) and `work` LAST (destroyed FIRST, running flush() in
135// ~DeferredRendererWork). No explicit destructor is needed: ~DeferredRendererWork
136// flushes, then the vector drops.
137//
138// Declare a RendererContractWorkQueue local BEFORE the
139// `ThreadOwnedMutexLock lock(world_mutex)` in every caller, exactly as its two
140// constituents required individually; the whole object is then destroyed after
141// the lock is released, so both the flush and the release happen unlocked.
142// tests/ci/check_renderer_contract_boundary.py enforces this declaration order
143// for RendererContractWorkQueue just as it does for a bare release vector.
144class GaussianSplatSceneDirector::RendererContractWorkQueue {
145public:
146 // --- deferred contract-work primitives (forwarded to the queue) ---
147 void queue_apply(const Ref<GaussianSplatRenderer> &p_renderer,
148 const GaussianSplatWorldSubmissionContract &p_contract) {

150 }
151 void queue_restore(const Ref<GaussianSplatRenderer> &p_renderer,
152 const GaussianSplatWorldSubmissionRuntimeStateSnapshot &p_snapshot) {
153 work.queue_restore(p_renderer, p_snapshot);
154 }
155 void queue_restore_first(const Ref<GaussianSplatRenderer> &p_renderer,
156 const GaussianSplatWorldSubmissionRuntimeStateSnapshot &p_snapshot) {
157 work.queue_restore_first(p_renderer, p_snapshot);
158 }
159 void queue_initialize(const Ref<GaussianSplatRenderer> &p_renderer) {
160 work.queue_initialize(p_renderer);
161 }
162 void cancel() { work.cancel(); }
163 // Runs and clears the queued contract work. Must not be called while
164 // world_mutex is held (it dispatches). The deferred-release Refs are NOT
165 // touched here; they drop with this object, after this flush.
166 void flush() { work.flush(); }
167
168 // --- deferred renderer-RELEASE vector (the #628 prune pattern) ---
169 // The vector _prune_world_if_unused moves freed renderer Refs into under
170 // world_mutex; they drop only when this object is destroyed — after the
171 // lock is released and after the queued work above has flushed.
172 LocalVector<Ref<GaussianSplatRenderer>> &release_vector() { return deferred_release; }
173
174 RendererContractWorkQueue() = default;
175 RendererContractWorkQueue(const RendererContractWorkQueue &) = delete;
176 RendererContractWorkQueue &operator=(const RendererContractWorkQueue &) = delete;
177
178private:
179 // Declared FIRST → destroyed LAST: the freed renderer Refs drop only after
180 // `work` (below) has flushed in its own destructor. See ORDER above.
181 LocalVector<Ref<GaussianSplatRenderer>> deferred_release;
182 // Declared LAST → destroyed FIRST: ~DeferredRendererWork runs flush().
183 DeferredRendererWork work;
184};
