5// The automatic HLOD route (docs/architecture/adr-one-route-many-producers.md §4.7, maintainer decision 5,
6// 2026-11-04). A GaussianSplatWorld3D whose world can carry an HLOD snapshot (an HLOD tree over a resident payload, or
7// a V4 paged package) takes the HLOD route by default, together with the node instances (rows 1..N of the same bundle), with no
8// project setting changed. Pure, so the routing matrix is a CPU doctest; GaussianSplatWorld3D::_register_shared_renderer()
9// reads the inputs and applies the plan.
10//
11// - rendering/gaussian_splatting/lod/hlod_runtime_cut_enabled (default true) is the explicit opt-out. false restores
12// the routing from before the automatic route exactly: no snapshot is captured, route_policy decides the backend and
13// a V4 paged world loads its payload into memory and takes the resident route (hlod_paged_requires_runtime_cut).
14// - rendering/gaussian_splatting/streaming/route_policy is not an opt-out. Its default is streaming, and a value set
15// equal to a default is indistinguishable from an unset one (docs/architecture/gaussian-project-settings-contract.md),
16// so "streaming" cannot mean "no HLOD". It keeps deciding the backend of every world that does not take the HLOD route.
17// - Node instances do not change the decision: since PR1b they are rows of the world's HLOD bundle.
18namespace gs_hlod_auto_route {

20struct WorldInputs {
21 bool cut_enabled = true; // lod/hlod_runtime_cut_enabled
22 bool has_hlod_tree = false; // GaussianSplatWorld::has_hlod_tree()
23 // GaussianSplatWorld::has_resident_gaussian_data(). A tree over a file-backed (streamable) payload - what a generic
24 // ResourceSaver::save() and the world importer write for a streamable source - cannot be snapshotted (the runtime
25 // cut needs the resident payload), so such a world is not an HLOD-route world: it keeps its route and reports nothing.
26 bool resident_payload = false;
27 bool paged = false; // GaussianSplatWorld::has_hlod_paged_source(), after the paged fallback load (if any) ran
28 bool identity_transform = true; // the world node's global transform is the identity
29 bool route_policy_streaming = true; // route_policy == streaming (the project default)
30 // A setup of this world that the HLOD route never serves, known at submission (nullptr, or one of the keys of
31 // GaussianSplatWorld3D::_hlod_auto_route_refusal(): lod_disabled, deformation_or_instance_domain, painterly_enabled,
32 // shadow_instance_filter).
33 const char *domain_refusal = nullptr;
34 // The node was demoted (see "Sticky demotion" below): it keeps the route it had without the automatic route until
35 // it is applied again. The cause is the caller's; the plan only needs to know.
36 bool demoted = false;
37};

39struct WorldPlan {
40 bool capture_snapshot = false;
41 // The submission's hlod_reason when no snapshot is captured. Empty for a world the HLOD route does not concern
42 // (no tree, or the opt-out set): such a world reports exactly what it reported before the automatic route.
43 String reason;

46// Prefix of the reason a world with an HLOD tree reports when the automatic route keeps it on the chunk-streaming route.
47static constexpr const char *AUTO_ROUTE_UNAVAILABLE_PREFIX = "hlod_auto_route_unavailable:";
48// Prefix of the reason of a demoted world (the cause follows).
49static constexpr const char *AUTO_ROUTE_DEMOTED_PREFIX = "hlod_auto_route_demoted:";

51inline WorldPlan plan_world(const WorldInputs &p_in) {
52 WorldPlan plan;
53 if (!p_in.cut_enabled || (!(p_in.has_hlod_tree && p_in.resident_payload) && !p_in.paged)) {
54 return plan;
55 }
56 if (p_in.demoted) {
57 return plan; // The caller states hlod_auto_route_demoted:<cause>; a paged world loaded its payload for it.
58 }
59 if (!p_in.identity_transform) {
60 plan.reason = "hlod_world_transform_unsupported";
61 return plan;
62 }
63 // Under the default streaming policy a world whose setup the HLOD route never serves keeps the chunk route it has
64 // without a snapshot: the HLOD route would fall back to a canonical resident copy, which the atlas budget thins
65 // where the chunk route streams. Under route_policy = resident the world is on the resident backend either way, and
66 // the HLOD publisher serves these setups with its canonical leaves and resumes the cut when they end (unchanged).
67 // A paged world here could not load its payload (its own reason is already set); it has no other route.
68 if (p_in.route_policy_streaming && p_in.domain_refusal != nullptr && !p_in.paged) {
69 plan.reason = String(AUTO_ROUTE_UNAVAILABLE_PREFIX) + p_in.domain_refusal;
70 return plan;
71 }
72 plan.capture_snapshot = true;
73 return plan;
74}

82// Sticky demotion. Some refusals only appear once the world is on the route: a view the route does not serve
83// (orthographic, multi-view/XR, painterly), a domain that starts after the submission (painterly, the shadow instance
84// filter, effector nodes, wind), or quantization over a tree with mixed DC encodings. Where the world has a canonical copy
85// that serves the frame (a whole-tree resident bundle) the publisher's own fallback draws it. Where it has none (a
86// streamed or paged world), or where that copy is a resident one the atlas budget thins (a world too large for its whole
87// tree), the world would draw less than without the automatic route. Such a refusal demotes the world: the renderer asks
88// once, the node resubmits it without a snapshot (a paged world loads its payload into memory), and it stays on that
89// route until it is applied again. Re-promotion is a follow-up. A request belongs to the snapshot that was refused: the
90// director queues it only while that snapshot is still the world's submitted one, and the node accepts it only for the
91// submission generation the callback was bound to, so a set_world()/apply_world() in between makes it stale.
92enum class RefusalStage {
93 FRAME, // a per-view refusal in prepare_hlod_frame()
94 PUBLISH, // a refusal while the bundle is published (try_publish_hlod_resident_contract())
95};

107// The cause to demote for, or nullptr.
108inline const char *demotion_cause(const RefusalInputs &p_in) {
109 if (p_in.reason == nullptr || p_in.committed_edit) {
110 return nullptr;
111 }
112 const String reason = p_in.reason;
113 if (reason == "hlod_quantized_mixed_dc_node") {
114 // The quantized route refuses the tree outright: the canonical copy is the resident one, which a paged world does
115 // not have and a large world gets thinned. master quantizes it on the chunk route or loads it into memory.
116 return p_in.stage == RefusalStage::PUBLISH ? "quantized_mixed_dc_node" : nullptr;
117 }
118 if (!p_in.no_canonical_copy) {
119 return nullptr;
120 }
121 if (p_in.stage == RefusalStage::FRAME) {
122 if (!p_in.main_view) {
123 return nullptr;
124 }
125 if (reason == "unsupported_view_or_material") { return "unsupported_view_or_material"; }
126 if (reason == "deformation_or_instance_domain" || reason == "hlod_paged_deformation_or_instance_domain") { return "deformation_or_instance_domain"; }
127 if (reason == "lod_disabled_canonical_leaves" || reason == "hlod_paged_lod_disabled") { return "lod_disabled"; }
128 return nullptr;
129 }
130 // PUBLISH: GaussianSplatRenderer::hlod_streaming_unsupported_reason() keys.
131 if (reason == "lod_disabled_canonical_leaves") { return "lod_disabled"; }
132 if (reason == "shadow_instance_filter") { return "shadow_instance_filter"; }
133 if (reason == "unsupported_view_or_material") { return "painterly_enabled"; }
134 if (reason == "deformation_or_instance_domain") { return "deformation_or_instance_domain"; }
135 if (reason == "unsupported_instance_transform_domain") { return "unsupported_instance_transform_domain"; }
136 return nullptr;
137}

139// One demotion request per submitted snapshot: the first refusal asks, later frames do not ask again. A new snapshot
140// (the world applied again, or replaced) re-arms it (install_snapshot()).
141struct DemotionLatch {
142 bool requested = false;
143 String cause;
144 bool request(const char *p_cause) {
145 if (requested || p_cause == nullptr) {
146 return false;
147 }
148 requested = true;
149 cause = p_cause;
150 return true;
151 }
152 void rearm() {
153 requested = false;
154 cause = String();
155 }
156};

158// The renderer installs a submitted snapshot into r_held. The comparison with the held snapshot is the only way to see a
159// replacement, so it runs here, before r_held is overwritten: a different, valid snapshot re-arms the latch (a demoted
160// world resubmits without one, which keeps the latch and its cause). Returns whether the snapshot changed. Templated on
161// the reference type so that the order is a CPU doctest.
162template <typename TSnapshotRef>
163inline bool install_snapshot(TSnapshotRef &r_held, const TSnapshotRef &p_next, DemotionLatch &r_latch) {
164 const bool changed = r_held != p_next;
165 if (changed && p_next.is_valid()) {
166 r_latch.rearm();
167 }
168 r_held = p_next;
169 return changed;
170}
