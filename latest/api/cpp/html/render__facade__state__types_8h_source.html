224// The publisher marks a frame active before the tile stage runs. When the tile stage then refuses it (its admitted
225// scratch or layout does not match the frame) nothing is drawn: the frame is not active, its reason names the refusal,
226// and it is not a ready (or first-ready) streamed frame. No-op for a frame that was not active.
227inline void hlod_mark_tile_stage_refused(ResourceState &r_state, const String &p_cause) {
228 if (!r_state.hlod_frame_active) { return; }
229 r_state.hlod_frame_active = false;
230 r_state.hlod_reason = String("hlod_tile_stage_refused:") + (p_cause.is_empty() ? String("tile_render_failed") : p_cause);
231 ++r_state.hlod_tile_stage_refused_frames;
232 if (r_state.hlod_streaming && r_state.hlod_stream_ready_frames > 0) {
233 // An active streamed frame was counted ready in this same frame (prepare_hlod_frame).
234 --r_state.hlod_stream_ready_frames;
235 if (uint64_t(r_state.hlod_stream_first_ready_frame) == r_state.hlod_stream_frames) { r_state.hlod_stream_first_ready_frame = 0; }
236 }
