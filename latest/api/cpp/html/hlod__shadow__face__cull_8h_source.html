4// Per light view (directional cascade, spot, omni cube face) subset of the HLOD caster's draw list (shadow split ADR
5// addendum, PR0). Device-free; shared by the renderer and the CPU doctests.
6//
7// The caster (shaders/gs_shadow_caster.glsl) issues draw_count * max_chunk_splats instanced quads for EVERY view and rejects
8// off-view splats one vertex at a time. This drops a record from a view only when no splat of it can pass the shader's
9// own clip tests, so the depth the view receives is bit-identical to drawing the whole list: the hardware depth test
10// keeps the maximum (reverse-Z) of every surviving fragment, independent of draw order.
11//
12// Why a dropped record has no surviving splat. Each record's world-space box contains the three-sigma support of every
13// splat it can emit, displaced by at most the row's wind/effector bound:
14// - world nodes: the node AABB contains its payload's three-sigma support (validated when the world is loaded,
15// io/gaussian_splat_world_paged.cpp, gaussian_splat_world_io.cpp; edits stay inside, lod/hlod_edit.h), and the
16// world row must be identity-placed (otherwise the record is kept);
17// - object chunks: a sphere about the chunk centre of radius max(|p - c| + 3 * max(scale)) over its payload, placed
18// by the row's translation with max(1, |uniform scale|) (both flag branches of gs_hlod_camera_relative_position).
19// The shader keeps a quad only if |ndc.xy| <= 1 + extent_ndc, extent = 1.177 sigma (alpha threshold 0.5) along the
20// footprint's eigen axes plus the 0.3 texel^2 low-pass. Per screen axis that reach is <= sqrt(2) * 1.177 sigma +
21// 1.3 texels, inside the projected three-sigma reach plus the guard band below. A box is dropped only when it lies
22// entirely in one outer half-space of the view (x or y beyond the bound, beyond the far plane, or behind a perspective
23// light); the near plane is never used (directional casters in front of it are pancaked, not clipped). Perspective views
24// linearise the footprint at the splat centre, which grows without bound as clip.w -> 0 (a splat beside the light, in the
25// face's own plane, can cover the whole face), so a perspective side test needs w > 0 over the whole box and widens its
26// bound by that footprint at the box's smallest w (View::box_outside). The CPU sweep in
27// tests/test_hlod_shadow_face_cull.cpp mirrors the shader's keep test against these rules.

55// Asset-local radius about p_center that contains the three-sigma support of every splat; false (record never culled)
56// for an empty or non-finite payload. T has float position[3] and scale[3] (PackedGaussian).
57template <typename T>
58inline bool local_support_radius(const T *p_splats, uint32_t p_count, const Vector3 &p_center, real_t &r_radius) {
59 if (!p_splats || p_count == 0) { return false; }
60 double radius = 0.0;
61 for (uint32_t i = 0; i < p_count; ++i) {
62 const T &g = p_splats[i];
63 double distance_sq = 0.0, max_scale = 0.0;
64 for (int a = 0; a < 3; ++a) {
65 const double d = double(g.position[a]) - double(p_center[a]);
66 distance_sq += d * d;
67 max_scale = std::fmax(max_scale, std::fabs(double(g.scale[a])));
68 }
69 const double r = std::sqrt(distance_sq) + kSupportSigmas * max_scale;
70 if (!std::isfinite(r)) { return false; }
71 radius = std::fmax(radius, r);
72 }
73 real_t rounded = real_t(radius);
74 if (double(rounded) < radius) { rounded = std::nextafter(rounded, std::numeric_limits<real_t>::infinity()); }
75 if (!std::isfinite(double(rounded))) { return false; }
76 r_radius = rounded;
77 return true;
78}

80// World-space caster support of one record, or false when it cannot be bounded (the record is then always drawn).
81inline bool record_support(const gs_shadow_refresh::Geometry &p_geometry, uint32_t p_row, const InstanceDataGPU &p_data,
82 real_t p_margin, AABB &r_box) {
83 if (!(p_margin >= 0) || !std::isfinite(double(p_margin))) { return false; }
84 if (p_row == 0) {
85 // World records: offsets are the node centres of an identity-placed world (resident_instance_contract_publisher:
86 // make_drawn_node with WORLD_ROW). Any other placement keeps every world record.
87 const uint32_t identity = GS_INSTANCE_FLAG_ROTATION_IDENTITY | GS_INSTANCE_FLAG_SCALE_IDENTITY | GS_INSTANCE_FLAG_TRANSLATION_ZERO;
88 if (p_geometry.support_kind != SUPPORT_WORLD_NODE_AABB || (p_data.ids[1] & identity) != identity) { return false; }
89 r_box = p_geometry.bounds;
90 r_box.grow_by(p_margin);
91 return true;
92 }
93 if (p_geometry.support_kind != SUPPORT_LOCAL_SPHERE || !(p_geometry.support_radius >= 0)) { return false; }
94 const Vector3 center = p_geometry.bounds.get_center();
95 const double scale = std::fmax(1.0, std::fabs(double(p_data.translation_scale[3])));
96 const double radius = scale * (double(center.length()) + double(p_geometry.support_radius)) + double(p_margin);
97 if (!std::isfinite(radius)) { return false; }
98 const Vector3 origin(p_data.translation_scale[0], p_data.translation_scale[1], p_data.translation_scale[2]);
99 if (!std::isfinite(double(origin.x)) || !std::isfinite(double(origin.y)) || !std::isfinite(double(origin.z))) { return false; }
100 real_t r = real_t(radius);
101 if (double(r) < radius) { r = std::nextafter(r, std::numeric_limits<real_t>::infinity()); }
102 r_box = gs_shadow_refresh::enclosing_bounds(origin - Vector3(r, r, r), origin + Vector3(r, r, r));
103 r_box.grow_by(real_t(std::fmax(std::fabs(double(origin.x)), std::fmax(std::fabs(double(origin.y)), std::fabs(double(origin.z)))) * 1e-6));
104 return true;
105}

107// The per-frame CPU snapshot one renderer's light views cull against. Render thread only; rebuilt with the per-light

128 void build(const gs_shadow_caster::FrameRecord &p_record, const gs_shadow_refresh::PublishedContent &p_content,
129 const LocalVector<InstanceDataGPU> &p_rows, const LocalVector<real_t> &p_margins, uint64_t p_server_frame) {
130 clear();
131 if (!p_record.valid || p_content.gpu_draws.size() != p_record.draw_count ||
132 p_content.geometry.size() != p_record.draw_count || p_margins.size() != p_rows.size()) {
133 return;
134 }
135 rows.resize(p_record.draw_count);
136 boxes.resize(p_record.draw_count);

139 const uint32_t row = p_content.gpu_draws[i].instance_id;
140 rows[i] = row;
141 known[i] = row < p_rows.size() && record_support(p_content.geometry[i], row, p_rows[row], p_margins[row], boxes[i]);
142 }
143 bundle_builds = p_record.bundle_builds;
144 draw_list_upload = p_record.draw_list_upload;
145 draw_count = p_record.draw_count;
146 server_frame = p_server_frame;
147 valid = true;
148 }

156// One light view: the caster's clip transform of frame-relative positions (frame_relative_shadow_clip) and its rect.
157struct View {
158 double rows[4][4] = {}; // rows[r][c]: clip_r = sum_c rows[r][c] * (p - frame_camera, 1)_c
159 double camera[3] = {};
160 double guard_x = 0.0;
161 double guard_y = 0.0;
162 double half_px[2] = {};
163 bool perspective = false;
164 bool valid = false;
165
166 View() = default;
167 View(const Projection &p_clip_from_frame, const Vector3 &p_frame_camera, const Vector2i &p_rect_size, bool p_perspective,
168 double p_guard_texels = kGuardTexels) {
169 for (int r = 0; r < 4; ++r) {
170 for (int c = 0; c < 4; ++c) {
171 rows[r][c] = double(p_clip_from_frame.columns[c][r]);
172 valid = valid || rows[r][c] != 0.0;
173 }
174 }
175 for (int a = 0; a < 3; ++a) { camera[a] = double(p_frame_camera[a]); }
176 perspective = p_perspective;
177 valid = valid && p_rect_size.x > 0 && p_rect_size.y > 0;
178 if (valid) {
179 guard_x = 2.0 * p_guard_texels / double(p_rect_size.x);
180 guard_y = 2.0 * p_guard_texels / double(p_rect_size.y);
181 half_px[0] = 0.5 * double(p_rect_size.x);
182 half_px[1] = 0.5 * double(p_rect_size.y);
183 }
184 for (int r = 0; r < 4 && valid; ++r) {
185 for (int c = 0; c < 4; ++c) { valid = valid && std::isfinite(rows[r][c]); }
186 }
187 }

189 // Minimum and maximum over the box of the affine form row . (p - camera, 1).
190 void range(const double p_row[4], const double p_center[3], const double p_half[3], double &r_min, double &r_max) const {
191 double mid = p_row[3], spread = 0.0;
192 for (int a = 0; a < 3; ++a) {
193 mid += p_row[a] * p_center[a];
194 spread += std::fabs(p_row[a]) * p_half[a];
195 }
196 r_min = mid - spread;
197 r_max = mid + spread;
198 }
199
200 // True only when every point of p_box is rejected by the caster's clip tests (see the file comment).
201 bool box_outside(const AABB &p_box) const {
202 if (!valid) { return false; }
203 double center[3], half[3];
204 for (int a = 0; a < 3; ++a) {
205 half[a] = 0.5 * double(p_box.size[a]);
206 center[a] = double(p_box.position[a]) + half[a] - camera[a];
207 if (!std::isfinite(center[a]) || !std::isfinite(half[a]) || half[a] < 0.0) { return false; }
208 }
209 double w_min, w_max;
210 range(rows[3], center, half, w_min, w_max);
211 if (perspective && w_max < 0.0) { return true; } // Behind the light: every centre has clip.w <= 1e-6.
212 const double w_abs = std::fmax(std::fabs(w_min), std::fabs(w_max));
213 // Largest sigma along any direction of a splat whose three-sigma support fits the box: sigma_dir^2 <= trace of
214 // the covariance <= 3 * (max half / 3)^2.
215 const double sigma_max = std::fmax(half[0], std::fmax(half[1], half[2])) / std::sqrt(3.0);
216 double row3_norm = 0.0;
217 for (int a = 0; a < 3; ++a) { row3_norm += rows[3][a] * rows[3][a]; }
218 row3_norm = std::sqrt(row3_norm);
219 for (int axis = 0; axis < 2 && (!perspective || w_min > 0.0); ++axis) {
220 double bound = 1.0 + (axis == 0 ? guard_x : guard_y);
221 if (perspective) {
222 // The shader keeps a quad when |ndc| <= 1 + E, E = 1.177 sigma * |(row - ndc * row_w) / w| summed over the
223 // two footprint axes (<= sqrt(2) * 1.177 = 1.665 sigma times the gradient norm) plus the low-pass. Near the
224 // face's own plane (w -> 0) that footprint grows without bound, so solve |ndc| (1 - a) > 1 + b for the
225 // smallest w in the box: a = 1.665 sigma |row_w| / w, b = 1.665 sigma |row| / w + 1.3 texels.
226 double row_norm = 0.0;
227 for (int a = 0; a < 3; ++a) { row_norm += rows[axis][a] * rows[axis][a]; }
228 row_norm = std::sqrt(row_norm);
229 const double a = 1.665 * sigma_max * row3_norm / w_min;
230 if (!(a < 0.5)) { continue; }
231 const double b = 1.665 * sigma_max * row_norm / w_min + 1.3 / half_px[axis];
232 bound = std::fmax(bound, (1.0 + b) / (1.0 - a) + (axis == 0 ? guard_x : guard_y));
233 if (!std::isfinite(bound)) { continue; }
234 }
235 double outer[4], inner[4];
236 for (int c = 0; c < 4; ++c) {
237 outer[c] = rows[axis][c] - bound * rows[3][c]; // x - bound * w > 0
238 inner[c] = -rows[axis][c] - bound * rows[3][c]; // -x - bound * w > 0
239 }
240 double lo, hi;
241 range(outer, center, half, lo, hi);
242 if (lo > 1e-9 * (1.0 + w_abs)) { return true; }
243 range(inner, center, half, lo, hi);
244 if (lo > 1e-9 * (1.0 + w_abs)) { return true; }
245 }
246 // Beyond the far plane (ndc.z < 0 with w > 0). The margin dwarfs float rounding of clip.z.
247 double z_min, z_max;
248 range(rows[2], center, half, z_min, z_max);
249 if (z_max < -1e-4 * std::fmax(w_abs, 1e-3)) { return true; }
250 return false;
251 }

263// The records of p_snapshot that the view can receive, in list order, as indices into the published list. A record
264// whose row does not cast is skipped exactly as the shader skips it (row_casts[row] == 0 for an uploaded row).
265// p_unsafe_centre_only is a TEST-ONLY negative control: it culls by each box's centre point, which drops records that
266// still reach the view, so a depth comparison against the whole list must see the difference.
267inline void build_view_list(const Snapshot &p_snapshot, const LocalVector<uint32_t> &p_row_casts, const View &p_view,
268 LocalVector<uint32_t> &r_list, ListStats &r_stats, bool p_unsafe_centre_only = false) {
269 r_list.clear();
270 r_stats = ListStats();
271 r_stats.total = p_snapshot.draw_count;
272 for (uint32_t i = 0; i < p_snapshot.draw_count; ++i) {
273 const uint32_t row = p_snapshot.rows[i];
274 if (row < p_row_casts.size() && p_row_casts[row] == 0) {
275 ++r_stats.culled_noncasting;
276 continue;
277 }
278 if (!p_snapshot.known[i]) {
279 ++r_stats.unbounded;
280 } else if (p_view.box_outside(p_unsafe_centre_only ? AABB(p_snapshot.boxes[i].get_center(), Vector3()) : p_snapshot.boxes[i])) {
281 ++r_stats.culled_outside;
282 continue;
283 }
284 r_list.push_back(i);
285 }
286 if (r_list.is_empty() && p_snapshot.draw_count > 0) {
287 r_list.push_back(0); // Proven to emit nothing into this view.
288 r_stats.placeholder = true;
289 }
290 r_stats.listed = r_list.size();
291}
