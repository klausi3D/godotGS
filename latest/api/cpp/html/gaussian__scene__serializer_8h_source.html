290 // Largest number of bytes codec `type` can PHYSICALLY emit from
291 // `compressed_size` input bytes, given its container format (Zstd: one 4-byte
292 // RLE block header per 128 KiB block; FastLZ level 2: one length byte per 255
293 // copied bytes). Saturates instead of overflowing. Because this is a property
294 // of the format, not of the data, it can never reject a stream our writer
295 // produced. See the derivation in gaussian_scene_serializer.cpp.
296 static uint64_t max_codec_expansion_bytes(CompressionType type, uint64_t compressed_size);
297
298 // Absolute ceiling, in bytes, that a single chunk of a load may claim. Unlike
299 // the codec/corroboration bounds -- which bound a chunk against the FILE --
300 // this bounds it against the MACHINE, and is the only thing that stops an
301 // internally CONSISTENT hostile file from forcing a multi-GiB allocation.
302 // Derived from OS::get_memory_info() rather than being a tuned constant, so it
303 // cannot re-introduce the "rejects legitimately large assets" regression.
304 static uint64_t get_load_allocation_budget_bytes();
305 // Test / embedder seam: pin the budget to an exact value (0 restores the
306 // OS-derived default). Process-global and NOT thread-safe; intended to be set
307 // once, from the main thread, around a load.
308 static void set_load_allocation_budget_override(uint64_t bytes);
309
310 // Anti-decompress-bomb bound for a compressed chunk (#603a): true when a chunk
311 // declaring `original_size` decompressed bytes from `compressed_size` bytes is
312 // acceptable. Bounded by (1) the uint32 on-disk field width, (2) the
313 // corroborated ceiling the surrounding file structure justifies, (3) the
314 // codec's physical maximum expansion, and (4) `memory_budget`, the absolute
315 // allocation ceiling for this load. (1)-(3) are NOT tunable ratio heuristics,
316 // which previously rejected this serializer's own highly-compressible output.
317 // Pure (the budget is an explicit parameter, not read from global state) and
318 // public + static so it can be unit-tested at the boundary without allocating
319 // multi-GiB buffers. Not a bound (script) method.
320 static bool is_decompressed_chunk_size_plausible(uint64_t original_size, uint64_t compressed_size,
321 CompressionType type, uint64_t corroborated_max_size, uint64_t memory_budget);
