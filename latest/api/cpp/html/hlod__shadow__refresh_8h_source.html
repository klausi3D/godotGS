12// Render-thread CPU state only. No GPU ownership, worker wait, or additional lock.
13// FNV-1a 64-bit digest: accidental collisions remain possible (~2^-64 per independent comparison).
14// Hash once per record/row, then combine only spatial hits per light; retaining exact bytes per light would
15// duplicate the intersecting draw/row data for every lamp. This is a cache key, not an adversarial hash.
16inline uint64_t hash_bytes(uint64_t p_hash, const void *p_data, size_t p_size) {
17 const uint8_t *bytes = static_cast<const uint8_t *>(p_data);
18 for (size_t i = 0; i < p_size; ++i) { p_hash = (p_hash ^ bytes[i]) * UINT64_C(1099511628211); }
19 return p_hash;
20}
21static constexpr uint64_t HASH_SEED = UINT64_C(14695981039346656037);

23template <typename T>
24inline uint64_t hash_value(uint64_t p_hash, const T &p_value) {
25 return hash_bytes(p_hash, &p_value, sizeof(T));
26}

28// Preserve both endpoints when the float span crosses a large origin (e.g. [-2e8, 1]). AABB(lo, hi-lo)
29// alone can round its computed end back to zero. Aggregate cells must not narrow and prune an edge caster.
30inline AABB enclosing_bounds(const Vector3 &p_lower, const Vector3 &p_upper) {
31 Vector3 size;
32 for (int axis = 0; axis < 3; ++axis) {
33 const double span = double(p_upper[axis]) - double(p_lower[axis]);
34 size[axis] = real_t(span);
35 if (double(size[axis]) < span || p_lower[axis] + size[axis] < p_upper[axis]) {
36 size[axis] = std::nextafter(size[axis], std::numeric_limits<real_t>::infinity());
37 }
38 }
39 return AABB(p_lower, size);
40}
41
42inline AABB enclosing_merge(const AABB &p_a, const AABB &p_b) {
43 return enclosing_bounds(p_a.position.min(p_b.position),
44 (p_a.position + p_a.size).max(p_b.position + p_b.size));
45}

96// Fixed-depth sparse octree of record centres, with aggregate *full inflated bounds* at every node (large records
97// cannot be lost by centre assignment). O(6*N) build, <=min(1+6*N, 299593) nodes, O(N) storage. Query prunes
98// disjoint cells and consumes fully enclosed cells in O(1), including their digest and animation flag. Only boundary
99// cells visit records; cost is O(N + sum_light(visited cells + boundary-leaf candidates)), not an unconditional
100// lights*records scan. Degenerate boundary clustering can still visit N records per light. Depth is fixed at six.
101struct ContentIndex {
102 struct Cell {
103 AABB bounds;
104 uint32_t children[8] = { UINT32_MAX, UINT32_MAX, UINT32_MAX, UINT32_MAX, UINT32_MAX, UINT32_MAX, UINT32_MAX, UINT32_MAX };
105 uint32_t head = UINT32_MAX;
106 uint64_t digest = 0;
107 uint32_t count = 0;
108 bool animated = false;
109 void add(const SpatialRecord &p_record) {
110 if (count == 0) { bounds = p_record.bounds; } else { bounds = enclosing_merge(bounds, p_record.bounds); }
111 digest += p_record.digest;
112 ++count;
113 animated |= p_record.animated;
114 }
115 };
116 LocalVector<SpatialRecord> records;
117 LocalVector<Cell> cells;
118 void build() {
119 cells.clear();
120 if (records.is_empty()) { return; }
121 Cell root;
122 for (const SpatialRecord &record : records) { root.add(record); }
123 cells.push_back(root);
124 for (uint32_t i = 0; i < records.size(); ++i) {
125 Vector3 lo = root.bounds.position, size = root.bounds.size;
126 const Vector3 center = records[i].bounds.get_center();
127 uint32_t cell = 0;
128 for (uint32_t depth = 0; depth < 6; ++depth) {
129 size *= real_t(0.5);
130 uint32_t octant = 0;
131 for (int axis = 0; axis < 3; ++axis) {
132 if (center[axis] >= lo[axis] + size[axis]) { octant |= 1u << axis; lo[axis] += size[axis]; }
133 }
134 uint32_t child = cells[cell].children[octant];
135 if (child == UINT32_MAX) {
136 child = cells.size();
137 cells[cell].children[octant] = child;
138 Cell added;
139 cells.push_back(added);
140 }
141 cells[child].add(records[i]);
142 cell = child;
143 }
144 records[i].next = cells[cell].head;
145 cells[cell].head = i;
146 }
147 }
148 void rebuild(const PublishedContent &p_content, const gs_shadow_caster::FrameRecord &p_record,

155// Godot's spot shadow projection has axial length range and base radius range*tan(angle). The apex-centred sphere
156// with radius range/cos(angle) encloses that cone, including its far rim. Omnis retain their exact range sphere.
157inline real_t positional_sphere_radius(real_t p_range, real_t p_spot_angle) {
158 return p_spot_angle < 0 ? p_range : p_range / MAX(Math::cos(Math::deg_to_rad(p_spot_angle)), real_t(1e-6));
159}

161// Includes touching boundaries and avoids the much looser light-AABB test for omnis.
162inline bool sphere_intersects_bounds(const Vector3 &p_center, real_t p_radius, const AABB &p_bounds) {
163 const Vector3 end = p_bounds.position + p_bounds.size;
164 const real_t dx = MAX(MAX(p_bounds.position.x - p_center.x, p_center.x - end.x), real_t(0));
165 const real_t dy = MAX(MAX(p_bounds.position.y - p_center.y, p_center.y - end.y), real_t(0));
166 const real_t dz = MAX(MAX(p_bounds.position.z - p_center.z, p_center.z - end.z), real_t(0));
167 return p_radius >= 0 && dx * dx + dy * dy + dz * dz <= p_radius * p_radius;
168}

170inline bool sphere_contains_bounds(const Vector3 &p_center, real_t p_radius, const AABB &p_bounds) {
171 const Vector3 end = p_bounds.position + p_bounds.size;
172 const real_t dx = MAX(Math::abs(p_bounds.position.x - p_center.x), Math::abs(end.x - p_center.x));
173 const real_t dy = MAX(Math::abs(p_bounds.position.y - p_center.y), Math::abs(end.y - p_center.y));
174 const real_t dz = MAX(Math::abs(p_bounds.position.z - p_center.z), Math::abs(end.z - p_center.z));
175 return p_radius >= 0 && dx * dx + dy * dy + dz * dz <= p_radius * p_radius;
176}

178inline void ContentIndex::query(uint32_t p_cell, const Vector3 &p_center, real_t p_radius,
179 uint64_t &r_digest, uint32_t &r_count, bool &r_animated) const {
180 const Cell &cell = cells[p_cell];
181 if (!sphere_intersects_bounds(p_center, p_radius, cell.bounds)) { return; }
182 if (sphere_contains_bounds(p_center, p_radius, cell.bounds)) {
183 r_digest += cell.digest;
184 r_count += cell.count;
185 r_animated |= cell.animated;
186 return;
187 }
188 for (uint32_t i = cell.head; i != UINT32_MAX; i = records[i].next) {
189 const SpatialRecord &record = records[i];
190 if (!sphere_intersects_bounds(p_center, p_radius, record.bounds)) { continue; }
191 // Commutative multiset digest: cut/list order and spatial-cell changes cannot dirty a map.
192 r_digest += record.digest;
193 ++r_count;
194 r_animated |= record.animated;
195 }
196 for (uint32_t child : cell.children) {
197 if (child != UINT32_MAX) { query(child, p_center, p_radius, r_digest, r_count, r_animated); }
198 }
199}

201// No camera, list upload/build counter, whole-cut revision, or dense row index in the content identity.
202inline uint64_t record_digest(const DrawnNodeGPU &p_draw, const Geometry &p_geometry, uint64_t p_row_digest) {
203 uint64_t digest = hash_value(p_row_digest, p_geometry.chunk_key);
204 digest = hash_value(digest, p_draw.fade);
205 digest = hash_value(digest, p_draw.level);
206 digest = hash_value(digest, p_geometry.payload);
207 for (int axis = 0; axis < 3; ++axis) {
208 digest = hash_value(digest, p_geometry.bounds.position[axis]);
209 digest = hash_value(digest, p_geometry.bounds.size[axis]);
210 }
211 // Avalanche before commutative addition, so neighbouring node/row keys do not have linear sums.
212 digest = (digest ^ (digest >> 33)) * UINT64_C(0xff51afd7ed558ccd);
213 digest = (digest ^ (digest >> 33)) * UINT64_C(0xc4ceb9fe1a85ec53);
214 return digest ^ (digest >> 33);
215}

217inline uint64_t row_digest(const InstanceDataGPU &p_row, uint64_t p_key, uint64_t p_asset) {
218 InstanceDataGPU row = p_row; // Layout has asserted size and no padding.
219 row.ids[0] = 0; // The stable asset identity below replaces the bundle's dense remap.
220 row.lod[0] = 0; // Ignore camera-selected LOD; lod.y remains the semantic override base.
221 row.params[1] = 0; // LOD bias affects admission, already represented by selected node/chunk identities.
222 return hash_value(hash_value(hash_value(HASH_SEED, p_key), p_asset), row);
223}

225inline void ContentIndex::rebuild(const PublishedContent &p_content, const gs_shadow_caster::FrameRecord &p_record,
226 const LocalVector<InstanceDataGPU> &p_rows, const LocalVector<real_t> &p_margins,
227 const LocalVector<uint32_t> &p_animated) {
228 LocalVector<uint64_t> row_digests;
229 row_digests.resize(p_rows.size());
230 for (uint32_t row = 0; row < p_rows.size(); ++row) {
231 row_digests[row] = row_digest(p_rows[row], p_record.row_keys[row], p_record.row_asset_ids[row]);
232 }
233 records.clear();
234 for (uint32_t i = 0; i < p_content.draws.size(); ++i) {
235 const auto &draw = p_content.draws[i];
236 const uint32_t row = draw.instance_id;
237 if (row >= p_rows.size() || !p_record.row_casts[row]) { continue; }
238 SpatialRecord spatial;
239 spatial.bounds = record_bounds(p_content.geometry[i], row, p_rows[row], p_margins[row]);
240 spatial.digest = record_digest(draw, p_content.geometry[i], row_digests[row]);
241 spatial.animated = p_animated[row] != 0;
242 records.push_back(spatial);
243 }
244 build();
245}

251inline bool effectors_active_on_casting_row(const InstanceDataGPU &p_row, bool p_casts, const float (&p_meta)[4],
252 const float (&p_spheres)[GS_MAX_SPHERE_EFFECTORS][4], const float (&p_configs)[GS_MAX_SPHERE_EFFECTORS][4],
253 const float (&p_opacity)[GS_MAX_SPHERE_EFFECTORS][4]) {
254 if (!p_casts) { return false; }
255 uint32_t mask = 0;
256 std::memcpy(&mask, &p_row.effect_params[2], sizeof(mask));
257 for (uint32_t slot = 0; slot < MIN(uint32_t(p_meta[0]), GS_MAX_SPHERE_EFFECTORS); ++slot) {
258 // Conservatively ignore sphere/splat overlap, time, and per-node fade. Scene masks retain packed slot indices;
259 // the legacy project sphere applies without a scene mask. Both position and opacity can change shadow content.
260 if (p_configs[slot][0] <= 0.5f || p_spheres[slot][3] <= 1e-6f ||
261 (p_meta[3] > 0.5f && (mask & (1u << slot)) == 0)) { continue; }
262 if ((p_row.effect_params[0] > 0.0f && p_opacity[slot][0] > 0.5f && std::abs(p_configs[slot][1]) > 1e-8f) ||
263 (p_row.effect_params[1] > 0.0f && p_opacity[slot][1] > 0.5f && p_opacity[slot][2] > 0.0f)) { return true; }
264 }
265 return false;
266}

289// The module's render-thread cache holds one state per live light, keyed by light-instance RID. Its last-pass list is
290// bounded by the intersecting visible owners of that pass; the scene-cull light-free hook erases the whole entry.
291struct LightState {
292 LocalVector<Owner> last_pass;
293 void commit(const LightUpdate &p_update) {
294 last_pass = p_update.owners;
295 for (Owner &owner : last_pass) { owner.content = nullptr; }
296 }
297};

299inline LightUpdate prepare_light(const LocalVector<Owner> &p_owners, const Vector3 &p_center, real_t p_radius,
300 const LightState &p_state) {
301 LightUpdate update;
302 for (const Owner &source : p_owners) {
303 if (!sphere_intersects_bounds(p_center, p_radius, source.bounds)) { continue; }
304 Owner owner = source;
305 if (source.content) {
306 owner.revision = 0;
307 owner.record_count = 0;
308 owner.animated = false;
309 if (!source.content->cells.is_empty()) {
310 source.content->query(0, p_center, p_radius, owner.revision, owner.record_count, owner.animated);
311 }
312 if (owner.record_count == 0) { continue; }
313 }
314 owner.content = nullptr;
315 const Owner *seen = nullptr;
316 for (const Owner &old : p_state.last_pass) {
317 if (old.owner == owner.owner) { seen = &old; break; }
318 }
319 update.refresh |= needs_refresh(true, seen != nullptr, seen && !owner.same_content(*seen), owner.animated);
320 update.owners.push_back(owner);
321 }
322 // A departed owner must be cleared even when it was the last one. Committing that empty pass ends the refresh.
323 // Equal-size replacements already refresh through the unseen-owner check above; two empty sets never refresh.
324 if (update.owners.size() != p_state.last_pass.size()) { update.refresh = true; }
325 return update;
326}
