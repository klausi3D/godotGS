4// ADR docs/architecture/adr-hlod-streaming.md sections 6.6, 14.1, 14.2 (package D1): glue between the CPU
5// residency model (lod/hlod_node_residency.*) and the HLOD relative bundle's GPU buffers.
6//
7// One streamer per prepared HLOD bundle. It is not a second streaming system: placement, generations,
8// retirement and eviction are the shared NodeResidency model on top of GaussianAtlasAllocator (#1088); this
9// class only (a) packs requested nodes on worker threads with the bundle's single packing path
10// (renderer/hlod_node_pack.*), (b) uploads completed payloads, bounded per frame, and (c) hands the selector
11// the resident mask. The atlas starts empty and every ChunkMetaGPU starts zeroed (splat_count 0), so a node
12// that is not resident-ready draws nothing even if a stale record named it.
13//
14// Package F1 (ADR 14.5, editable regions): stage_edit() takes an edited snapshot (lod/hlod_edit.*), reserves shadow
15// runs for the edited leaves, loads the replacement cut and packs the new leaf payloads on the same worker path;
16// commit_staged_edit() is the atomic publication boundary. Until it commits, nothing drawable changes: the old
17// snapshot, its resident mask and its GPU metadata keep serving a consistent old cut. A paged (V4) file is never
18// written: unedited nodes keep reading from it, edited leaves come from the in-memory overlay of the edited snapshot.
19//
20// Threading: update()/begin_frame()/shutdown() belong to the render thread. Workers only read the immutable
21// runtime snapshot and write into a mailbox they co-own; they never touch the residency model or the device.

222 // `p_reserved` is the job's whole CPU working set while it runs: the file record plus the packed copy.
223 void begin_read(uint64_t p_reserved) {
224 in_flight.fetch_add(1);
225 const uint64_t now = in_flight_bytes.fetch_add(p_reserved) + p_reserved;
226 uint64_t peak = in_flight_bytes_peak.load();
227 while (now > peak && !in_flight_bytes_peak.compare_exchange_weak(peak, now)) {
228 }
229 }
230 void end_read(uint64_t p_reserved, uint64_t p_file_bytes, bool p_ok) {
231 in_flight_bytes.fetch_sub(p_reserved);
232 in_flight.fetch_sub(1);
233 if (p_ok) {
234 reads.fetch_add(1);
235 bytes.fetch_add(p_file_bytes);
236 uint64_t seen = largest.load();
237 while (p_file_bytes > seen && !largest.compare_exchange_weak(seen, p_file_bytes)) {
238 }
239 } else {
240 failures.fetch_add(1);
241 }
242 }
