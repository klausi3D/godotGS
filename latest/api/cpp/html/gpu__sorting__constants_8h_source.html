8static constexpr uint32_t DEFAULT_WORKGROUP_SIZE = 256;
9// Radix-sort digit width PER PASS (4-bit = 16 passes for a 64-bit key, 8-bit = 8 passes). This is a
10// performance/mechanics knob ONLY — it does NOT change the sorted output or key precision (precision is
11// key_bits/tile_bits/depth_bits). Both 4 and 8 are valid (validate() accepts both) and both are now
12// CORRECT at any workgroup_size (the strided per-bin shader loops in gpu_sorter.cpp + the shared-memory
13// probe in RadixSort::is_supported handle 8-bit's 256 bins). DEFAULT IS 4-BIT: measured A/B on
14// GrandmasHouse (2026-06-06) showed 8-bit is SLOWER, not faster — fewer passes are outweighed by 16x
15// larger histograms, more per-pass bin work, and lower occupancy (~10 KB shared mem). Keep 4-bit unless
16// a future measurement on different hardware/workload shows otherwise.
17static constexpr uint32_t DEFAULT_RADIX_BITS = 4;
18static constexpr uint32_t RADIX_BITS = 8;
19static constexpr uint32_t RADIX_SIZE = 1u << RADIX_BITS;
20static constexpr uint32_t MAX_WORKGROUP_SIZE = 1024;
21static constexpr uint32_t HISTOGRAM_BINS = RADIX_SIZE;

23// Sort-key bit layout defaults. 64-bit (32 tile + 32 depth) is the ONLY
24// shippable layout: 32-bit quantized depth keys flicker/band on real-scan data
25// (GS-298). These are the single source of truth for the key-width defaults so
26// the ProjectSettings GLOBAL_DEF and every get_setting() fallback agree — a read
27// before registration can then never silently select the broken 32-bit path.
28// (32-bit remains reachable only via an explicit gpu_preset="custom" opt-in.)
29static constexpr uint32_t DEFAULT_KEY_BITS = 64;
30static constexpr uint32_t DEFAULT_TILE_BITS = 32;
31static constexpr uint32_t DEFAULT_DEPTH_BITS = 32;

33// ---------------------------------------------------------------------------
34// Sort-path allocation bound
35// ---------------------------------------------------------------------------
36// RenderingDevice::storage_buffer_create() takes the size as a **uint32_t**
37// (servers/rendering/rendering_device.h). Every buffer the sort path allocates is
38// sized from the element count, so a large enough element count produces a size
39// that silently TRUNCATES modulo 2^32: the device hands back a buffer far smaller
40// than the shaders index, and the sort writes out of bounds. That is silent VRAM
41// corruption, which is strictly worse than a clean rejection — so the element
42// count must be bounded by the LARGEST buffer the path allocates, not just by the
43// key buffer.
44//
45// The dominant term is NOT the key buffer. RadixSort preallocates a per-workgroup,
46// per-bin, per-pass histogram (and an identically sized workgroup-prefix buffer):
47//
48// workgroups = ceil(N / workgroup_size)
49// radix_size = 1 << radix_bits
50// num_passes = ceil(key_bits / radix_bits)
51// histogram_bytes = workgroups * radix_size * num_passes * 4
52//
53// which scales as N * (radix_size * num_passes * 4 / workgroup_size). At the
54// permissive end of the validated knob ranges (workgroup_size=64, radix_bits=8,
55// key_bits=64) that is 128 bytes PER ELEMENT — 16x the 8-byte 64-bit key buffer.
56// The safe element count therefore depends on radix_bits / workgroup_size /
57// key_bits and cannot be expressed as one scalar constant.
58//
59// Sentinel returned by sort_path_max_buffer_bytes() when the configuration is not
60// one the sort path can build at all (an unsupported radix_bits). It is
61// deliberately UINT64_MAX so that every "does this fit in RenderingDevice's
62// uint32_t size parameter?" comparison FAILS CLOSED: an unsupported config can
63// never be mistaken for a small, allocatable one. Callers that want to *report*
64// the condition (rather than just reject it) must compare against this sentinel
65// before formatting a byte count — see GPUSortingConfig::get_validation_errors().
66static constexpr uint64_t SORT_PATH_SIZE_UNSUPPORTED = UINT64_MAX;
67
68// The radix widths the sort path can actually build. RadixSort::create_variant()
69// and GPUSortingConfig::validate() accept exactly this set. Anything else has no
70// defined buffer sizing, and evaluating `1ull << radix_bits` for it would be
71// UNDEFINED BEHAVIOUR once radix_bits >= 64 (a shift count at or above the width
72// of the promoted type). Every shift by a radix width must go through this check
73// first, so the helper is total for ANY input instead of safe only when the
74// caller happens to pre-validate.
75inline bool is_supported_radix_bits(uint32_t p_radix_bits) {
76 return p_radix_bits == 4 || p_radix_bits == 8;
77}
78
79// Returns the largest single storage_buffer_create() size, in bytes, computed in
80// 64-bit so the result is the TRUE required size even when it exceeds uint32.
81// Callers reject when the result exceeds UINT32_MAX. Bounding the byte total also
82// bounds RadixSort's uint32 `histogram_stride` intermediate (stride <= bytes / 4).
83//
84// TOTAL: defined for every possible argument. An unsupported radix_bits returns
85// SORT_PATH_SIZE_UNSUPPORTED instead of shifting by it. The radix check is FIRST
86// — before the zero-element short-circuit — so an unsupported configuration is
87// reported as unsupported no matter what element count accompanies it.
88inline uint64_t sort_path_max_buffer_bytes(uint64_t p_max_elements, uint32_t p_radix_bits,
89 uint32_t p_workgroup_size, uint32_t p_key_bits) {
90 if (!is_supported_radix_bits(p_radix_bits)) {
91 return SORT_PATH_SIZE_UNSUPPORTED;
92 }
93 if (p_max_elements == 0) {
94 return 0;
95 }
96 const uint32_t radix_bits = p_radix_bits;
97 // workgroup_size only ever appears as a DIVISOR here, so it cannot invoke UB
98 // the way a shift count can; it is still coerced away from zero to keep the
99 // division defined for any input.
100 const uint32_t workgroup_size = p_workgroup_size > 0 ? p_workgroup_size : DEFAULT_WORKGROUP_SIZE;
101 const uint32_t key_bits = p_key_bits > 32 ? 64u : 32u;
102
103 // --- RadixSort (gpu_sorter.cpp RadixSort::initialize) ---
104 const uint64_t radix_size = 1ull << radix_bits; // radix_bits is 4 or 8 here — never >= 64.
105 uint64_t num_passes = (uint64_t(key_bits) + radix_bits - 1ull) / radix_bits;
106 if (num_passes == 0) {
107 num_passes = 1;
108 }
109 uint64_t workgroups = (p_max_elements + workgroup_size - 1ull) / workgroup_size;
110 if (workgroups == 0) {
111 workgroups = 1;
112 }
113 // histogram_buffer and wg_prefix_buffer are both this size.
114 const uint64_t histogram_bytes = workgroups * radix_size * num_passes * sizeof(uint32_t);
115 const uint64_t key_stride_bytes = (key_bits > 32) ? 8ull : 4ull;
116 const uint64_t temp_keys_bytes = p_max_elements * key_stride_bytes;
117 const uint64_t temp_values_bytes = p_max_elements * sizeof(uint32_t);
118
119 // --- OneSweepSort (fixed WORKGROUP_SIZE/RADIX_SIZE, 32-bit keys) ---
120 const uint64_t onesweep_workgroups =
121 (p_max_elements + DEFAULT_WORKGROUP_SIZE - 1ull) / DEFAULT_WORKGROUP_SIZE;
122 const uint64_t onesweep_histogram_bytes = onesweep_workgroups * RADIX_SIZE * sizeof(uint32_t);
123
124 uint64_t largest = histogram_bytes;
125 if (temp_keys_bytes > largest) {
126 largest = temp_keys_bytes;
127 }
128 if (temp_values_bytes > largest) {
129 largest = temp_values_bytes;
130 }
131 if (onesweep_histogram_bytes > largest) {
132 largest = onesweep_histogram_bytes;
133 }
134 return largest;
135}

147// ---------------------------------------------------------------------------
148// RadixSort scatter-kernel shared-memory bound (#634)
149// ---------------------------------------------------------------------------
150// The RadixSort scatter kernel is the shared-memory PEAK of the sort path. It
151// allocates, in compute shared memory:
152//
153// local_bases[radix_size] + local_offsets[radix_size] + bin_masks[radix_size * mask_words]
154//
155// where radix_size = 1 << radix_bits and mask_words = ceil(workgroup_size / 32)
156// (one uint bitmask word per 32 workgroup lanes). radix_bits and workgroup_size
157// are each validated INDEPENDENTLY, but the constraint is on their PRODUCT: e.g.
158// (radix_bits=8, workgroup_size=512) needs 256*(2+16)*4 = 18432 bytes, which
159// exceeds the shared-memory budget of common Intel/AMD parts. When the scatter
160// pipeline cannot be built the renderer falls through to compositing translucent
161// splats UNSORTED — mathematically wrong output. See #634 / #586.
162//
163// This helper is the SINGLE SOURCE OF TRUTH for that derived quantity:
164// * RadixSort::is_supported() (gpu_sorter.cpp) probes the ACTUAL device limit
165// against this value, and
166// * GPUSortingConfig::get_validation_errors() flags any pair whose requirement
167// exceeds VULKAN_MIN_COMPUTE_SHARED_MEMORY_BYTES at configuration time.
168// Neither re-derives the formula by hand.
169
170// Sentinel returned for a radix width the scatter kernel cannot build. Same
171// fail-closed rationale as SORT_PATH_SIZE_UNSUPPORTED: it is UINT32_MAX so every
172// "does this fit in <limit>?" comparison FAILS CLOSED — an unsupported radix can
173// never be mistaken for a small, buildable footprint. `1u << radix_bits` is UB
174// once radix_bits >= 32, so the helper returns this instead of shifting.
175static constexpr uint32_t RADIX_SCATTER_SHARED_MEMORY_UNSUPPORTED = UINT32_MAX;
176
177// Vulkan 1.1 guarantees maxComputeSharedMemorySize >= 16384 bytes (16 KB) on every
178// conformant device (docs/PLATFORM_COMPATIBILITY.md requires Vulkan 1.1+); common
179// Intel/AMD parts sit exactly at this floor. A (radix_bits, workgroup_size) pair
180// whose scatter footprint exceeds this PORTABLE minimum is guaranteed to fail
181// RadixSort::is_supported() on some supported hardware, so it can be rejected at
182// CONFIG time — before any device exists — instead of surfacing as silently
183// unsorted output many frames later.
184static constexpr uint32_t VULKAN_MIN_COMPUTE_SHARED_MEMORY_BYTES = 16384u;
185
186// Bytes of compute shared memory the RadixSort scatter kernel requires for a given
187// (radix_bits, workgroup_size). TOTAL: an unsupported radix_bits returns
188// RADIX_SCATTER_SHARED_MEMORY_UNSUPPORTED instead of shifting by it. A zero
189// workgroup_size is coerced to DEFAULT_WORKGROUP_SIZE (matching the runtime probe
190// and sort_path_max_buffer_bytes) so the result is defined for any input.
191inline uint32_t radix_scatter_shared_memory_bytes(uint32_t p_radix_bits, uint32_t p_workgroup_size) {
192 if (!is_supported_radix_bits(p_radix_bits)) {
193 return RADIX_SCATTER_SHARED_MEMORY_UNSUPPORTED;
194 }
195 const uint32_t workgroup_size = p_workgroup_size > 0 ? p_workgroup_size : DEFAULT_WORKGROUP_SIZE;
196 const uint32_t radix_size = 1u << p_radix_bits; // radix_bits is 4 or 8 here — never >= 32.
197 const uint32_t mask_words = (workgroup_size + 31u) / 32u;
198 const uint32_t scatter_shared_uints = radix_size * (2u + mask_words);
199 return scatter_shared_uints * uint32_t(sizeof(uint32_t));
200}
