27// Static device-local VRAM capacity below which RenderingDevice::get_device_memory_budget()
28// is treated as unknown. Mirrors streaming_vram_regulator.cpp's
29// MIN_TRUSTED_DEVICE_CAPACITY_BYTES (#321): a degenerate small-positive value would
30// otherwise clamp the atlas to nonsense.
31static constexpr uint64_t kMinTrustedDeviceBudgetBytes = uint64_t(256) * 1024 * 1024;
32
33// Absolute single-upload staging ceiling, shared with the resident publisher's
34// kMaxResidentUploadBytes. Datasets above this still hard-reject (streaming is the
35// right answer); the budget cap below only ever tightens beneath it.
36static constexpr uint64_t kResidentStagingCeilingBytes = uint64_t(2) * 1024 * 1024 * 1024;
37
38// Fraction of device-local VRAM the resident atlas may occupy. The remainder is
39// reserved for the same frame's sort / splat-ref / visible-chunk buffers, engine
40// framebuffers, and driver reserve (get_device_memory_budget() is the device heap,
41// not net of this app's non-atlas allocations). The 0.60 * budget term only binds
42// below ~3.3 GB device VRAM (0.60 * budget < 2 GB); at/above that, the staging
43// ceiling dominates and capable GPUs are byte-identical to today.
44static constexpr double kResidentAtlasHeadroomFraction = 0.60;
45
46// Effective byte cap for the packed resident atlas.
47// - device_budget_bytes == 0 or below the sane floor => capacity unknown =>
48// cap stays at the staging ceiling (do-no-harm on UMA/iGPU/software/non-Vulkan).
49// - override_mb > 0 pins an explicit cap (low-end clamp + on-capable-GPU validation).
50inline uint64_t compute_effective_atlas_cap_bytes(uint64_t device_budget_bytes, uint32_t override_mb,
51 uint64_t staging_ceiling_bytes = kResidentStagingCeilingBytes,
52 double headroom_fraction = kResidentAtlasHeadroomFraction,
53 uint64_t min_trusted_budget_bytes = kMinTrustedDeviceBudgetBytes) {
54 uint64_t cap = staging_ceiling_bytes;
55 if (device_budget_bytes >= min_trusted_budget_bytes) {
56 const uint64_t budget_cap = uint64_t(double(device_budget_bytes) * headroom_fraction);
57 cap = MIN(cap, budget_cap);
58 }
59 if (override_mb > 0) {
60 const uint64_t override_bytes = uint64_t(override_mb) * 1024u * 1024u;
61 cap = MIN(cap, override_bytes);
62 }
63 return cap;
64}

66struct SubsetPlan {
67 bool reduced = false; // true when the atlas must be thinned to fit
68 double keep_ratio = 1.0; // global fraction of splats to keep (1.0 == full)
69 uint64_t target_keep = 0; // soft splat target (cap_bytes / packed_size)
70 uint64_t source_count = 0; // full atlas splat count
71 uint64_t cap_bytes = 0; // effective byte cap that produced this plan
72};

74// Decide whether the full atlas fits the cap and, if not, the global keep ratio.
75inline SubsetPlan compute_subset_plan(uint64_t total_gaussians, uint64_t effective_cap_bytes,
76 uint32_t packed_gaussian_size = 144u) {
77 SubsetPlan plan;
78 plan.source_count = total_gaussians;
79 plan.cap_bytes = effective_cap_bytes;
80 plan.target_keep = total_gaussians;
81 if (packed_gaussian_size == 0u || total_gaussians == 0u) {
82 return plan;
83 }
84 const uint64_t cap_splats = effective_cap_bytes / uint64_t(packed_gaussian_size);
85 if (cap_splats >= total_gaussians) {
86 return plan; // fits: keep_ratio == 1.0, reduced == false
87 }
88 plan.reduced = true;
89 plan.target_keep = cap_splats;
90 plan.keep_ratio = double(cap_splats) / double(total_gaussians);
91 return plan;
92}

94// Per-asset slice of the global keep budget when the resident atlas spans multiple assets.
95// Each asset gets a share proportional to its splat count (ceil, so a non-empty asset keeps
96// >=1), capped by the budget still globally available. This keeps any one asset -- including
97// the currently visible/casting ones -- from being starved to zero by an earlier large
98// (possibly hidden) asset processed first in the stable superset. The split is a pure function
99// of per-device counts, NOT visibility, so the atlas stays stable across visibility flips
100// (Codex #420 round 3).
101inline uint32_t resident_asset_budget(uint64_t p_target_keep, uint64_t p_asset_count,
102 uint64_t p_total_gaussians, uint32_t p_global_remaining) {
103 if (p_asset_count == 0u || p_total_gaussians == 0u) {
104 return 0u;
105 }
106 const uint64_t proportional = (p_target_keep * p_asset_count + p_total_gaussians - 1u) / p_total_gaussians;
107 const uint64_t share = MAX<uint64_t>(uint64_t(1), proportional);
108 return uint32_t(MIN<uint64_t>(uint64_t(p_global_remaining), share));
109}

111// Per-chunk keep count under a global keep_ratio. Floors at 1 for any non-empty
112// chunk so thinning never deletes a whole spatial region (preserves coverage on
113// real-scan content); the per-chunk floor remainder is the documented soft
114// overshoot bounded by the hard staging ceiling.
115inline uint32_t compute_chunk_keep_count(uint32_t count, double keep_ratio) {
116 if (count == 0u) {
117 return 0u;
118 }
119 if (keep_ratio >= 1.0) {
120 return count;
121 }
122 if (keep_ratio <= 0.0) {
123 return 1u;
124 }
125 uint32_t keep = uint32_t(Math::round(double(count) * keep_ratio));
126 keep = MAX(1u, keep);
127 return MIN(keep, count);
128}

134// Compact a captured chunk snapshot down to its top-importance subset in place,
135// moving both the Gaussian payload and its parallel high-order SH block (stride =
136// p_sh_stride coefficients per gaussian, gaussian-major, per capture_*_snapshot).
137// `r_remaining_budget` is the global cap on cumulative kept splats across all chunks: the
138// per-chunk keep is clamped to it and it is decremented by the kept count, so the resident
139// atlas honors the device VRAM target even when many per-chunk floors would otherwise
140// overshoot it (e.g. a dataset of many one-splat static chunks) -- Codex #420. Returns the
141// number of kept gaussians (0 when the budget is exhausted -> the caller drops the chunk).
142inline uint32_t compact_chunk_by_importance(LocalVector<Gaussian> &r_gaussians,
143 LocalVector<Vector3> &r_sh_high_order, uint32_t p_sh_stride, double p_keep_ratio,
144 uint32_t &r_remaining_budget) {
145 const uint32_t count = r_gaussians.size();
146 uint32_t keep = compute_chunk_keep_count(count, p_keep_ratio);
147 keep = MIN(keep, r_remaining_budget); // hard global cap
148 if (keep >= count) {
149 r_remaining_budget -= count;
150 return count;
151 }
152 if (keep == 0) {
153 r_gaussians.clear();
154 r_sh_high_order.clear();
155 return 0;
156 }
157
158 LocalVector<float> importance;
159 importance.resize(count);
160 for (uint32_t i = 0; i < count; i++) {
161 importance[i] = gaussian_importance(r_gaussians[i]);
162 }
163
164 LocalVector<uint32_t> keep_indices;
165 select_top_k_indices(importance.ptr(), count, keep, keep_indices);
166
167 // Forward in-place compaction: keep_indices is ascending and keep <= count, so
168 // keep_indices[j] >= j for every j -- writing slot j never clobbers an as-yet
169 // unread source slot.
170 for (uint32_t j = 0; j < keep_indices.size(); j++) {
171 const uint32_t src = keep_indices[j];
172 if (src != j) {
173 r_gaussians[j] = r_gaussians[src];
174 }
175 }
176 r_gaussians.resize(keep);
177
178 if (p_sh_stride > 0u && uint64_t(r_sh_high_order.size()) >= uint64_t(count) * uint64_t(p_sh_stride)) {
179 Vector3 *sh = r_sh_high_order.ptr();
180 for (uint32_t j = 0; j < keep_indices.size(); j++) {
181 const uint32_t src = keep_indices[j];
182 if (src != j) {
183 const uint64_t dst_base = uint64_t(j) * uint64_t(p_sh_stride);
184 const uint64_t src_base = uint64_t(src) * uint64_t(p_sh_stride);
185 for (uint32_t c = 0; c < p_sh_stride; c++) {
186 sh[dst_base + c] = sh[src_base + c];
187 }
188 }
189 }
190 r_sh_high_order.resize(keep * p_sh_stride);
191 }
192 r_remaining_budget -= keep;
193 return keep;
194}
