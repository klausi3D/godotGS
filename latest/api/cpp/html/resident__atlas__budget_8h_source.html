4// Pure, device-independent helpers for clamping the resident atlas to the
5// device VRAM budget (#321) and packing an importance-ordered subset (a coarse
6// LOD) when the full atlas would not fit. Deliberately free of RenderingDevice /
7// RID / global-config dependencies so the budget math and the subset selection
8// are unit-testable on the host without a GPU. See
9// resident_instance_contract_publisher.cpp for the single production caller.

27// Static device-local VRAM capacity below which RenderingDevice::get_device_memory_budget()
28// is treated as unknown. Mirrors streaming_vram_regulator.cpp's
29// MIN_TRUSTED_DEVICE_CAPACITY_BYTES (#321): a degenerate small-positive value would
30// otherwise clamp the atlas to nonsense.
31static constexpr uint64_t kMinTrustedDeviceBudgetBytes = uint64_t(256) * 1024 * 1024;
32
33// Absolute single-upload staging ceiling, shared with the resident publisher's
34// kMaxResidentUploadBytes. Datasets above this still hard-reject (streaming is the
35// right answer); the budget cap below only ever tightens beneath it.
36static constexpr uint64_t kResidentStagingCeilingBytes = uint64_t(2) * 1024 * 1024 * 1024;
37
38// Fraction of device-local VRAM the resident atlas may occupy. The remainder is
39// reserved for the same frame's sort / splat-ref / visible-chunk buffers, engine
40// framebuffers, and driver reserve (get_device_memory_budget() is the device heap,
41// not net of this app's non-atlas allocations). The 0.60 * budget term only binds
42// below ~3.3 GB device VRAM (0.60 * budget < 2 GB); at/above that, the staging
43// ceiling dominates and capable GPUs are byte-identical to today.
44static constexpr double kResidentAtlasHeadroomFraction = 0.60;
45
46// Effective byte cap for the packed resident atlas.
47// - device_budget_bytes == 0 or below the sane floor => capacity unknown =>
48// cap stays at the staging ceiling (do-no-harm on UMA/iGPU/software/non-Vulkan).
49// - override_mb > 0 pins an explicit cap (low-end clamp + on-capable-GPU validation).
50inline uint64_t compute_effective_atlas_cap_bytes(uint64_t device_budget_bytes, uint32_t override_mb,
51 uint64_t staging_ceiling_bytes = kResidentStagingCeilingBytes,
52 double headroom_fraction = kResidentAtlasHeadroomFraction,
53 uint64_t min_trusted_budget_bytes = kMinTrustedDeviceBudgetBytes) {
54 uint64_t cap = staging_ceiling_bytes;
55 if (device_budget_bytes >= min_trusted_budget_bytes) {
56 const uint64_t budget_cap = uint64_t(double(device_budget_bytes) * headroom_fraction);
57 cap = MIN(cap, budget_cap);
58 }
59 if (override_mb > 0) {
60 const uint64_t override_bytes = uint64_t(override_mb) * 1024u * 1024u;
61 cap = MIN(cap, override_bytes);
62 }
63 return cap;
64}

66// Byte limit for the HLOD route's device-wide ledger (every RenderingDevice buffer plus the tile and
67// compositor textures, get_hlod_owned_byte_upper_bound()). The ledger includes the canonical
68// route's own sort / tile / atlas buffers (819 MB on the 252k-leaf holzbank scan), so it cannot be
69// bounded by the HLOD-owned byte cap; that cap bounds the HLOD bundle on its own. The ledger is
70// bounded by the same documented device-heap fraction as the atlas (kResidentAtlasHeadroomFraction;
71// ADR 6.5 / 14.1: budget checks use allocated bytes against one physical budget). An unknown or
72// untrusted heap size keeps the fixed hlod_owned_cap (fail closed, as before).
73inline uint64_t compute_hlod_device_ledger_limit_bytes(uint64_t device_budget_bytes, uint64_t hlod_owned_cap_bytes,
74 double headroom_fraction = kResidentAtlasHeadroomFraction,
75 uint64_t min_trusted_budget_bytes = kMinTrustedDeviceBudgetBytes) {
76 if (device_budget_bytes >= min_trusted_budget_bytes) {
77 return uint64_t(double(device_budget_bytes) * headroom_fraction);
78 }
79 return device_budget_bytes ? MIN(device_budget_bytes, hlod_owned_cap_bytes) : hlod_owned_cap_bytes;
80}

82// Admission of a candidate HLOD bundle: its own bytes must fit the HLOD-owned cap, and the
83// device-wide ledger plus the candidate must fit the ledger limit. Pure so the boundaries are testable.
84inline bool hlod_candidate_fits(uint64_t candidate_bytes, uint64_t device_bytes, uint64_t ledger_limit_bytes,
85 uint64_t hlod_owned_cap_bytes) {
86 return candidate_bytes <= hlod_owned_cap_bytes && candidate_bytes <= ledger_limit_bytes &&
87 device_bytes <= ledger_limit_bytes - candidate_bytes;
88}

90// Same for an already prepared bundle: device_bytes already includes the bundle.
91inline bool hlod_prepared_fits(uint64_t owned_bytes, uint64_t device_bytes, uint64_t ledger_limit_bytes,
92 uint64_t hlod_owned_cap_bytes) {
93 return owned_bytes <= hlod_owned_cap_bytes && device_bytes <= ledger_limit_bytes;
94}

96// Streaming decision for an HLOD tree (package D): stream whenever the whole-tree bundles cannot be admitted -

102// Pages a streamed HLOD atlas may hold. `fixed_bundle_bytes` is the relative bundle without its atlas records.
103// The result is clamped by the atlas cap, the tree itself, the HLOD-owned cap and the device ledger headroom
104// (the same limits hlod_candidate_fits applies); 0 means not even one page is admissible.
105// The ledger headroom must also hold the frame scratch the route allocates before its first frame
106// (TileRenderer::prepare_hlod_scratch): `scratch_fixed_bytes` plus `scratch_bytes_per_record` per atlas record. A page
107// budget that left less than that would be refused by the scratch admission on every frame and draw nothing for good.
108// The scratch is device memory, not HLOD-owned bundle bytes and not atlas bytes, so only the ledger term reserves it.
109inline uint64_t hlod_stream_page_budget(uint64_t atlas_limit_bytes, uint64_t record_stride, uint32_t page_splats,
110 uint64_t total_node_pages, uint64_t fixed_bundle_bytes, uint64_t hlod_owned_cap_bytes,
111 uint64_t ledger_limit_bytes, uint64_t device_bytes, uint64_t scratch_fixed_bytes, uint64_t scratch_bytes_per_record) {
112 if (record_stride == 0 || page_splats == 0) { return 0; }
113 const uint64_t page_bytes = record_stride * page_splats;
114 const uint64_t page_ledger_bytes = page_bytes + scratch_bytes_per_record * page_splats;
115 if (fixed_bundle_bytes >= hlod_owned_cap_bytes || device_bytes > ledger_limit_bytes ||
116 ledger_limit_bytes - device_bytes <= fixed_bundle_bytes ||
117 ledger_limit_bytes - device_bytes - fixed_bundle_bytes <= scratch_fixed_bytes) { return 0; }
118 uint64_t pages = MIN(atlas_limit_bytes / page_bytes, total_node_pages);
119 pages = MIN(pages, (hlod_owned_cap_bytes - fixed_bundle_bytes) / page_bytes);
120 pages = MIN(pages, (ledger_limit_bytes - device_bytes - fixed_bundle_bytes - scratch_fixed_bytes) / page_ledger_bytes);
121 return pages;
122}

124// Records the relative HLOD atlas holds. The tile frame indexes its per-splat scratch (subpixel history, SH colour
125// cache, the 16-bit packed-stage index gate) by atlas record, so HLOD scratch admission must cover exactly this.
126// A resident atlas holds the tree (leaf + parent splats). A streamed atlas holds whole pages and every node rounds
127// up to its own pages, so once the budget reaches the tree's node pages it holds MORE records than the tree has
128// splats: sizing the scratch from the tree then refuses every frame (a black frame), see ADR 14.4.1.
129inline uint32_t hlod_relative_atlas_records(bool streaming, uint32_t stream_atlas_records, uint32_t leaf_count,
130 uint32_t parent_count) {
131 return streaming ? stream_atlas_records : leaf_count + parent_count;
132}

134struct SubsetPlan {
135 bool reduced = false; // true when the atlas must be thinned to fit
136 double keep_ratio = 1.0; // global fraction of splats to keep (1.0 == full)
137 uint64_t target_keep = 0; // soft splat target (cap_bytes / packed_size)
138 uint64_t source_count = 0; // full atlas splat count
139 uint64_t cap_bytes = 0; // effective byte cap that produced this plan
140};
141
142// Decide whether the full atlas fits the cap and, if not, the global keep ratio.
143inline SubsetPlan compute_subset_plan(uint64_t total_gaussians, uint64_t effective_cap_bytes,
144 uint32_t packed_gaussian_size = 144u) {
145 SubsetPlan plan;
146 plan.source_count = total_gaussians;
147 plan.cap_bytes = effective_cap_bytes;
148 plan.target_keep = total_gaussians;
149 if (packed_gaussian_size == 0u || total_gaussians == 0u) {
150 return plan;
151 }
152 const uint64_t cap_splats = effective_cap_bytes / uint64_t(packed_gaussian_size);
153 if (cap_splats >= total_gaussians) {
154 return plan; // fits: keep_ratio == 1.0, reduced == false
155 }
156 plan.reduced = true;
157 plan.target_keep = cap_splats;
158 plan.keep_ratio = double(cap_splats) / double(total_gaussians);
159 return plan;
160}

162// Per-asset slice of the global keep budget when the resident atlas spans multiple assets.
163// Each asset gets a share proportional to its splat count (ceil, so a non-empty asset keeps
164// >=1), capped by the budget still globally available. This keeps any one asset -- including
165// the currently visible/casting ones -- from being starved to zero by an earlier large
166// (possibly hidden) asset processed first in the stable superset. The split is a pure function
167// of per-device counts, NOT visibility, so the atlas stays stable across visibility flips
168// (Codex #420 round 3).
169inline uint32_t resident_asset_budget(uint64_t p_target_keep, uint64_t p_asset_count,
170 uint64_t p_total_gaussians, uint32_t p_global_remaining) {
171 if (p_asset_count == 0u || p_total_gaussians == 0u) {
172 return 0u;
173 }
174 const uint64_t proportional = (p_target_keep * p_asset_count + p_total_gaussians - 1u) / p_total_gaussians;
175 const uint64_t share = MAX<uint64_t>(uint64_t(1), proportional);
176 return uint32_t(MIN<uint64_t>(uint64_t(p_global_remaining), share));
177}

179// Per-chunk keep count under a global keep_ratio. Floors at 1 for any non-empty
180// chunk so thinning never deletes a whole spatial region (preserves coverage on
181// real-scan content); the per-chunk floor remainder is the documented soft
182// overshoot bounded by the hard staging ceiling.
183inline uint32_t compute_chunk_keep_count(uint32_t count, double keep_ratio) {
184 if (count == 0u) {
185 return 0u;
186 }
187 if (keep_ratio >= 1.0) {
188 return count;
189 }
190 if (keep_ratio <= 0.0) {
191 return 1u;
192 }
193 uint32_t keep = uint32_t(Math::round(double(count) * keep_ratio));
194 keep = MAX(1u, keep);
195 return MIN(keep, count);
196}

202// Compact a captured chunk snapshot down to its top-importance subset in place,
203// moving both the Gaussian payload and its parallel high-order SH block (stride =
204// p_sh_stride coefficients per gaussian, gaussian-major, per capture_*_snapshot).
205// `r_remaining_budget` is the global cap on cumulative kept splats across all chunks: the
206// per-chunk keep is clamped to it and it is decremented by the kept count, so the resident
207// atlas honors the device VRAM target even when many per-chunk floors would otherwise
208// overshoot it (e.g. a dataset of many one-splat static chunks) -- Codex #420. Returns the
209// number of kept gaussians (0 when the budget is exhausted -> the caller drops the chunk).
210inline uint32_t compact_chunk_by_importance(LocalVector<Gaussian> &r_gaussians,
211 LocalVector<Vector3> &r_sh_high_order, uint32_t p_sh_stride, double p_keep_ratio,
212 uint32_t &r_remaining_budget) {
213 const uint32_t count = r_gaussians.size();
214 uint32_t keep = compute_chunk_keep_count(count, p_keep_ratio);
215 keep = MIN(keep, r_remaining_budget); // hard global cap
216 if (keep >= count) {
217 r_remaining_budget -= count;
218 return count;
219 }
220 if (keep == 0) {
221 r_gaussians.clear();
222 r_sh_high_order.clear();
223 return 0;
224 }
225
226 LocalVector<float> importance;
227 importance.resize(count);
228 for (uint32_t i = 0; i < count; i++) {
229 importance[i] = gaussian_importance(r_gaussians[i]);
230 }
231
232 LocalVector<uint32_t> keep_indices;
233 select_top_k_indices(importance.ptr(), count, keep, keep_indices);
234
235 // Forward in-place compaction: keep_indices is ascending and keep <= count, so
236 // keep_indices[j] >= j for every j -- writing slot j never clobbers an as-yet
237 // unread source slot.
238 for (uint32_t j = 0; j < keep_indices.size(); j++) {
239 const uint32_t src = keep_indices[j];
240 if (src != j) {
241 r_gaussians[j] = r_gaussians[src];
242 }
243 }
244 r_gaussians.resize(keep);
245
246 if (p_sh_stride > 0u && uint64_t(r_sh_high_order.size()) >= uint64_t(count) * uint64_t(p_sh_stride)) {
247 Vector3 *sh = r_sh_high_order.ptr();
248 for (uint32_t j = 0; j < keep_indices.size(); j++) {
249 const uint32_t src = keep_indices[j];
250 if (src != j) {
251 const uint64_t dst_base = uint64_t(j) * uint64_t(p_sh_stride);
252 const uint64_t src_base = uint64_t(src) * uint64_t(p_sh_stride);
253 for (uint32_t c = 0; c < p_sh_stride; c++) {
254 sh[dst_base + c] = sh[src_base + c];
255 }
256 }
257 }
258 r_sh_high_order.resize(keep * p_sh_stride);
259 }
260 r_remaining_budget -= keep;
261 return keep;
262}

264// Texture bytes a tile grid of `new_px` pixels charges to the HLOD ledger when `live_bytes` are charged for `cur_px`
265// pixels now (the output, resolved, depth and normal targets scale with the pixel count). A smaller or equal grid never
266// costs more than it does now; an unknown current size (0) keeps the live bytes. Rounded up, saturating.
267inline uint64_t hlod_scale_texture_bytes_for_pixels(uint64_t live_bytes, uint64_t cur_px, uint64_t new_px) {
268 if (cur_px == 0 || new_px == 0 || new_px <= cur_px || live_bytes == 0) { return live_bytes; }
269 if (live_bytes > UINT64_MAX / new_px) { return UINT64_MAX; }
270 return (live_bytes * new_px + cur_px - 1u) / cur_px;
271}

273// What a streamed HLOD view must do with the tile grid before its scratch is admitted. A streamed world never runs a
274// canonical frame, so nothing else resizes the grid: a view whose size differs from the grid, or a grid without overlap
275// storage, must be provisioned for THIS view, per view and per frame. The new targets must fit the ledger headroom
276// (`wanted_texture_bytes` is what the grid would charge afterwards); otherwise the view is refused, honestly.
277enum class HlodViewGridAction { KEEP, RESIZE, REFUSE_MEMORY };
278inline HlodViewGridAction hlod_view_grid_action(int p_grid_x, int p_grid_y, int p_view_x, int p_view_y, uint32_t p_overlap_capacity,
279 uint64_t p_live_texture_bytes, uint64_t p_wanted_texture_bytes, uint64_t p_headroom_bytes) {
280 if (p_grid_x == p_view_x && p_grid_y == p_view_y && p_overlap_capacity != 0) { return HlodViewGridAction::KEEP; }
281 if (p_wanted_texture_bytes > p_live_texture_bytes && p_wanted_texture_bytes - p_live_texture_bytes > p_headroom_bytes) {
282 return HlodViewGridAction::REFUSE_MEMORY;
283 }
284 return HlodViewGridAction::RESIZE;
285}
