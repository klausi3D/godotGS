80 // #1086: forward-progress signals that separate a FAILURE from designed
81 // backpressure. The "needed set" is the visible chunks inside the load distance
82 // (exactly the set _load_visible_chunks treats as load candidates).
83 //
84 // Scan starvation: needed chunks sit unserved (not loaded, not upload-pending)
85 // while the visible scan had enqueue headroom, yet the scan found no load
86 // candidate -- the scan window never reached the demand. This is the case a
87 // throttled or capped scan produces when it restarts at a nearest prefix that is
88 // already resident. A scan cut to one chunk because pack jobs in flight hit
89 // max_pack_jobs_in_flight (headroom 0) is designed backpressure, not starvation.
90 // A frame is ELIGIBLE (could have starved) when the scan had capacity and needed
91 // chunks were unserved; with an uncapped, unthrottled scan every eligible frame
92 // finds candidates, so starvation needs a scan budget below the visible count.
93 struct VisibleScanStarvationInput {
94 uint32_t needed_unserved_chunks = 0;
95 bool scan_had_capacity = false;
96 uint32_t load_candidates = 0;
97 };

99 static bool is_visible_scan_starved(const VisibleScanStarvationInput &p_input);

104 // Needed-set progress. A frame makes progress when a NEEDED chunk completes that
105 // did not merely replace another needed chunk: completions of non-needed chunks
106 // (prefetch) do not count, and a completion first pays off "displacement debt"
107 // -- one unit per needed (visible) chunk evicted to make room. Evict/reload
108 // churn of the needed set therefore makes no progress. Debt no pending load can
109 // repay is dropped (capped at the in-flight load count).
110 //
111 // The stall is the sum of streaming frame deltas (each clamped by
112 // _resolve_frame_delta_seconds to [0.0005, 0.25] s) for which the needed set has
113 // been incomplete without progress; it resets on progress or a complete needed
114 // set. The threshold is the engine's STALL_THRESHOLD_FRAMES (30) at the 60 fps
115 // reference delta, in seconds so the onset does not depend on the frame rate.
116 static constexpr float NEEDED_SET_STALL_THRESHOLD_SECONDS = 0.5f;
117 struct NeededSetProgressInput {
118 uint32_t needed_chunks = 0;

122 uint32_t in_flight_loads = 0;
123 float frame_delta_seconds = 0.0f;
124 };
