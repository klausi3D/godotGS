179 // #714: quiesces every live file-backed source reading `p_path` for this object's
180 // lifetime, so a writer can atomically replace that file.
181 //
182 // Windows opens FileAccess without FILE_SHARE_DELETE (file_access_windows.cpp:
183 // _SH_DENYNO / _SH_DENYWR / _SH_DENYRW), so an open reader blocks MoveFileExW
184 // replacement AND the backup-swap fallback rename. Merely dropping the cached
185 // handles is not enough: _acquire_pooled_file() would immediately reopen and re-cache
186 // mid-copy, making reimport intermittent under continuous streaming.
187 //
188 // So this sets each matching source's lock-free `suspend_opens` flag (which stops
189 // NEW opens), takes over the cached handles, and then WAITS for reads already in
190 // flight to release their own Refs before returning to the caller's rename.
191 //
192 // It deliberately does NOT hold file_mutex while waiting. A read already past
193 // _acquire_pooled_file() still needs that mutex to finish -- _record_io_counters()
194 // takes it while the reader's Ref is alive -- so holding it would block the very
195 // reads being drained and make the timeout certain instead of rare.
196 //
197 // LOCK ORDER: registry mutex -> instance file_mutex, and file_mutex is only ever
198 // taken briefly, never across the wait. The registry mutex IS held for the
199 // guard's lifetime, purely so a source cannot be destroyed mid-drain.
200 //
201 // Scope it to the RENAME ONLY, never across the copy: the wait is bounded, but
202 // the copy is file-sized.
203 // The wait is bounded, so it can expire with a reader still holding the file open
204 // (a read slower than the drain budget, or one that never returns). That outcome
205 // is reported by drained() and is part of the contract rather than a caveat:
206 // gs_atomic_file_write() refuses the replace instead of attempting a rename whose
207 // precondition is known to be violated, so the timeout tunes how long an import
208 // may stall, NOT whether the result is correct.
209 //
210 // WINDOWS ONLY in effect. The constraint being worked around is Windows-specific
211 // (FileAccessWindows opens via _wfsopen with _SH_DENY*, none of which share delete),
212 // whereas POSIX rename() replaces the destination with readers still open -- they
213 // keep the old inode until they close. On POSIX the constructor is therefore an
214 // early-out no-op that reports drained; suspending and draining there would burn up
215 // to the drain budget per reimport and could refuse a replace that would have
216 // succeeded.
217 class ScopedReaderSuspend {
218 LocalVector<StagedFileChunkPayloadSource *> held;
219 // False when constructed with an empty path, which locks nothing at all.
220 bool registry_locked = false;
221 // True unless the bounded wait expired with a handle still in flight. Starts
222 // true so the "nothing to drain" case (no matching source) is drained.
223 bool readers_drained = true;
224
225 public:
226 explicit ScopedReaderSuspend(const String &p_path);
227 ~ScopedReaderSuspend();
228
229 ScopedReaderSuspend(const ScopedReaderSuspend &) = delete;
230 ScopedReaderSuspend &operator=(const ScopedReaderSuspend &) = delete;
231
232 uint32_t suspended_count() const { return held.size(); }
233 // False => at least one reader still had the destination open when the wait
234 // expired. The caller must NOT attempt the replace; see gs_atomic_file_writer.h.
235 bool drained() const { return readers_drained; }
236 };
