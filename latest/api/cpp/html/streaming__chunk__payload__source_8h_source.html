129 // #714: quiesces every live file-backed source reading `p_path` for this object's
130 // lifetime, so a writer can atomically replace that file.
131 //
132 // Windows opens FileAccess without FILE_SHARE_DELETE (file_access_windows.cpp:
133 // _SH_DENYNO / _SH_DENYWR / _SH_DENYRW), so an open reader blocks MoveFileExW
134 // replacement AND the backup-swap fallback rename. Merely dropping the cached
135 // handles is not enough: _get_thread_file() would immediately reopen and re-cache
136 // mid-copy, making reimport intermittent under continuous streaming.
137 //
138 // So this sets each matching source's lock-free `suspend_opens` flag (which stops
139 // NEW opens), takes over the cached handles, and then WAITS for reads already in
140 // flight to release their own Refs before returning to the caller's rename.
141 //
142 // It deliberately does NOT hold file_mutex while waiting. A read already past
143 // _get_thread_file() still needs that mutex to finish -- _record_io_counters()
144 // takes it while the reader's Ref is alive -- so holding it would block the very
145 // reads being drained and make the timeout certain instead of rare.
146 //
147 // LOCK ORDER: registry mutex -> instance file_mutex, and file_mutex is only ever
148 // taken briefly, never across the wait. The registry mutex IS held for the
149 // guard's lifetime, purely so a source cannot be destroyed mid-drain.
150 //
151 // Scope it to the RENAME ONLY, never across the copy: the wait is bounded, but
152 // the copy is file-sized.
153 // The wait is bounded, so it can expire with a reader still holding the file open
154 // (a read slower than the drain budget, or one that never returns). That outcome
155 // is reported by drained() and is part of the contract rather than a caveat:
156 // gs_atomic_file_write() refuses the replace instead of attempting a rename whose
157 // precondition is known to be violated, so the timeout tunes how long an import
158 // may stall, NOT whether the result is correct.
159 //
160 // WINDOWS ONLY in effect. The constraint being worked around is Windows-specific
161 // (FileAccessWindows opens via _wfsopen with _SH_DENY*, none of which share delete),
162 // whereas POSIX rename() replaces the destination with readers still open -- they
163 // keep the old inode until they close. On POSIX the constructor is therefore an
164 // early-out no-op that reports drained; suspending and draining there would burn up
165 // to the drain budget per reimport and could refuse a replace that would have
166 // succeeded.
167 class ScopedReaderSuspend {
168 LocalVector<StagedFileChunkPayloadSource *> held;
169 // False when constructed with an empty path, which locks nothing at all.
170 bool registry_locked = false;
171 // True unless the bounded wait expired with a handle still in flight. Starts
172 // true so the "nothing to drain" case (no matching source) is drained.
173 bool readers_drained = true;
174
175 public:
176 explicit ScopedReaderSuspend(const String &p_path);
177 ~ScopedReaderSuspend();
178
179 ScopedReaderSuspend(const ScopedReaderSuspend &) = delete;
180 ScopedReaderSuspend &operator=(const ScopedReaderSuspend &) = delete;
181
182 uint32_t suspended_count() const { return held.size(); }
183 // False => at least one reader still had the destination open when the wait
184 // expired. The caller must NOT attempt the replace; see gs_atomic_file_writer.h.
185 bool drained() const { return readers_drained; }
186 };
