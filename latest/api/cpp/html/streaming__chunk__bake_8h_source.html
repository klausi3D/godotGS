14// Per-chunk spatial bookkeeping baked at import time so runtime
15// _build_chunks_for_data() can skip the per-splat center / bounds pass on

25// 4 (start_idx) + 4 (count) + 12 (center) + 4 (max_radius) + 24 (bounds) = 48 bytes in
26// single-precision (real_t = float) builds. Double-precision builds widen Vector3/AABB
27// and naturally bump this number; the bake format is a per-build runtime cache so
28// drift across precision builds is acceptable, but drift within a precision flavor
29// would silently corrupt loads.
30#ifndef REAL_T_IS_DOUBLE
31static_assert(sizeof(StreamingChunkBakeRecord) == 48,
32 "StreamingChunkBakeRecord layout drifted; bake on-disk format would silently break.");
33#endif
