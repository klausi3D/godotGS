79// Represents a single inflated gaussian. Each gaussian has 344 bytes (position, rotation, scale,
80// color, alpha, and 24 SH coeffs x 3 channels). Although the data is easier to interpret in this
81// format, it is not more precise than the packed format, since it was inflated.
82struct UnpackedGaussian {
83 std::array<float, 3> position; // x, y, z
84 std::array<float, 4> rotation; // x, y, z, w
85 std::array<float, 3> scale; // std::log(scale)
86 std::array<float, 3> color; // rgb sh0 encoding
87 float alpha; // inverse logistic
88 std::array<float, SH_MAX_COEFFS> shR;
89 std::array<float, SH_MAX_COEFFS> shG;
90 std::array<float, SH_MAX_COEFFS> shB;
91};

93// Represents a single low precision gaussian. Each gaussian has exactly 92 bytes (for degree 4
94// spherical harmonics); the struct layout is fixed so that at() can index into non-interleaved buffers.
95struct PackedGaussian {
96 std::array<uint8_t, 9> position{};
97 std::array<uint8_t, 4> rotation{};
98 std::array<uint8_t, 3> scale{};
99 std::array<uint8_t, 3> color{};
100 uint8_t alpha = 0;
101 std::array<uint8_t, SH_MAX_COEFFS> shR{};
102 std::array<uint8_t, SH_MAX_COEFFS> shG{};
103 std::array<uint8_t, SH_MAX_COEFFS> shB{};
104
105 UnpackedGaussian unpack(
106 bool usesFloat16, bool usesQuaternionSmallestThree, int32_t fractionalBits, const CoordinateConverter &c) const;
107};

109// Represents a full splat with lower precision. Each splat has at most 92 bytes (for degree 4
110// spherical harmonics), although splats with fewer spherical harmonics degrees will have less.
111// The data is stored non-interleaved.
112struct PackedGaussians {
113 uint32_t version = LATEST_SPZ_HEADER_VERSION; // Version of the packed format
114 int32_t numPoints = 0; // Total number of points (gaussians)
115 int32_t shDegree = 0; // Degree of spherical harmonics
116 int32_t fractionalBits = 0; // Number of bits used for fractional part of fixed-point coords
117 bool antialiased = false; // Whether gaussians should be rendered with mip-splat antialiasing
118 bool usesQuaternionSmallestThree = true; // Whether gaussians use the smallest three method to store quaternions
119 bool hadSkippedExtensions = false; // True when extensions were present in the file but ignored at load time
120
121 std::vector<uint8_t> positions;
122 std::vector<uint8_t> scales;
123 std::vector<uint8_t> rotations;
124 std::vector<uint8_t> alphas;
125 std::vector<uint8_t> colors;
126 std::vector<uint8_t> sh;
127
128#ifdef SPZ_BUILD_EXTENSIONS
129 std::vector<SpzExtensionBasePtr> extensions; // List of extensions, if any
130#endif
131
132 bool usesFloat16() const;
133 PackedGaussian at(int32_t i) const;
134 UnpackedGaussian unpack(int32_t i, const CoordinateConverter &c) const;
135};
