4// Splat shadow casters on the HLOD route (#1095, #1163; docs/architecture/adr-one-route-many-producers.md, F5 "light
5// views"). Device-free planning for the light-view caster pass, shared by the renderer and the CPU doctests.
6//
7// Caster list: the main view's HLOD draw list, exactly as that view uploaded it (world records of its cut plus the
8// admitted node instances' records, one DrawnNodeGPU per node or chunk). Every caster the shadow map receives is a splat
9// the color view drew; objects the budget rule refused cast nothing, and nothing outside the view's cut casts (the
10// light-view cut is a later slice). Forward+ renders the shadow passes of a frame BEFORE that frame's GS main view, so a
11// shadow pass consumes the record of the previous main view: the draw list, rows and atlas the GPU still holds, with
12// camera offsets formed against that view's camera position (recorded here).
13//
14// Representation: depth only, written straight into Godot's directional shadow atlas rect with Godot's own projection
15// and depth convention (reverse-Z, depth correction, pancake clamp), so every mesh receiver samples it unchanged. A
16// fragment of a splat is an occluder when its Gaussian alpha reaches kAlphaThreshold.

30// alpha = opacity * exp(-r^2 / 2) >= threshold marks an occluding fragment (threshold coverage, #1095 section 1).
31static constexpr float kAlphaThreshold = 0.5f;
32// Godot's pancake clamp for depth casters (scene_forward_clustered.glsl, MODE_RENDER_DEPTH).
33static constexpr float kPancakeDepth = 0.9999f;
34// Low-pass dilation added to the projected 2D covariance, in shadow-map texels squared (as the color raster does).
35static constexpr float kLowPassTexels2 = 0.3f;
36// Instance slots one draw may issue. Slots are draw_count * max_chunk_splats (the depth pass's 2D dispatch, flattened).
37static constexpr uint64_t kMaxInstanceSlots = uint64_t(1) << 31;

39enum class Refusal : uint8_t {
40 NONE = 0,
41 NOT_DIRECTIONAL, // Spot / omni passes: perspective light projections are not cast yet.
42 NO_ACTIVE_HLOD_VIEW, // No main view published an active HLOD draw list since the bundle was (re)built.
43 QUANTIZED_PAYLOAD, // The caster shader reads the raw relative payload only.
44 EMPTY_DRAW_LIST,
45 INSTANCE_SLOT_OVERFLOW,
46 BUNDLE_CHANGED, // The relative bundle's buffers are not the ones the record names (rebuilt or released).
47 PIPELINE_UNAVAILABLE,
48 INVALID_ATLAS_RECT,
49};

75// What the last main view uploaded (written on the render thread by prepare_hlod_frame, read by the shadow passes).
76struct FrameRecord {
77 bool valid = false;
78 bool quantized = false;
79 uint32_t draw_count = 0; // World records + instance records in the uploaded list.
80 uint32_t world_records = 0;
81 uint32_t instance_records = 0;
82 uint32_t max_chunk_splats = 0; // Largest node/chunk payload of the bundle (the per-record slot stride).
83 uint64_t world_splats = 0; // Splats of the world records (sum of the selected nodes' payloads).
84 uint64_t instance_splats = 0; // Splats of the admitted objects whose rows cast (refused objects are not in the list).
85 Vector3 camera; // Camera position the list's offsets were formed against (origin 0: the route refuses a shift).
86 uint64_t origin_revision = 0;
87 uint64_t renderer_frame = 0;
88 uint64_t bundle_builds = 0; // ResourceState::hlod_bundle_builds when recorded.
89 // ResourceState::hlod_draw_list_uploads when the list was uploaded. A later upload rewrites the list the record names;
90 // an upload-free rebind (the streamed world's end-of-frame idle restore zeroes only the GPU counter) does not.
91 uint64_t draw_list_upload = 0;
92 // Per bundle row (InstanceDataGPU index): 1 = the row casts. Row 0 follows the world's own cast_shadow (Godot gives the
93 // shared renderer a shadow pass when ANY of its Gaussian instances casts, so that alone says nothing about the
94 // world); rows 1..N follow their node's cast_shadow.
95 LocalVector<uint32_t> row_casts;
96
97 uint64_t casters() const { return world_splats + instance_splats; }
98};

100// Per-row cast flags for the rows as uploaded (row 0 = the world). Row 0 casts iff the world casts; a node row casts iff
101// its stable key is among the director's visible && casts_shadow records.
102inline void build_row_casts(const LocalVector<uint64_t> &p_row_keys, bool p_world_casts,
103 const LocalVector<uint64_t> &p_caster_keys, LocalVector<uint32_t> &r_row_casts) {
104 r_row_casts.resize(p_row_keys.size());
105 for (uint32_t r = 0; r < p_row_keys.size(); ++r) {
106 uint32_t casts = r == 0 ? (p_world_casts ? 1u : 0u) : 0u;
107 for (uint32_t k = 0; r != 0 && k < p_caster_keys.size() && !casts; ++k) {
108 casts = p_caster_keys[k] == p_row_keys[r] ? 1u : 0u;
109 }
110 r_row_casts[r] = casts;
111 }
112}

114// Splats of the admitted objects whose rows cast: whole assets (the budget rule admits or refuses an object whole), so
115// the caster set is exactly what the color view draws minus the rows with cast_shadow off; a refused object casts
116// nothing.
117inline uint64_t admitted_instance_splats(const gs_hlod_bundle::InstanceRegion &p_region,
118 const gs_hlod_bundle::ObjectDemand *p_demands, uint32_t p_demand_count, const gs_hlod_bundle::Admission &p_admission,
119 const LocalVector<uint32_t> &p_row_casts) {
120 uint64_t splats = 0;
121 for (uint32_t index : p_admission.admitted) {
122 if (index >= p_demand_count || p_demands[index].asset_index >= p_region.assets.size()) {
123 continue;
124 }
125 const uint32_t row = p_demands[index].row;
126 if (row < p_row_casts.size() && p_row_casts[row] != 0) {
127 splats += p_region.assets[p_demands[index].asset_index].splat_count;
128 }
129 }
130 return splats;
131}

133// Completes a record for the list a main view uploaded: `p_world_records` world records (their payloads summing to
134// `p_world_record_splats`) followed by the instance records the record already counts (instance_records,
135// instance_splats, row_casts). The world's splats are casters only when row 0 casts. An idle streamed world uploads
136// its instance records alone (p_world_records = 0): the objects it still draws keep casting (ADR 4.4).
137inline void complete_record(FrameRecord &r_record, uint32_t p_world_records, uint64_t p_world_record_splats) {
138 const bool world_casts = !r_record.row_casts.is_empty() && r_record.row_casts[0] != 0;
139 r_record.world_records = p_world_records;
140 r_record.world_splats = world_casts ? p_world_record_splats : 0;
141 r_record.draw_count = p_world_records + r_record.instance_records;
142 r_record.valid = true;
143}

153// One light pass. The light type follows from the projection Godot hands the pass: directional cascades are orthographic.
154inline PassPlan plan_pass(const FrameRecord &p_record, bool p_orthographic_light, bool p_bundle_current) {
155 PassPlan plan;
156 if (!p_orthographic_light) {
157 plan.refusal = Refusal::NOT_DIRECTIONAL;
158 } else if (!p_record.valid) {
159 plan.refusal = Refusal::NO_ACTIVE_HLOD_VIEW;
160 } else if (!p_bundle_current) {
161 plan.refusal = Refusal::BUNDLE_CHANGED;
162 } else if (p_record.quantized) {
163 plan.refusal = Refusal::QUANTIZED_PAYLOAD;
164 } else if (p_record.draw_count == 0 || p_record.max_chunk_splats == 0 || p_record.casters() == 0) {
165 plan.refusal = Refusal::EMPTY_DRAW_LIST;
166 } else {
167 const uint64_t slots = uint64_t(p_record.draw_count) * uint64_t(p_record.max_chunk_splats);
168 if (slots > kMaxInstanceSlots) {
169 plan.refusal = Refusal::INSTANCE_SLOT_OVERFLOW;
170 } else {
171 plan.instance_slots = uint32_t(slots);
172 plan.casters = p_record.casters();
173 }
174 }
175 return plan;
176}

178// Godot's shadow-pass clip transform (render_forward_clustered.cpp): _render_shadow_pass hands _render_shadow_append
179// flip_y = true for a directional pass, _render_shadow_append sets scene_data.flip_y = !p_flip_y, and
180// RenderSceneDataRD::update_ubo applies set_depth_correction(scene_data.flip_y). So mesh casters of a directional
181// cascade use correction(false): clip = correction(!p_flip_y) * light_projection * light_transform^-1 * world.
182// The GS hook passes the same p_flip_y (true for directional). The receivers sample with
183// rectm * bias * correction(false, true, false) * camera * modelview (light_storage.cpp), which this matches texel for
184// texel. A correction(true) here mirrored every cascade about its light-space centre line (#1095 slice 1 review).
185inline Projection godot_shadow_clip(const Projection &p_light_projection, const Transform3D &p_light_transform, bool p_flip_y) {
186 Projection correction;
187 correction.set_depth_correction(!p_flip_y);
188 return correction * p_light_projection * Projection(p_light_transform.affine_inverse());
189}
190
191// The same transform for positions relative to the frame camera C (world = p + C), as the HLOD draw list stores them.
192inline Projection frame_relative_shadow_clip(const Projection &p_light_projection, const Transform3D &p_light_transform,
193 bool p_flip_y, const Vector3 &p_frame_camera) {
194 return godot_shadow_clip(p_light_projection, p_light_transform, p_flip_y) * Projection(Transform3D(Basis(), p_frame_camera));
195}

197// Where Godot's directional receivers sample a world point: atlas-normalised uv and depth of the shadow matrix of
198// light_storage.cpp (rectm * bias * correction(false, true, false) * camera * light_transform^-1), for a cascade rect
199// in normalised atlas coordinates. Used by the CPU tests to check the caster against the receiver.
200inline Vector3 godot_directional_receiver_sample(const Projection &p_light_projection, const Transform3D &p_light_transform,
201 const Rect2 &p_atlas_rect_normalized, const Vector3 &p_world) {
202 Projection correction;
203 correction.set_depth_correction(false, true, false);
204 Projection bias;
205 bias.set_light_bias();
206 Projection rectm;
207 rectm.set_light_atlas_rect(p_atlas_rect_normalized);
208 const Projection shadow = rectm * bias * correction * p_light_projection * Projection(p_light_transform.affine_inverse());
209 return shadow.xform(p_world);
210}

212// Where the caster writes a clip-space point: the draw list's viewport is the cascade rect (RenderingDevice
213// draw_list_begin region), framebuffer row 0 at ndc y = -1, so uv = rect.pos + (ndc.xy * 0.5 + 0.5) * rect.size.
214inline Vector3 caster_atlas_sample(const Projection &p_clip, const Rect2 &p_atlas_rect_normalized, const Vector3 &p_point) {
215 const Vector3 ndc = p_clip.xform(p_point);
216 return Vector3(p_atlas_rect_normalized.position.x + (ndc.x * 0.5f + 0.5f) * p_atlas_rect_normalized.size.x,
217 p_atlas_rect_normalized.position.y + (ndc.y * 0.5f + 0.5f) * p_atlas_rect_normalized.size.y, ndc.z);
218}

220// Pancake: casters in front of the cascade's near plane are clamped, never clipped (Godot's mesh casters do the same).
221inline float pancake_depth(float p_ndc_z) {
222 return p_ndc_z >= kPancakeDepth ? kPancakeDepth : p_ndc_z;
223}

225// A caster fragment's depth, as gs_shadow_caster.glsl's fragment stage writes it: the centre depth plus the mean-depth
226// plane offset p_plane_dz (interpolated unclamped, so the plane is exact), capped per fragment at the Gaussian's own
227// depth extent p_cap. Capping at the quad corners instead bent the plane and flattened its slope inside the ellipse.
228inline float caster_fragment_depth(float p_center, float p_plane_dz, float p_cap, bool p_pancake_pass) {
229 float depth = p_center + std::fmax(-p_cap, std::fmin(p_plane_dz, p_cap));
230 if (p_pancake_pass) {
231 depth = pancake_depth(depth);
232 }
233 return std::fmax(0.0f, std::fmin(depth, 1.0f));
234}

236// Eigenvalues of the dilated 2D footprint covariance [[a + w, b], [b, c + w]] (a, b, c without the low-pass w), in
237// float32 exactly as gs_shadow_caster.glsl computes them. The minor one is det / major, never mid - disc: for an
238// elongated splat mid - disc cancels catastrophically (a = 26214400, c ~ 0.3 gave 1e-8 instead of 0.3) and the quad
239// collapses to no fragments. det adds w's contribution separately from the cancelling a * c - b * b, and the dilated
240// covariance's minor eigenvalue is >= w by construction, so w is an exact floor.
241struct FootprintEigen {
242 float major = 0.0f;
243 float minor = 0.0f;
244};
245
246inline FootprintEigen footprint_eigenvalues(float p_a, float p_b, float p_c, float p_low_pass) {
247 const float a = p_a + p_low_pass;
248 const float c = p_c + p_low_pass;
249 const float mid = 0.5f * (a + c);
250 const float disc = std::sqrt(std::fmax(mid * mid - (a * c - p_b * p_b), 0.0f));
251 FootprintEigen e;
252 e.major = mid + disc;
253 const float det = std::fmax(p_a * p_c - p_b * p_b, 0.0f) + p_low_pass * (p_a + p_c) + p_low_pass * p_low_pass;
254 e.minor = e.major > 0.0f ? std::fmax(det / e.major, std::fmax(p_low_pass, 1e-8f)) : 0.0f;
255 return e;
256}

258// The caster's mean-depth slope g = S'^-1 c (texel offset -> light depth), as gs_shadow_caster.glsl computes it:
259// through the dilated footprint's eigenbasis (footprint_eigenvalues and the shader's axis0), never through the
260// adjugate / det, which cancels to det = 0 in float32 for a needle tilted across both axes and drops the plane. S' is
261// the dilated covariance the quad already uses (minor eigenvalue >= the low-pass), so g is the conditional mean slope
262// of the blurred Gaussian and stays bounded.
263struct DepthGradient {
264 float x = 0.0f;
265 float y = 0.0f;
266};
267
268inline DepthGradient caster_depth_gradient(float p_a, float p_b, float p_c, float p_low_pass, float p_cx, float p_cy) {
269 const FootprintEigen e = footprint_eigenvalues(p_a, p_b, p_c, p_low_pass);
270 DepthGradient g;
271 if (!(e.major > 0.0f) || !(e.minor > 0.0f)) {
272 return g;
273 }
274 const float a = p_a + p_low_pass;
275 const float c = p_c + p_low_pass;
276 float ax = 0.0f;
277 float ay = 0.0f;
278 if (std::fabs(p_b) > 1e-12f) {
279 // The formulation without cancellation: (major - c, b) when a dominates, else (b, major - a).
280 ax = a >= c ? e.major - c : p_b;
281 ay = a >= c ? p_b : e.major - a;
282 const float len = std::sqrt(ax * ax + ay * ay);
283 ax /= len;
284 ay /= len;
285 } else if (a >= c) {
286 ax = 1.0f;
287 } else {
288 ay = 1.0f;
289 }
290 const float p0 = (ax * p_cx + ay * p_cy) / e.major;
291 const float p1 = (-ay * p_cx + ax * p_cy) / e.minor;
292 g.x = ax * p0 - ay * p1;
293 g.y = ay * p0 + ax * p1;
294 return g;
295}

297// Per-renderer-frame caster accounting (stats shadow_caster_*). Shadow passes of a frame run before its main view, at
298// renderer frame counter F; get_render_stats() after that frame shows frame_count F + 1, so shadow_caster_frame is
299// frame_count - 1 exactly like sort_dispatch_frame.
300struct FrameStats {
301 uint64_t frame = UINT64_MAX;
302 uint32_t passes_cast = 0;
303 uint32_t passes_refused = 0;
304 uint64_t casters = 0; // Sum over the frame's cast passes (a caster counts once per cascade it is drawn into).
305 uint64_t instance_slots = 0;
306 Refusal last_refusal = Refusal::NONE;
307 uint64_t total_passes_cast = 0;
308 uint64_t total_passes_refused = 0;
309
310 void record(uint64_t p_frame, const PassPlan &p_plan) {
311 if (frame != p_frame) {
312 frame = p_frame;
313 passes_cast = passes_refused = 0;
314 casters = instance_slots = 0;
315 last_refusal = Refusal::NONE;
316 }
317 if (p_plan.draws()) {
318 ++passes_cast;
319 ++total_passes_cast;
320 casters += p_plan.casters;
321 instance_slots += p_plan.instance_slots;
322 } else {
323 ++passes_refused;
324 ++total_passes_refused;
325 last_refusal = p_plan.refusal;
326 }
327 }
