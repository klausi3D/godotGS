4// Splat shadow casters on the HLOD route (#1095, #1163; docs/architecture/adr-one-route-many-producers.md, F5 "light
5// views"). Device-free planning for the light-view caster pass, shared by the renderer and the CPU doctests.
6//
7// Caster list: the main view's HLOD draw list, exactly as that view uploaded it (world records of its cut plus the
8// admitted node instances' records, one DrawnNodeGPU per node or chunk). Every caster the shadow map receives is a splat
9// the color view drew; objects the budget rule refused cast nothing, and nothing outside the view's cut casts (the
10// light-view cut is a later slice). Forward+ renders the shadow passes of a frame BEFORE that frame's GS main view, so a
11// shadow pass consumes the record of the previous main view: the draw list, rows and atlas the GPU still holds, with
12// camera offsets formed against that view's camera position (recorded here).
13//
14// Representation: depth only, written straight into Godot's directional shadow atlas rect with Godot's own projection
15// and depth convention (reverse-Z, depth correction, pancake clamp), so every mesh receiver samples it unchanged. A
16// fragment of a splat is an occluder when its Gaussian alpha reaches kAlphaThreshold.

76// What the last main view uploaded (written on the render thread by prepare_hlod_frame, read by the shadow passes).
77struct FrameRecord {
78 bool valid = false;
79 bool quantized = false;
80 uint32_t draw_count = 0; // World records + instance records in the uploaded list.
81 uint32_t world_records = 0;
82 uint32_t instance_records = 0;
83 uint32_t max_chunk_splats = 0; // Largest node/chunk payload of the bundle (the per-record slot stride).
84 uint64_t world_splats = 0; // Splats of the world records (sum of the selected nodes' payloads).
85 uint64_t instance_splats = 0; // Splats of the admitted objects whose rows cast (refused objects are not in the list).
86 Vector3 camera; // Camera position the list's offsets were formed against (origin 0: the route refuses a shift).
87 uint64_t origin_revision = 0;
88 uint64_t renderer_frame = 0;

94 // shared renderer a shadow pass when ANY of its Gaussian instances casts, so that alone says nothing about the

96 LocalVector<uint32_t> row_casts;

98 uint64_t casters() const { return world_splats + instance_splats; }

101// Per-row cast flags for the rows as uploaded (row 0 = the world). Row 0 casts iff the world casts; a node row casts iff
102// its stable key is among the director's visible && casts_shadow records.
103inline void build_row_casts(const LocalVector<uint64_t> &p_row_keys, bool p_world_casts,
104 const LocalVector<uint64_t> &p_caster_keys, LocalVector<uint32_t> &r_row_casts) {
105 r_row_casts.resize(p_row_keys.size());
106 for (uint32_t r = 0; r < p_row_keys.size(); ++r) {
107 uint32_t casts = r == 0 ? (p_world_casts ? 1u : 0u) : 0u;
108 for (uint32_t k = 0; r != 0 && k < p_caster_keys.size() && !casts; ++k) {
109 casts = p_caster_keys[k] == p_row_keys[r] ? 1u : 0u;
110 }
111 r_row_casts[r] = casts;
112 }
113}

115// Splats of the admitted objects whose rows cast: whole assets (the budget rule admits or refuses an object whole), so
116// the caster set is exactly what the color view draws minus the rows with cast_shadow off; a refused object casts
117// nothing.
118inline uint64_t admitted_instance_splats(const gs_hlod_bundle::InstanceRegion &p_region,
119 const gs_hlod_bundle::ObjectDemand *p_demands, uint32_t p_demand_count, const gs_hlod_bundle::Admission &p_admission,
120 const LocalVector<uint32_t> &p_row_casts) {
121 uint64_t splats = 0;
122 for (uint32_t index : p_admission.admitted) {
123 if (index >= p_demand_count || p_demands[index].asset_index >= p_region.assets.size()) {
124 continue;
125 }
126 const uint32_t row = p_demands[index].row;
127 if (row < p_row_casts.size() && p_row_casts[row] != 0) {
128 splats += p_region.assets[p_demands[index].asset_index].splat_count;
129 }
130 }
131 return splats;
132}

134// Completes a record for the list a main view uploaded: `p_world_records` world records (their payloads summing to
135// `p_world_record_splats`) followed by the instance records the record already counts (instance_records,
136// instance_splats, row_casts). The world's splats are casters only when row 0 casts. An idle streamed world uploads
137// its instance records alone (p_world_records = 0): the objects it still draws keep casting (ADR 4.4).
138inline void complete_record(FrameRecord &r_record, uint32_t p_world_records, uint64_t p_world_record_splats) {
139 const bool world_casts = !r_record.row_casts.is_empty() && r_record.row_casts[0] != 0;
140 r_record.world_records = p_world_records;
141 r_record.world_splats = world_casts ? p_world_record_splats : 0;
142 r_record.draw_count = p_world_records + r_record.instance_records;
143 r_record.valid = true;
144}

180// One light pass. The caller must check both the explicit light type and the projection for directional eligibility.
181inline PassPlan plan_pass(const FrameRecord &p_record, bool p_orthographic_light, bool p_bundle_current) {
182 PassPlan plan;
183 if (!p_orthographic_light) {
184 plan.refusal = Refusal::NOT_DIRECTIONAL;
185 } else if (!p_record.valid) {
186 plan.refusal = Refusal::NO_ACTIVE_HLOD_VIEW;
187 } else if (!p_bundle_current) {
188 plan.refusal = Refusal::BUNDLE_CHANGED;
189 } else if (p_record.quantized) {
190 plan.refusal = Refusal::QUANTIZED_PAYLOAD;
191 } else if (p_record.draw_count == 0 || p_record.max_chunk_splats == 0 || p_record.casters() == 0) {
192 plan.refusal = Refusal::EMPTY_DRAW_LIST;
193 } else {
194 const uint64_t slots = uint64_t(p_record.draw_count) * uint64_t(p_record.max_chunk_splats);
195 if (slots > kMaxInstanceSlots) {
196 plan.refusal = Refusal::INSTANCE_SLOT_OVERFLOW;
197 } else {
198 plan.instance_slots = uint32_t(slots);
199 plan.casters = p_record.casters();
200 }
201 }
202 return plan;
203}

205// Godot's shadow-pass clip transform (render_forward_clustered.cpp): _render_shadow_pass hands _render_shadow_append
206// flip_y = true for a directional pass, _render_shadow_append sets scene_data.flip_y = !p_flip_y, and
207// RenderSceneDataRD::update_ubo applies set_depth_correction(scene_data.flip_y). So mesh casters of a directional
208// cascade use correction(false): clip = correction(!p_flip_y) * light_projection * light_transform^-1 * world.
209// The GS hook passes the same p_flip_y (true for directional). The receivers sample with
210// rectm * bias * correction(false, true, false) * camera * modelview (light_storage.cpp), which this matches texel for
211// texel. A correction(true) here mirrored every cascade about its light-space centre line (#1095 slice 1 review).
212inline Projection godot_shadow_clip(const Projection &p_light_projection, const Transform3D &p_light_transform, bool p_flip_y) {
213 Projection correction;
214 correction.set_depth_correction(!p_flip_y);
215 return correction * p_light_projection * Projection(p_light_transform.affine_inverse());
216}

218// The same transform for positions relative to the frame camera C (world = p + C), as the HLOD draw list stores them.
219inline Projection frame_relative_shadow_clip(const Projection &p_light_projection, const Transform3D &p_light_transform,
220 bool p_flip_y, const Vector3 &p_frame_camera) {
221 return godot_shadow_clip(p_light_projection, p_light_transform, p_flip_y) * Projection(Transform3D(Basis(), p_frame_camera));
222}

224// Where Godot's directional receivers sample a world point: atlas-normalised uv and depth of the shadow matrix of
225// light_storage.cpp (rectm * bias * correction(false, true, false) * camera * light_transform^-1), for a cascade rect
226// in normalised atlas coordinates. Used by the CPU tests to check the caster against the receiver.
227inline Vector3 godot_directional_receiver_sample(const Projection &p_light_projection, const Transform3D &p_light_transform,
228 const Rect2 &p_atlas_rect_normalized, const Vector3 &p_world) {
229 Projection correction;
230 correction.set_depth_correction(false, true, false);
231 Projection bias;
232 bias.set_light_bias();
233 Projection rectm;
234 rectm.set_light_atlas_rect(p_atlas_rect_normalized);
235 const Projection shadow = rectm * bias * correction * p_light_projection * Projection(p_light_transform.affine_inverse());
236 return shadow.xform(p_world);
237}

239// Where the caster writes a clip-space point: the draw list's viewport is the cascade rect (RenderingDevice
240// draw_list_begin region), framebuffer row 0 at ndc y = -1, so uv = rect.pos + (ndc.xy * 0.5 + 0.5) * rect.size.
241inline Vector3 caster_atlas_sample(const Projection &p_clip, const Rect2 &p_atlas_rect_normalized, const Vector3 &p_point) {
242 const Vector3 ndc = p_clip.xform(p_point);
243 return Vector3(p_atlas_rect_normalized.position.x + (ndc.x * 0.5f + 0.5f) * p_atlas_rect_normalized.size.x,
244 p_atlas_rect_normalized.position.y + (ndc.y * 0.5f + 0.5f) * p_atlas_rect_normalized.size.y, ndc.z);
245}

247// Pancake: casters in front of the cascade's near plane are clamped, never clipped (Godot's mesh casters do the same).
248inline float pancake_depth(float p_ndc_z) {
249 return p_ndc_z >= kPancakeDepth ? kPancakeDepth : p_ndc_z;
250}

252// A caster fragment's depth, as gs_shadow_caster.glsl's fragment stage writes it: the centre depth plus the mean-depth
253// plane offset p_plane_dz (interpolated unclamped, so the plane is exact), capped per fragment at the Gaussian's own
254// depth extent p_cap. Capping at the quad corners instead bent the plane and flattened its slope inside the ellipse.
255inline float caster_fragment_depth(float p_center, float p_plane_dz, float p_cap, bool p_pancake_pass) {
256 float depth = p_center + std::fmax(-p_cap, std::fmin(p_plane_dz, p_cap));
257 if (p_pancake_pass) {
258 depth = pancake_depth(depth);
259 }
260 return std::fmax(0.0f, std::fmin(depth, 1.0f));
261}

263// Eigenvalues of the dilated 2D footprint covariance [[a + w, b], [b, c + w]] (a, b, c without the low-pass w), in
264// float32 exactly as gs_shadow_caster.glsl computes them. The minor one is det / major, never mid - disc: for an
265// elongated splat mid - disc cancels catastrophically (a = 26214400, c ~ 0.3 gave 1e-8 instead of 0.3) and the quad
266// collapses to no fragments. det adds w's contribution separately from the cancelling a * c - b * b, and the dilated
267// covariance's minor eigenvalue is >= w by construction, so w is an exact floor.
268struct FootprintEigen {
269 float major = 0.0f;
270 float minor = 0.0f;
271};
272
273inline FootprintEigen footprint_eigenvalues(float p_a, float p_b, float p_c, float p_low_pass) {
274 const float a = p_a + p_low_pass;
275 const float c = p_c + p_low_pass;
276 const float mid = 0.5f * (a + c);
277 const float disc = std::sqrt(std::fmax(mid * mid - (a * c - p_b * p_b), 0.0f));
278 FootprintEigen e;
279 e.major = mid + disc;
280 const float det = std::fmax(p_a * p_c - p_b * p_b, 0.0f) + p_low_pass * (p_a + p_c) + p_low_pass * p_low_pass;
281 e.minor = e.major > 0.0f ? std::fmax(det / e.major, std::fmax(p_low_pass, 1e-8f)) : 0.0f;
282 return e;
283}

285// The caster's mean-depth slope g = S'^-1 c (texel offset -> light depth), as gs_shadow_caster.glsl computes it:
286// through the dilated footprint's eigenbasis (footprint_eigenvalues and the shader's axis0), never through the
287// adjugate / det, which cancels to det = 0 in float32 for a needle tilted across both axes and drops the plane. S' is
288// the dilated covariance the quad already uses (minor eigenvalue >= the low-pass), so g is the conditional mean slope
289// of the blurred Gaussian and stays bounded.
290struct DepthGradient {
291 float x = 0.0f;
292 float y = 0.0f;
293};
294
295inline DepthGradient caster_depth_gradient(float p_a, float p_b, float p_c, float p_low_pass, float p_cx, float p_cy) {
296 const FootprintEigen e = footprint_eigenvalues(p_a, p_b, p_c, p_low_pass);
297 DepthGradient g;
298 if (!(e.major > 0.0f) || !(e.minor > 0.0f)) {
299 return g;
300 }
301 const float a = p_a + p_low_pass;
302 const float c = p_c + p_low_pass;
303 float ax = 0.0f;
304 float ay = 0.0f;
305 if (std::fabs(p_b) > 1e-12f) {
306 // The formulation without cancellation: (major - c, b) when a dominates, else (b, major - a).
307 ax = a >= c ? e.major - c : p_b;
308 ay = a >= c ? p_b : e.major - a;
309 const float len = std::sqrt(ax * ax + ay * ay);
310 ax /= len;
311 ay /= len;
312 } else if (a >= c) {
313 ax = 1.0f;
314 } else {
315 ay = 1.0f;
316 }
317 const float p0 = (ax * p_cx + ay * p_cy) / e.major;
318 const float p1 = (-ay * p_cx + ax * p_cy) / e.minor;
319 g.x = ax * p0 - ay * p1;
320 g.y = ay * p0 + ax * p1;
321 return g;
322}

360// Invalid light IDs (legacy direct hook calls) are ignored, including for the frame stamp and no-caster count.
