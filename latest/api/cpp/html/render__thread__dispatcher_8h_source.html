10class IRenderThreadDispatcher {
11public:
12 virtual ~IRenderThreadDispatcher() = default;

14 // Argument-order contract. The dispatcher submits
15 // `p_callable.bind(request_id)` and the render thread calls it with no
16 // arguments. Godot passes call arguments BEFORE bound ones
17 // (CallableCustomBind::call, core/variant/callable_bind.cpp), so when
18 // p_callable already carries binds, e.g.
19 // `callable_mp(this, &T::f).bind(data)`, the dispatcher's request_id is
20 // passed ahead of them: the target is called as f(request_id, data).
21 // Every target therefore takes `uint64_t p_request_id` as its FIRST
22 // parameter, followed by its own bound arguments in bind order, and must
23 // call notify_completed(p_request_id). A target with p_request_id last
24 // receives its arguments swapped and nothing reports the call error. An
25 // int argument becomes the request id and the int is "completed" in its
26 // place, which can push completed_request_id past every outstanding
27 // request so that later blocking dispatches return before their callback
28 // has run. An Object or Transform3D argument converts to null or identity
29 // while p_request_id becomes 0, so that request never completes and the
30 // caller blocks for the full timeout.
31 virtual bool dispatch_call_on_render_thread_blocking(const Callable &p_callable, bool *r_dispatched,
32 bool p_allow_timeout, uint64_t *r_request_id, const char *p_log_prefix) = 0;
33 // Returns true iff a call to dispatch_call_on_render_thread_blocking()
34 // would actually attempt a dispatch (i.e. the early-exit guard would
35 // not short-circuit it). This is the discrimination signal callers need
36 // when reasoning about "is the render-thread dispatch path live?",
37 // because the bool return of dispatch_call_on_render_thread_blocking()
38 // conflates "no dispatch path active" (early-exit returned false) with
39 // "dispatched but timed out waiting for completion" (also returned
40 // false). The fixture in test_renderer_lifetime_proof.h depends on
41 // this distinction to decide whether the renderer dtor will fall
42 // through to the synchronous teardown path. (Codex PR #386 review,
43 // P1.)
44 virtual bool is_render_thread_dispatch_path_active() const = 0;
45 virtual void notify_completed(uint64_t p_request_id) = 0;
46 virtual void set_timeout_usec(uint64_t p_timeout_usec) = 0;
47 virtual uint64_t get_timeout_usec() const = 0;
48 virtual uint64_t get_next_request_id() const = 0;
49 virtual uint64_t get_completed_request_id() const = 0;
50 virtual void promote_latest_data_request_id(uint64_t p_request_id) = 0;
51 virtual uint64_t get_latest_data_request_id() const = 0;
52 virtual void set_latest_data_result(Error p_error) = 0;
53 virtual Error get_latest_data_result() const = 0;
54};
