150// Every member must carry a default initializer. Callers routinely build a
151// Gaussian by declaration ("Gaussian g;") and assign only the fields they care
152// about, so a member without one silently inherits whatever occupied that
153// storage: area and stroke_age are range-checked by
154// GaussianData::_validate_gpu_payload_locked() and a garbage value makes
155// create_gpu_buffer() return a null RID, while painterly_meta and _padding are
156// not validated at all and reach the GPU or the serialized payload verbatim.
157// Godot's Vector2/Vector3/Quaternion/Color zero themselves through their own
158// union initializers, so only the scalars and arrays need one here. See #687.
159struct Gaussian {
160 Vector3 position;
161 float opacity = 0.0f;
162
163 Vector3 scale;
164 float area = 0.0f;
165
166 Quaternion rotation;
167
168 // Spherical harmonics coefficients
169 Color sh_dc; // DC term
170 Vector3 sh_1[3] = {}; // First-order SH
171
172 // 2D Gaussian support (surfels)
173 Vector3 normal;
174 float stroke_age = 0.0f;
175
176 // Padding for alignment
177 float _padding = 0.0f; // Ensures brush_axes is 8-byte aligned
178
179 // Painterly rendering extensions (packed for GPU alignment)
180 Vector2 brush_axes;
181 uint32_t painterly_meta = 0; // lower 16 bits: palette id, upper 16 bits: painterly flags / brush override ids
182
183 // Render metadata and final padding to reach 144 bytes.
184 uint32_t render_meta = 0; // lower bit stores GaussianDCEncoding
185 float _padding2[2] = { 0.0f, 0.0f };
186};

188// NOTE: 144 is this AUTHORING struct's size. It is deliberately no longer the
189// GPU layout size -- `PackedGaussian` is 128 bytes and is the only thing a
190// shader reads. The two were equal by coincidence, never by contract: even
191// while both were 144 the field layouts differed (`Vector3 sh_1[3]` here, 36
192// bytes, against `sh_encoded[12]` there, 48), so this struct was never safe to
193// upload raw. Go through `pack_gaussian()`. Keeping the size pinned is still
194// worthwhile -- it keeps the 16-byte alignment and catches accidental growth --
195// but do not read it as a GPU-compatibility guarantee.
196static_assert(sizeof(Gaussian) % 16 == 0, "Gaussian struct must remain 16-byte aligned");
197static_assert(sizeof(Gaussian) == 144, "Gaussian authoring struct size changed; it is NOT the GPU layout (see PackedGaussian)");
198static_assert(offsetof(Gaussian, brush_axes) % 8 == 0, "Gaussian::brush_axes must stay 8-byte aligned for std430 layout");

200// Persisted layout fingerprint (#1172).
201//
202// `.gsplatworld` (and the `.gsplatcache` PLY sidecar, which is a .gsplatworld)
203// and the GSF GAUSSIAN_DATA chunk store this struct as raw bytes. The size
204// assert above cannot see a same-size change -- swapping painterly_meta and
205// render_meta, or repurposing a padding slot as happened when render_meta took
206// part of _padding2 -- and such a change would misdecode every existing file
207// without any check firing. So every writer records
208// GAUSSIAN_STRUCT_LAYOUT_VERSION in its header and every reader rejects a
209// payload recorded under a different layout.
210//
211// The asserts below pin every member's offset and type. If one fails you are
212// changing the persisted layout or the meaning of a field: bump
213// GAUSSIAN_STRUCT_LAYOUT_VERSION, update the asserts, and decide what happens
214// to files written under the old layout (they are rejected by default). Do not
215// edit the asserts without the bump.
216static constexpr uint32_t GAUSSIAN_STRUCT_LAYOUT_VERSION = 1u;
217
218// Files written before the layout word existed (.gsplatworld v1 and HLOD v2, GSF headers
219// whose layout word is 0) were all written with layout 1. They resolve to that
220// version, so they keep loading today and are rejected as soon as the layout
221// version moves on. Never change this value.
222static constexpr uint32_t GAUSSIAN_STRUCT_LAYOUT_VERSION_UNRECORDED = 1u;
223
224// Layout a persisted payload is decoded with: the recorded word, or the legacy
225// layout when the writer predates the word (0).
226constexpr uint32_t gaussian_resolve_persisted_layout_version(uint32_t p_recorded) {
227 return p_recorded == 0u ? GAUSSIAN_STRUCT_LAYOUT_VERSION_UNRECORDED : p_recorded;
228}
