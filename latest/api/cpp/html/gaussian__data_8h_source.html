149// Every member must carry a default initializer. Callers routinely build a
150// Gaussian by declaration ("Gaussian g;") and assign only the fields they care
151// about, so a member without one silently inherits whatever occupied that
152// storage: area and stroke_age are range-checked by
153// GaussianData::_validate_gpu_payload_locked() and a garbage value makes
154// create_gpu_buffer() return a null RID, while painterly_meta and _padding are
155// not validated at all and reach the GPU or the serialized payload verbatim.
156// Godot's Vector2/Vector3/Quaternion/Color zero themselves through their own
157// union initializers, so only the scalars and arrays need one here. See #687.
158struct Gaussian {
159 Vector3 position;
160 float opacity = 0.0f;
161
162 Vector3 scale;
163 float area = 0.0f;
164
165 Quaternion rotation;
166
167 // Spherical harmonics coefficients
168 Color sh_dc; // DC term
169 Vector3 sh_1[3] = {}; // First-order SH
170
171 // 2D Gaussian support (surfels)
172 Vector3 normal;
173 float stroke_age = 0.0f;
174
175 // Padding for alignment
176 float _padding = 0.0f; // Ensures brush_axes is 8-byte aligned
177
178 // Painterly rendering extensions (packed for GPU alignment)
179 Vector2 brush_axes;
180 uint32_t painterly_meta = 0; // lower 16 bits: palette id, upper 16 bits: painterly flags / brush override ids
181
182 // Render metadata and final padding to reach 144 bytes.
183 uint32_t render_meta = 0; // lower bit stores GaussianDCEncoding
184 float _padding2[2] = { 0.0f, 0.0f };
185};

187// NOTE: 144 is this AUTHORING struct's size. It is deliberately no longer the
188// GPU layout size -- `PackedGaussian` is 128 bytes and is the only thing a
189// shader reads. The two were equal by coincidence, never by contract: even
190// while both were 144 the field layouts differed (`Vector3 sh_1[3]` here, 36
191// bytes, against `sh_encoded[12]` there, 48), so this struct was never safe to
192// upload raw. Go through `pack_gaussian()`. Keeping the size pinned is still
193// worthwhile -- it keeps the 16-byte alignment and catches accidental growth --
194// but do not read it as a GPU-compatibility guarantee.
195static_assert(sizeof(Gaussian) % 16 == 0, "Gaussian struct must remain 16-byte aligned");
196static_assert(sizeof(Gaussian) == 144, "Gaussian authoring struct size changed; it is NOT the GPU layout (see PackedGaussian)");
197static_assert(offsetof(Gaussian, brush_axes) % 8 == 0, "Gaussian::brush_axes must stay 8-byte aligned for std430 layout");
