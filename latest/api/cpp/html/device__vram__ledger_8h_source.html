4// The one device VRAM ledger (ADR adr-beyond-vram-worlds.md D-11, BV-11). It owns the GS share of the device heap, the
5// atlas limit derived from it, the device-wide byte measurement the HLOD route is admitted against, and the split of
6// that measurement by category. Everything else reports through it. The class is pure (no RenderingDevice, no project
7// settings) so that the admission math is testable on the host; renderer/resident_instance_contract_publisher.cpp feeds it.
8//
9// How bytes are counted, so that nothing is counted twice:
10// - observe_device_bytes() is the measured device-wide upper bound (every RD buffer plus the tracked textures). It is
11// the admission basis. The categories only EXPLAIN it: report() replaces a category's live bytes, it never adds, and
12// live bytes are never summed on top of the measurement. What no category explains is `unattributed_bytes`.
13// - Without a measurement (nothing observed yet) the sum of the live categories is the basis.
14// - reserve() pre-flights bytes that are not allocated yet. It replaces the category's own reservation. The caller
15// releases it once the allocation exists or failed, and the next measurement carries the allocated bytes. While the
16// old allocation of a resize is still alive it stays in the measurement, so a reservation of the whole new size
17// pre-flights old + new (I-6).
18//
19// D-11: enforced on the HLOD route, report-only on the chunk route until Step C. CHUNK_POOL never refuses; going over the
20// limit there only counts report_only_overruns. The engine and driver reserves are the rest of the heap (1 - 0.60): the
21// repo has no measurement that splits them, so they are published as one number.
22//
23// Threads: the render thread is the only writer (I-11). The fields are relaxed atomics so that the diagnostics may read a
24// snapshot from any thread; a snapshot is therefore per-field, not a consistent cut of one frame.

45 // The GS-owned share of the device heap (D-11). The remainder is the engine and driver reserve:
46 // get_device_memory_budget() is the device heap, not net of anything.
47 static constexpr double kGsDeviceShare = 0.60;
48 // Below this floor a reported device heap is treated as unknown: a degenerate small-positive value would otherwise
49 // clamp every budget to nonsense.
50 static constexpr uint64_t kMinTrustedDeviceBytes = uint64_t(256) * 1024 * 1024;
51 // Absolute single-upload staging ceiling of the resident atlas; a budget only ever tightens beneath it.
52 static constexpr uint64_t kResidentStagingCeilingBytes = uint64_t(2) * 1024 * 1024 * 1024;
53
54 // Byte cap of a packed resident atlas.
55 // - An unknown or untrusted heap keeps the staging ceiling (do-no-harm on UMA/iGPU/software/non-Vulkan).
56 // - override_mb > 0 pins an explicit cap (low-end clamp + on-capable-GPU validation).
57 // The 0.60 term only binds below ~3.3 GB of device VRAM; above that the staging ceiling dominates.
58 static uint64_t atlas_limit_for_device(uint64_t p_device_bytes, uint32_t p_override_mb,
59 uint64_t p_staging_ceiling_bytes = kResidentStagingCeilingBytes, double p_share = kGsDeviceShare,
60 uint64_t p_min_trusted_bytes = kMinTrustedDeviceBytes) {
61 uint64_t cap = p_staging_ceiling_bytes;
62 if (p_device_bytes >= p_min_trusted_bytes) {
63 const uint64_t share_cap = uint64_t(double(p_device_bytes) * p_share);
64 cap = share_cap < cap ? share_cap : cap;
65 }
66 if (p_override_mb > 0) {
67 const uint64_t override_bytes = uint64_t(p_override_mb) * 1024u * 1024u;
68 cap = override_bytes < cap ? override_bytes : cap;
69 }
70 return cap;
71 }
72
73 // The GS limit. It holds the canonical route's own sort, tile and atlas buffers too, so it cannot be bounded by the
74 // HLOD-owned byte cap; that cap bounds the HLOD bundle on its own. An unknown or untrusted heap size keeps the fixed
75 // cap (fail closed).
76 static uint64_t gs_limit_for_device(uint64_t p_device_bytes, uint64_t p_fixed_cap_bytes, double p_share = kGsDeviceShare,
77 uint64_t p_min_trusted_bytes = kMinTrustedDeviceBytes) {
78 if (p_device_bytes >= p_min_trusted_bytes) {
79 return uint64_t(double(p_device_bytes) * p_share);
80 }
81 if (p_device_bytes == 0) { return p_fixed_cap_bytes; }
82 return p_device_bytes < p_fixed_cap_bytes ? p_device_bytes : p_fixed_cap_bytes;
83 }

109 // Releases its reservation when it goes out of scope, so that no failure path can leak one.
110 class ScopedReservation {
111 DeviceVramLedger &ledger;
112 Category category;
113 bool held = false;
114
115 public:
116 ScopedReservation(DeviceVramLedger &p_ledger, Category p_category) :
117 ledger(p_ledger), category(p_category) {}
118 ScopedReservation(const ScopedReservation &) = delete;
119 ScopedReservation &operator=(const ScopedReservation &) = delete;
120 ~ScopedReservation() {
121 if (held) { ledger.release(category); }
122 }
123 bool acquire(uint64_t p_bytes) {
124 const bool ok = ledger.reserve(category, p_bytes);
125 held = held || ok;
126 return ok;
127 }
128 };

130 // `p_limit_clamp` != 0 lowers the limit (the test hook of the HLOD frame route).
131 void configure(uint64_t p_device_heap_bytes, uint64_t p_fixed_cap_bytes, uint64_t p_limit_clamp = 0) {
132 uint64_t limit = gs_limit_for_device(p_device_heap_bytes, p_fixed_cap_bytes);
133 if (p_limit_clamp != 0 && p_limit_clamp < limit) { limit = p_limit_clamp; }
134 device_heap_bytes.store(p_device_heap_bytes, std::memory_order_relaxed);
135 limit_bytes.store(limit, std::memory_order_relaxed);
136 configured.store(true, std::memory_order_relaxed);
137 }
138 uint64_t limit() const { return limit_bytes.load(std::memory_order_relaxed); }
139 // The atlas limit of the configured heap. The override setting is read by the caller; this class reads no settings.
140 uint64_t atlas_limit(uint32_t p_override_mb) const {
141 return atlas_limit_for_device(device_heap_bytes.load(std::memory_order_relaxed), p_override_mb);
142 }

154 // The bytes the category holds now. Replaces the previous report: the same memory reported again is counted once.
155 void report(Category p_category, uint64_t p_bytes) {
156 live[p_category].store(p_bytes, std::memory_order_relaxed);
157 if (!is_enforced(p_category) && configured.load(std::memory_order_relaxed) && _used() > limit()) {
158 report_only_overruns.fetch_add(1, std::memory_order_relaxed);
159 }
160 }

162 // The HLOD bundles were released: their categories explain nothing any more. The chunk route keeps its own report.
163 void clear_hlod_reports() {
164 for (uint32_t i = 0; i < CATEGORY_COUNT; ++i) {
165 if (is_enforced(Category(i))) { live[i].store(0, std::memory_order_relaxed); }
166 }
167 }

169 // Pre-flight `p_bytes` not yet allocated. Replaces the category's own reservation, so a retry never stacks. True when
170 // it fits next to the measurement and the other categories' reservations. A refused reservation changes nothing but
171 // the refusal counter. An unconfigured ledger has limit 0 and refuses (fail closed). A report-only category always
172 // succeeds and counts the overrun.
173 bool reserve(Category p_category, uint64_t p_bytes) {
174 const uint64_t used = _saturating_add(_basis(), _pending_total(p_category));
175 const uint64_t max_bytes = limit();
176 const bool fits = used <= max_bytes && p_bytes <= max_bytes - used;
177 if (!fits && is_enforced(p_category)) {
178 refused_reservations.fetch_add(1, std::memory_order_relaxed);
179 return false;
180 }
181 if (!fits && configured.load(std::memory_order_relaxed)) {
182 report_only_overruns.fetch_add(1, std::memory_order_relaxed);
183 }
184 pending[p_category].store(p_bytes, std::memory_order_relaxed);
185 return true;
186 }
187 void release(Category p_category) { pending[p_category].store(0, std::memory_order_relaxed); }
188
189 // What can still be reserved. 0 when the limit is already exceeded.
190 uint64_t headroom() const {
191 const uint64_t used = _used();
192 const uint64_t max_bytes = limit();
193 return used < max_bytes ? max_bytes - used : 0;
194 }

196 Snapshot snapshot() const {
197 Snapshot out;
198 out.device_heap_bytes = device_heap_bytes.load(std::memory_order_relaxed);
199 out.limit_bytes = limit();
200 out.engine_driver_reserve_bytes = out.device_heap_bytes > out.limit_bytes ? out.device_heap_bytes - out.limit_bytes : 0;
201 out.device_observed = device_observed.load(std::memory_order_relaxed);
202 out.device_bytes = device_bytes.load(std::memory_order_relaxed);
203 out.admitted_device_bytes = admitted_device_bytes.load(std::memory_order_relaxed);
204 for (uint32_t i = 0; i < CATEGORY_COUNT; ++i) {
205 out.live[i] = live[i].load(std::memory_order_relaxed);
206 out.pending[i] = pending[i].load(std::memory_order_relaxed);
207 out.attributed_bytes = _saturating_add(out.attributed_bytes, out.live[i]);
208 out.pending_bytes = _saturating_add(out.pending_bytes, out.pending[i]);
209 }
210 out.unattributed_bytes = out.device_bytes > out.attributed_bytes ? out.device_bytes - out.attributed_bytes : 0;
211 out.headroom_bytes = headroom();
212 out.refused_reservations = refused_reservations.load(std::memory_order_relaxed);
213 out.report_only_overruns = report_only_overruns.load(std::memory_order_relaxed);
214 return out;
215 }

217private:
218 static uint64_t _saturating_add(uint64_t p_a, uint64_t p_b) { return p_b > UINT64_MAX - p_a ? UINT64_MAX : p_a + p_b; }
219
220 uint64_t _basis() const {
221 if (device_observed.load(std::memory_order_relaxed)) { return device_bytes.load(std::memory_order_relaxed); }
222 uint64_t sum = 0;
223 for (uint32_t i = 0; i < CATEGORY_COUNT; ++i) {
224 sum = _saturating_add(sum, live[i].load(std::memory_order_relaxed));
225 }
226 return sum;
227 }
228 uint64_t _pending_total(int p_except = -1) const {
229 uint64_t sum = 0;
230 for (int i = 0; i < CATEGORY_COUNT; ++i) {
231 if (i != p_except) { sum = _saturating_add(sum, pending[i].load(std::memory_order_relaxed)); }
232 }
233 return sum;
234 }
235 uint64_t _used() const { return _saturating_add(_basis(), _pending_total()); }
