26// Device-free lifecycle policy, consumed by BOTH publisher stale-contract paths.
27// ROWS_ONLY + UNCHANGED needs no upload. Reservation is an input here; allocation,
28// project settings and byte accounting remain the resource owner's responsibility.
29inline ContractChangeDecision hlod_contract_change_action(const ContractChange &p_change) {
30 if (!p_change.world_device_source_current) {
31 return { ContractChangeAction::FULL_REBUILD, ContractChangeReason::WORLD_DEVICE_SOURCE_CHANGED };
32 }
33 if (!p_change.resources_current) {
34 return { ContractChangeAction::FULL_REBUILD, ContractChangeReason::RESOURCES_CHANGED };
35 }
36 if (p_change.instance_refresh_failed_asset_generation != 0 &&
37 p_change.instance_refresh_failed_asset_generation == p_change.asset_generation) {
38 return { ContractChangeAction::FULL_REBUILD, ContractChangeReason::INSTANCE_REFRESH_FAILED };
39 }
40 if (p_change.prepared_asset_generation != p_change.asset_generation) {
41 if (p_change.object_pages > p_change.reserved_object_pages) {
42 return { ContractChangeAction::FULL_REBUILD, ContractChangeReason::OBJECT_TAIL_OVERFLOW };
43 }
44 return { ContractChangeAction::INSTANCE_REGION_ONLY, ContractChangeReason::ASSETS_CHANGED };
45 }
46 return { ContractChangeAction::ROWS_ONLY, p_change.prepared_row_generation == p_change.row_generation ?
47 ContractChangeReason::UNCHANGED : ContractChangeReason::ROWS_CHANGED };
48}

50// The last step may partially write the retained atlas. A failure must discard
51// new tables before tearing down the old owner; it cannot roll back that write.
52template <typename Upload, typename Commit, typename Discard, typename Teardown>
53bool publish_instance_tail(Upload p_upload, Commit p_commit, Discard p_discard, Teardown p_teardown) {
54 if (!p_upload()) {
55 p_discard();
56 p_teardown();
57 return false;
58 }
59 p_commit();
60 return true;
61}

63inline uint64_t object_tail_pages(uint64_t p_pages, uint32_t p_percent) {
64 const uint32_t percent = p_percent > 100 ? 100 : p_percent;
65 return (p_pages / 100) * percent + ((p_pages % 100) * percent) / 100;
66}

73// Percentage base: the existing tree-capped world page budget. Unused atlas
74// headroom supplies that target first; it never enlarges the percentage target.
75inline ObjectTailPlacement place_object_tail(uint64_t p_total_node_pages, uint64_t p_page_budget, uint32_t p_percent) {
76 const uint64_t world = p_total_node_pages < p_page_budget ? p_total_node_pages : p_page_budget;
77 const uint64_t tail = object_tail_pages(world, p_percent);
78 const uint64_t headroom = p_page_budget > p_total_node_pages ? p_page_budget - p_total_node_pages : 0;
79 const uint64_t from_headroom = tail < headroom ? tail : headroom;
80 return { world - (tail - from_headroom), tail };
81}
