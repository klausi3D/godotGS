91 // Build a coarse CPU hierarchy for live candidate generation. GPUCuller still performs the
92 // final per-splat frustum, distance, and budget filtering on the query output.
93 //
94 // #794: returns false ONLY when the build could not allocate, leaving no usable
95 // hierarchy. Empty input is not a failure (nothing to build) and returns true.
96 //
97 // [[nodiscard]] is deliberate. This used to return void, so an allocation failure
98 // left a null root that the caller could not see -- GPUCuller then marked the
99 // hierarchy clean, recorded the source revision, and every later frame took the
100 // hierarchical path against an empty tree, making the whole asset invisible with no
101 // retry until the revision changed. A silently-ignorable status is what allowed
102 // that, so ignoring it is now a compile error.
103 [[nodiscard]] bool build_hierarchy(const Vector<GaussianData>& splats, const BuildParams& params);
104 [[nodiscard]] bool build_hierarchy(const Vector<GaussianData>& splats) { return build_hierarchy(splats, BuildParams()); }

200inline uint32_t HierarchicalSplatStructure::OctreeNode::get_child_index(const Vector3& pos) const {
201 Vector3 center = bounds.get_center();
202 uint32_t index = 0;
203 if (pos.x > center.x) index |= 1;
204 if (pos.y > center.y) index |= 2;
205 if (pos.z > center.z) index |= 4;
206 return index;
207}
