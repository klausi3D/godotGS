17// `p_camera_offset` is DrawRecord::camera_offset: R s c + T - camera, formed in double by
18// gs_world_frame::camera_relative_offset and converted to float last. It is view dependent, so it is written here and
19// never into the shared ChunkMetaGPU payload.
20// `p_level` (BV-04b) is the HLOD level of a world node (its tree height, 0 = leaf); object (instance) records keep
21// GS_DRAWN_NODE_LEVEL_NONE. Only the emit pass's records-per-level counters read it.
22inline DrawnNodeGPU make_drawn_node(uint32_t p_instance_id, uint32_t p_chunk_id, const Vector3 &p_camera_offset,
23 uint32_t p_level = GS_DRAWN_NODE_LEVEL_NONE) {
24 DrawnNodeGPU node = {};
25 node.instance_id = p_instance_id;
26 node.chunk_id = p_chunk_id;
27 node.fade = 1.0f; // S6 cross-fade is not wired; fully opaque.
28 node.level = p_level;
29 node.camera_offset[0] = p_camera_offset.x;
30 node.camera_offset[1] = p_camera_offset.y;
31 node.camera_offset[2] = p_camera_offset.z;
32 node.camera_offset[3] = 0.0f;

36// Frustum planes for the camera-relative depth pass: the planes of the camera-translated frustum (rotation of the
37// camera, no translation). The four side planes pass through the origin (d == 0), which is where the camera sits in
38// that frame; near and far keep d = +-distance (a few metres to km, never the camera's distance from the world
39// origin, which is the large number a float would round).
40inline Vector<Plane> camera_relative_frustum_planes(const Projection &p_projection, const Transform3D &p_camera_to_world) {
41 return p_projection.get_projection_planes(Transform3D(p_camera_to_world.basis, Vector3()));
42}

44// Cache key for state that depends on the HLOD frame (cached sort order, sort reuse). Outside the HLOD route the
45// generation is returned unchanged, so the canonical routes keep their keys. In the HLOD route two different origin
46// revisions give different keys, so a cache built against one origin can never be served against another.
47inline uint64_t origin_keyed_generation(uint64_t p_generation, bool p_hlod_relative, uint64_t p_origin_revision) {
48 if (!p_hlod_relative) {
49 return p_generation;
50 }
51 uint64_t mixed = p_generation;
52 mixed ^= p_origin_revision + 0x9e3779b97f4a7c15ULL + (mixed << 6) + (mixed >> 2);
53 return mixed;
54}
