45inline Gaussian decode_record(const uint8_t *p_bytes) {
46 float fields[31];
47 for (uint32_t i = 0; i < 31; i++) {
48 fields[i] = decode_float(p_bytes + i * 4);
49 }
50 Gaussian g;
51 g.position = Vector3(fields[0], fields[1], fields[2]);
52 g.opacity = fields[3];
53 g.scale = Vector3(fields[4], fields[5], fields[6]);
54 g.area = fields[7];
55 g.rotation = Quaternion(fields[8], fields[9], fields[10], fields[11]);
56 g.sh_dc = Color(fields[12], fields[13], fields[14], fields[15]);
57 for (uint32_t i = 0; i < 3; i++) {
58 g.sh_1[i] = Vector3(fields[16 + i * 3], fields[17 + i * 3], fields[18 + i * 3]);
59 }
60 g.normal = Vector3(fields[25], fields[26], fields[27]);
61 g.stroke_age = fields[28];
62 g.brush_axes = Vector2(fields[29], fields[30]);
63 g.painterly_meta = decode_uint32(p_bytes + 124);
64 g.render_meta = decode_uint32(p_bytes + 128);
65 return g;

68inline Error encode(const ::GaussianData::SaveSnapshot &p_snapshot, PackedByteArray &r_payload) {
69 const Vector<Gaussian> &gaussians = p_snapshot.get_gaussians();
70 const Vector<Vector3> &high_order = p_snapshot.get_sh_high_order();
71 const uint32_t first = p_snapshot.get_sh_first_order_count();
72 const uint32_t high = p_snapshot.get_sh_high_order_count();
73 ERR_FAIL_COND_V(first > 3 || high > MAX_HIGH_ORDER || p_snapshot.get_sh_degree() != degree_for_layout(first, high), ERR_INVALID_DATA);
74 const uint64_t high_elements = uint64_t(gaussians.size()) * high;
75 ERR_FAIL_COND_V(uint64_t(high_order.size()) != high_elements, ERR_INVALID_DATA);
76 const uint64_t bytes = PREFIX_SIZE + uint64_t(gaussians.size()) * RECORD_SIZE + high_elements * 12;
77 ERR_FAIL_COND_V(bytes > UINT32_MAX, ERR_INVALID_DATA);
78 if (!gs_resize_or_fail(r_payload, int64_t(bytes), "GaussianSceneSerializer::_write_gaussian_data_chunk payload")) {
79 return ERR_OUT_OF_MEMORY;
80 }
81 uint8_t *w = r_payload.ptrw();
82 encode_uint32(gaussians.size(), w);
83 encode_uint32(1, w + 4);
84 encode_uint32(RECORD_SIZE, w + 8);
85 encode_uint32(p_snapshot.get_sh_degree(), w + 12);
86 encode_uint32(first, w + 16);
87 encode_uint32(high, w + 20);
88 encode_uint32(uint32_t(p_snapshot.get_2d_mode()) | (uint32_t(p_snapshot.get_antialiased()) << 1), w + 24);
89 encode_uint32(0, w + 28);
90 uint64_t offset = PREFIX_SIZE;
91 for (int i = 0; i < gaussians.size(); i++) {
92 encode_record(gaussians[i], w + offset);
93 offset += RECORD_SIZE;
94 }
95 for (int i = 0; i < high_order.size(); i++) {
96 const Vector3 &coefficient = high_order[i];
97 encode_float(coefficient.x, w + offset);
98 encode_float(coefficient.y, w + offset + 4);
99 encode_float(coefficient.z, w + offset + 8);
100 offset += 12;
101 }
102 return OK;
103}

106inline Error decode(const PackedByteArray &p_payload, uint32_t p_declared_count, uint64_t p_memory_budget,
107 LocalVector<Gaussian> &r_gaussians, LocalVector<Vector3> &r_high_order,
108 uint32_t &r_first, uint32_t &r_high, bool &r_mode_2d, bool &r_antialiased) {
109 ERR_FAIL_COND_V(p_payload.size() < PREFIX_SIZE, ERR_FILE_CORRUPT);
110 const uint8_t *r = p_payload.ptr();
111 const uint32_t count = decode_uint32(r);
112 const uint32_t degree = decode_uint32(r + 12);
113 const uint32_t first = decode_uint32(r + 16);
114 const uint32_t high = decode_uint32(r + 20);
115 const uint32_t flags = decode_uint32(r + 24);
116 ERR_FAIL_COND_V(count != p_declared_count || decode_uint32(r + 4) != 1 || decode_uint32(r + 8) != RECORD_SIZE, ERR_FILE_CORRUPT);
117 ERR_FAIL_COND_V(first > 3 || high > MAX_HIGH_ORDER || degree > 4 || degree != degree_for_layout(first, high), ERR_FILE_CORRUPT);
118 ERR_FAIL_COND_V((flags & ~3u) != 0 || decode_uint32(r + 28) != 0, ERR_FILE_CORRUPT);
119 const uint64_t high_elements = uint64_t(count) * high;
120 const uint64_t expected_size = PREFIX_SIZE + uint64_t(count) * RECORD_SIZE + high_elements * 12;
121 ERR_FAIL_COND_V(expected_size != uint64_t(p_payload.size()), ERR_FILE_CORRUPT);
122 // Canonical float32 records are smaller than the in-memory Gaussian ABI.
123 // Bound staging independently so the decode buffer plus staging remain <= 2B.
124 const uint64_t staging_bytes = uint64_t(count) * sizeof(Gaussian) + high_elements * sizeof(Vector3);
125 ERR_FAIL_COND_V(staging_bytes > p_memory_budget, ERR_FILE_CORRUPT);
126 r_gaussians.resize(count);
127 r_high_order.resize(high_elements);
128 uint64_t offset = PREFIX_SIZE;
129 for (uint32_t i = 0; i < count; i++) {
130 r_gaussians[i] = decode_record(r + offset);
131 offset += RECORD_SIZE;
132 }
133 for (uint32_t i = 0; i < r_high_order.size(); i++) {
134 r_high_order[i] = Vector3(decode_float(r + offset), decode_float(r + offset + 4), decode_float(r + offset + 8));
135 offset += 12;
136 }
137 r_first = first;
138 r_high = high;
139 r_mode_2d = flags & 1;
140 r_antialiased = flags & 2;
141 return OK;
142}
