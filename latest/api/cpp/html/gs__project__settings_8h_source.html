28// Canonical user-facing lod_bias range for the scene-facing nodes
29// (GaussianSplatNode3D / GaussianSplatWorld3D). Single source of truth for the
30// setter CLAMP and the editor PROPERTY_HINT_RANGE so they can never drift.
31// The low-level GaussianSplatRenderer intentionally accepts a wider internal
32// range ([0.01, 8.0]) as a permissive superset — node values are always
33// pre-clamped to this range before reaching it, so the renderer's range never
34// narrows a node value; it only allows programmatic/preset callers a wider span.
35static constexpr float GS_LOD_BIAS_MIN = 0.1f;
36static constexpr float GS_LOD_BIAS_MAX = 4.0f;
37// Keep this hint string numerically equal to [GS_LOD_BIAS_MIN, GS_LOD_BIAS_MAX]
38// (PROPERTY_HINT_RANGE needs a compile-time string literal, so it can't embed
39// the constexpr floats directly).
40#define GS_LOD_BIAS_HINT_RANGE "0.1,4.0,0.1"

57static inline bool get_bool(ProjectSettings *p_ps, const StringName &p_name, bool p_fallback) {
58 ERR_FAIL_NULL_V(p_ps, p_fallback);
59
60 if (!p_ps->has_setting(p_name)) {
61 return p_fallback;
62 }
63
64 Variant value = p_ps->get_setting_with_override(p_name);
65 if (value.get_type() == Variant::BOOL) {
66 return (bool)value;
67 }
68 if (value.get_type() == Variant::INT) {
69 return value.operator int64_t() != 0;
70 }
71 if (value.get_type() == Variant::FLOAT) {
72 return !Math::is_zero_approx((float)value.operator double());
73 }
74
75 return p_fallback;
76}

83static inline uint32_t get_uint(ProjectSettings *p_ps, const StringName &p_name, uint32_t p_fallback) {
84 if (!p_ps || !p_ps->has_setting(p_name)) {
85 return p_fallback;
86 }
87
88 Variant value = p_ps->get_setting_with_override(p_name);
89 if (value.get_type() == Variant::INT) {
90 int64_t v = value.operator int64_t();
91 return v < 0 ? p_fallback : uint32_t(v);
92 }
93 if (value.get_type() == Variant::FLOAT) {
94 double v = value.operator double();
95 return v < 0.0 ? p_fallback : uint32_t(Math::round(v));
96 }
97
98 return p_fallback;
99}

128static inline int get_int(ProjectSettings *p_ps, const StringName &p_name, int p_fallback) {
129 if (!p_ps || !p_ps->has_setting(p_name)) {
130 return p_fallback;
131 }
132 Variant value = p_ps->get_setting_with_override(p_name);
133 if (value.get_type() == Variant::INT) {
134 return static_cast<int>(value.operator int64_t());
135 }
136 if (value.get_type() == Variant::FLOAT) {
137 return static_cast<int>(Math::round(value.operator double()));
138 }
139 if (value.get_type() == Variant::BOOL) {
140 return value.operator bool() ? 1 : 0;
141 }
142 return p_fallback;
143}

370static inline GSSphereEffectorSettings get_sphere_effector_settings(ProjectSettings *p_ps, bool p_warn_on_clamp = false) {
371 GSSphereEffectorSettings settings;
372 if (!p_ps) {
373 return settings;
374 }
375
376 const int requested_max_effectors = get_int(p_ps, "rendering/gaussian_splatting/effects/max_effectors", settings.max_effectors);
377 settings.max_effectors = CLAMP(requested_max_effectors, 0, 1);
378 if (p_warn_on_clamp && requested_max_effectors > 1) {
379 WARN_PRINT_ONCE("[GaussianSplatting] Only one sphere effector is currently supported. "
380 "'rendering/gaussian_splatting/effects/max_effectors' was clamped to 1.");
381 }
382
383 settings.enabled = get_bool(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_enabled", settings.enabled);
384 settings.center.x = get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_center_x", settings.center.x);
385 settings.center.y = get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_center_y", settings.center.y);
386 settings.center.z = get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_center_z", settings.center.z);
387 settings.radius = MAX(get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_radius", settings.radius), 0.0f);
388 settings.strength = get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_strength", settings.strength);
389 settings.falloff = MAX(get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_falloff", settings.falloff), 0.001f);
390 settings.frequency = MAX(get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_frequency", settings.frequency), 0.1f);
391 settings.affect_position = get_bool(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_affect_position", settings.affect_position);
392 settings.affect_opacity = get_bool(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_affect_opacity", settings.affect_opacity);
393 settings.opacity_strength = CLAMP(
394 get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_opacity_strength", settings.opacity_strength),
395 0.0f, 1.0f);
396 settings.target_opacity = CLAMP(
397 get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_target_opacity", settings.target_opacity),
398 0.0f, 1.0f);
399 settings.enabled = settings.enabled && settings.max_effectors > 0;
400 return settings;
401}
