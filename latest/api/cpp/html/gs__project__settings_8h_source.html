28// Canonical user-facing lod_bias range for the scene-facing nodes
29// (GaussianSplatNode3D / GaussianSplatWorld3D). Single source of truth for the
30// setter CLAMP and the editor PROPERTY_HINT_RANGE so they can never drift.
31// The low-level GaussianSplatRenderer intentionally accepts a wider internal
32// range ([0.01, 8.0]) as a permissive superset — node values are always
33// pre-clamped to this range before reaching it, so the renderer's range never
34// narrows a node value; it only allows programmatic/preset callers a wider span.
35static constexpr float GS_LOD_BIAS_MIN = 0.1f;
36static constexpr float GS_LOD_BIAS_MAX = 4.0f;
37// Keep this hint string numerically equal to [GS_LOD_BIAS_MIN, GS_LOD_BIAS_MAX]
38// (PROPERTY_HINT_RANGE needs a compile-time string literal, so it can't embed
39// the constexpr floats directly).
40#define GS_LOD_BIAS_HINT_RANGE "0.1,4.0,0.1"

51static inline bool get_bool(ProjectSettings *p_ps, const StringName &p_name, bool p_fallback) {
52 ERR_FAIL_NULL_V(p_ps, p_fallback);
53
54 if (!p_ps->has_setting(p_name)) {
55 return p_fallback;
56 }
57
58 Variant value = p_ps->get_setting_with_override(p_name);
59 if (value.get_type() == Variant::BOOL) {
60 return (bool)value;
61 }
62 if (value.get_type() == Variant::INT) {
63 return value.operator int64_t() != 0;
64 }
65 if (value.get_type() == Variant::FLOAT) {
66 return !Math::is_zero_approx((float)value.operator double());
67 }
68
69 return p_fallback;
70}

77static inline uint32_t get_uint(ProjectSettings *p_ps, const StringName &p_name, uint32_t p_fallback) {

83 if (value.get_type() == Variant::INT) {
84 int64_t v = value.operator int64_t();
85 return v < 0 ? p_fallback : uint32_t(v);

87 if (value.get_type() == Variant::FLOAT) {

89 return v < 0.0 ? p_fallback : uint32_t(Math::round(v));

91
92 return p_fallback;

122static inline int get_int(ProjectSettings *p_ps, const StringName &p_name, int p_fallback) {
123 if (!p_ps || !p_ps->has_setting(p_name)) {
124 return p_fallback;

126 Variant value = p_ps->get_setting_with_override(p_name);
127 if (value.get_type() == Variant::INT) {

347static inline GSSphereEffectorSettings get_sphere_effector_settings(ProjectSettings *p_ps, bool p_warn_on_clamp = false) {
348 GSSphereEffectorSettings settings;
349 if (!p_ps) {
350 return settings;
351 }
352
353 const int requested_max_effectors = get_int(p_ps, "rendering/gaussian_splatting/effects/max_effectors", settings.max_effectors);
354 settings.max_effectors = CLAMP(requested_max_effectors, 0, 1);
355 if (p_warn_on_clamp && requested_max_effectors > 1) {
356 WARN_PRINT_ONCE("[GaussianSplatting] Only one sphere effector is currently supported. "
357 "'rendering/gaussian_splatting/effects/max_effectors' was clamped to 1.");
358 }
359
360 settings.enabled = get_bool(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_enabled", settings.enabled);
361 settings.center.x = get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_center_x", settings.center.x);
362 settings.center.y = get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_center_y", settings.center.y);
363 settings.center.z = get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_center_z", settings.center.z);
364 settings.radius = MAX(get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_radius", settings.radius), 0.0f);
365 settings.strength = get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_strength", settings.strength);
366 settings.falloff = MAX(get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_falloff", settings.falloff), 0.001f);
367 settings.frequency = MAX(get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_frequency", settings.frequency), 0.1f);
368 settings.affect_position = get_bool(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_affect_position", settings.affect_position);
369 settings.affect_opacity = get_bool(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_affect_opacity", settings.affect_opacity);
370 settings.opacity_strength = CLAMP(
371 get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_opacity_strength", settings.opacity_strength),
372 0.0f, 1.0f);
373 settings.target_opacity = CLAMP(
374 get_float(p_ps, "rendering/gaussian_splatting/effects/sphere_effector_target_opacity", settings.target_opacity),
375 0.0f, 1.0f);
376 settings.enabled = settings.enabled && settings.max_effectors > 0;
377 return settings;
378}
