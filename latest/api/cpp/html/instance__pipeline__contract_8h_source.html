373inline InvariantViolationReason first_raster_violation(const GaussianSplatRenderer::InstancePipelineBuffers &p_buffers) {
374 if (!p_buffers.atlas_gaussian_buffer.is_valid()) {
375 return InvariantViolationReason::RASTER_ATLAS_GAUSSIAN_BUFFER_MISSING;
376 }
377 if (!p_buffers.instance_buffer.is_valid()) {
378 return InvariantViolationReason::RASTER_INSTANCE_BUFFER_MISSING;
379 }
380 // Tile binning binds `instance_grading_buffer` at set=0 binding=20 and reads
381 // per-splat grading via `splat_ref.instance_id`. It is a hard requirement for
382 // raster readiness — without this check a caller can pass has_raster_buffers()
383 // yet still trip the runtime ERR_FAIL_COND in tile_render_binning.cpp at bind
384 // time, producing late ERR_UNCONFIGURED failures and fallback loops.
385 if (!p_buffers.instance_grading_buffer.is_valid()) {
386 return InvariantViolationReason::RASTER_INSTANCE_GRADING_BUFFER_MISSING;
387 }
388 if (!p_buffers.splat_ref_buffer.is_valid()) {
389 return InvariantViolationReason::RASTER_SPLAT_REF_BUFFER_MISSING;
390 }
391 if (!p_buffers.instance_count_buffer.is_valid()) {
392 return InvariantViolationReason::RASTER_INSTANCE_COUNT_BUFFER_MISSING;
393 }
394 if (p_buffers.quantization_required && !p_buffers.quantization_buffer.is_valid()) {
395 return InvariantViolationReason::RASTER_QUANTIZATION_BUFFER_MISSING;
396 }
397 return InvariantViolationReason::NONE;
398}

400inline InvariantViolation evaluate_streaming_activation(const GaussianSplatRenderer::InstancePipelineBuffers &p_buffers,
401 bool p_atlas_buffers_required) {
402 if (p_atlas_buffers_required) {
403 const InvariantViolationReason atlas_violation = first_atlas_violation(p_buffers);
404 if (atlas_violation != InvariantViolationReason::NONE) {
405 return make_violation(atlas_violation);
406 }
407 }
408
409 const InvariantViolationReason cull_violation = first_cull_violation(p_buffers);
410 if (cull_violation != InvariantViolationReason::NONE) {
411 return make_violation(cull_violation);
412 }
413
414 const InvariantViolationReason sort_violation = first_sort_violation(p_buffers);
415 if (sort_violation != InvariantViolationReason::NONE) {
416 return make_violation(sort_violation);
417 }
418
419 const InvariantViolationReason raster_violation = first_raster_violation(p_buffers);
420 if (raster_violation != InvariantViolationReason::NONE) {
421 return make_violation(raster_violation);
422 }
423
424 return InvariantViolation();
425}

427inline InvariantViolationReason first_tile_runtime_violation(const RID &p_instance_buffer,
428 const RID &p_instance_grading_buffer,
429 const RID &p_splat_ref_buffer,
430 const RID &p_indirect_count_buffer, const RID &p_indirect_dispatch_buffer,
431 bool p_quantization_required, const RID &p_quantization_buffer,
432 const RID &p_chunk_meta_buffer = RID()) {
433 if (!p_instance_buffer.is_valid()) {
434 return InvariantViolationReason::TILE_INSTANCE_BUFFER_MISSING;
435 }
436 // Tile binning binds `instance_grading_buffer` at set=0 binding=20 and reads
437 // it per splat, so a missing RID here is a hard contract violation. Fail
438 // fast during preflight instead of at uniform-set acquisition time, which
439 // otherwise surfaces as late fallback/error behavior.
440 if (!p_instance_grading_buffer.is_valid()) {
441 return InvariantViolationReason::TILE_INSTANCE_GRADING_BUFFER_MISSING;
442 }
443 if (!p_splat_ref_buffer.is_valid()) {
444 return InvariantViolationReason::TILE_SPLAT_REF_BUFFER_MISSING;
445 }
446 if (!p_indirect_count_buffer.is_valid()) {
447 return InvariantViolationReason::TILE_INDIRECT_COUNT_BUFFER_MISSING;
448 }
449 if (!p_indirect_dispatch_buffer.is_valid()) {
450 return InvariantViolationReason::TILE_INDIRECT_DISPATCH_BUFFER_MISSING;
451 }
452 if (p_quantization_required && !p_quantization_buffer.is_valid()) {
453 return InvariantViolationReason::TILE_QUANTIZATION_BUFFER_MISSING;
454 }
455 if (p_quantization_required && !p_chunk_meta_buffer.is_valid()) {
456 return InvariantViolationReason::TILE_CHUNK_META_BUFFER_MISSING;
457 }
458 return InvariantViolationReason::NONE;
