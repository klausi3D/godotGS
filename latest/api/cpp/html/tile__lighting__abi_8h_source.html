37 // Conservative shader-layout minimums for the zero fallback buffers bound by
38 // create_lighting_uniform_set(). They intentionally stay at or above the
39 // historical fallback sizes and the Godot 4.x shader structs used by
40 // gs_lighting_bridge.glsl:
41 // - SceneDataBlock is two std140 SceneData structs, about 5.7 KiB with MAX_VIEWS=2.
42 // - DirectionalLightData is a std140 array entry, about 464 bytes.
43 // - LightData is 192 bytes in std430; 32 entries cover offline shader variants.
44 // - ReflectionData is about 128 bytes in std430.
45 // - ClusterBuffer needs at least one 32-slice record for omni and spot masks.

56 static constexpr bool is_power_of_two(uint32_t p_value) {

60 static constexpr uint32_t shift_from_power_of_two(uint32_t p_value) {
61 uint32_t shift = 0;
62 while ((uint32_t(1) << shift) < p_value && shift < 31) {

74 static TileLightingClusterABIConfig compute_cluster_config(const Vector2i &p_viewport_size,
75 uint32_t p_cluster_size, uint32_t p_cluster_max_elements, bool p_cluster_buffer_valid) {
76 TileLightingClusterABIConfig config;
77 if (!p_cluster_buffer_valid) {
78 config.disabled_reason = "missing cluster_buffer RID";
79 return config;
80 }
81 if (p_viewport_size.x <= 0 || p_viewport_size.y <= 0) {
82 config.disabled_reason = "non-positive viewport size";
83 return config;
84 }
85 if (!is_power_of_two(p_cluster_size)) {
86 config.disabled_reason = "cluster_size must be a non-zero power of two";
87 return config;
88 }
89 if (p_cluster_max_elements < 32 || (p_cluster_max_elements % 32) != 0) {
90 config.disabled_reason = "cluster_max_elements must be a multiple of 32";
91 return config;
92 }
93
94 const uint32_t width = (uint32_t(p_viewport_size.x) + p_cluster_size - 1) / p_cluster_size;
95 const uint32_t height = (uint32_t(p_viewport_size.y) + p_cluster_size - 1) / p_cluster_size;
96 const uint32_t max_div_32 = p_cluster_max_elements / 32;
97 const uint64_t type_size = uint64_t(width) * uint64_t(height) * uint64_t(max_div_32 + CLUSTER_Z_SLICES);
98 if (type_size > UINT32_MAX) {
99 config.disabled_reason = "cluster buffer layout exceeds 32-bit shader offsets";
100 return config;
101 }
102
103 config.enabled = true;
104 config.cluster_shift = shift_from_power_of_two(p_cluster_size);
105 config.cluster_width = width;
106 config.cluster_type_size = uint32_t(type_size);
107 config.max_cluster_element_count_div_32 = max_div_32;
108 config.required_storage_bytes = cluster_storage_bytes(width, height, max_div_32);
109 return config;
110 }
