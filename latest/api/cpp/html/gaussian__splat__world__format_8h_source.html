28inline Error select_header(uint32_t p_version, uint32_t p_flags, uint64_t p_file_bytes, HeaderLayout &r_layout) {
29 HeaderLayout layout;
30 if (p_version != kVersionV1 && p_version != kVersionV2 && p_version != kVersion) {
31 return ERR_FILE_CORRUPT;
32 }
33 layout.has_hlod = (p_flags & kFlagHasHlod) != 0u;
34 if ((p_flags & ~(kPlainFlags | kFlagHasHlod)) != 0u) {
35 return layout.has_hlod ? ERR_FILE_CORRUPT : ERR_FILE_UNRECOGNIZED;
36 }
37 if (p_version == kVersionV1 && layout.has_hlod) {
38 return ERR_FILE_CORRUPT;
39 }
40 if (p_version == kVersionV2 && !layout.has_hlod) {
41 // The 112-byte plain-v2 prototype collided with the released HLOD-v2
42 // identity. It requires re-export; never infer it from its payload.
43 return ERR_FILE_UNRECOGNIZED;
44 }
45 layout.has_layout_word = p_version == kVersion;
46 layout.prefix_bytes = layout.has_layout_word ? kPrefixBytes : kLegacyPrefixBytes;
47 layout.header_bytes = layout.prefix_bytes + (layout.has_hlod ? kHlodExtensionBytes : 0u);
48 if (p_file_bytes < layout.header_bytes) {
49 return ERR_FILE_CORRUPT;
50 }
51 r_layout = layout;
52 return OK;
53}
