25// Bit 7 (BV-C5a, ADR D-12 "Containers"): the payload is a run of GSQ v1 pages per chunk
26// instead of raw Gaussian + SH sections. Readers built before it refuse the file, because
27// select_header() refuses every flag it does not know (ERR_FILE_UNRECOGNIZED).
28static constexpr uint32_t kFlagPagedChunks = 1u << 7u;
29static constexpr uint32_t kPlainFlags = (1u << 6u) - 1u;
30static constexpr uint64_t kLegacyPrefixBytes = 104u;
31static constexpr uint64_t kPrefixBytes = 112u;
32static constexpr uint64_t kHlodExtensionBytes = 80u;
33// One plain chunk-table record: bounds position, bounds size, centre (3 x vec3 f32),
34// radius f32, indices_offset u64 (in indices, not bytes), index_count u32, reserved u32.
35static constexpr uint64_t kChunkRecordBytes = 56u;

37// Plain v3 + bit 7 (paged GSQ chunks). File layout:
38//
39// prefix (112 B, the plain v3 header; sh_offset and indices_offset are 0, gaussian_offset is
40// the first page, chunk_table_offset the paged chunk table)
41// paged extension (64 B, below)
42// zero padding up to kPagedPageAlignment
43// pages: GSQ v1 pages (io/gsq/gsq_codec.h) in chunk order, each starting on a
44// kPagedPageAlignment boundary, zero padding between them
45// paged chunk table: chunk_count x kPagedChunkRecordBytes
46// page table: page_count x kPagedPageRecordBytes
47// metadata (optional, as plain v3)
48//
49// Paged extension, little endian:
50// 0 u32 codec id (1 = GSQ v1)
51// 4 u8[3] SH bits per band, u8 opacity bits } the W16 codec block, as the writer
52// 8 f32 pos_step_m } was given it
53// 12 i32 zstd level }
54// 16 u32 page count
55// 20 u32 page alignment (kPagedPageAlignment)
56// 24 u64 page table offset
57// 32 u8[32] reserved, zero
58//
59// Paged chunk record: bounds position, bounds size, centre (3 x vec3 f32), radius f32,
60// first page u32, page count u32, splat count u32, reserved u32 (zero). A chunk's splats are the
61// concatenation of its pages; the index lists of the uncompressed container do not exist.
62//
63// Page record: offset u64, stored bytes u32, splat count u32, stored SHA-256 [32],
64// decoded-canonical SHA-256 [32], position extent min xyz / max xyz (6 x f32, equal to the
65// page header's), chunk index u32, reserved u32 (zero).
66static constexpr uint64_t kPagedExtensionBytes = 64u;
67static constexpr uint64_t kPagedPageAlignment = 4096u;
68static constexpr uint64_t kPagedChunkRecordBytes = 56u;
69static constexpr uint64_t kPagedPageRecordBytes = 112u;

79inline Error select_header(uint32_t p_version, uint32_t p_flags, uint64_t p_file_bytes, HeaderLayout &r_layout) {
80 HeaderLayout layout;
81 if (p_version != kVersionV1 && p_version != kVersionV2 && p_version != kVersion) {
82 return ERR_FILE_CORRUPT;
83 }
84 layout.has_hlod = (p_flags & kFlagHasHlod) != 0u;
85 layout.has_paged_chunks = (p_flags & kFlagPagedChunks) != 0u;
86 if ((p_flags & ~(kPlainFlags | kFlagHasHlod | kFlagPagedChunks)) != 0u) {
87 return layout.has_hlod ? ERR_FILE_CORRUPT : ERR_FILE_UNRECOGNIZED;
88 }
89 if (layout.has_paged_chunks) {
90 // Bit 7 exists only in plain v3, never with HLOD, the whole-section gzip (bit 4) or the
91 // resident-payload marker (bit 5), and always with its chunk table (bit 2).
92 if (p_version != kVersion || layout.has_hlod || (p_flags & (kFlagCompressed | kFlagResidentPayload)) != 0u ||
93 (p_flags & kFlagHasChunks) == 0u) {
94 return ERR_FILE_CORRUPT;
95 }
96 }
97 if (p_version == kVersionV1 && layout.has_hlod) {
98 return ERR_FILE_CORRUPT;
99 }
100 if (p_version == kVersionV2 && !layout.has_hlod) {
101 // The 112-byte plain-v2 prototype collided with the released HLOD-v2
102 // identity. It requires re-export; never infer it from its payload.
103 return ERR_FILE_UNRECOGNIZED;
104 }
105 layout.has_layout_word = p_version == kVersion;
106 layout.prefix_bytes = layout.has_layout_word ? kPrefixBytes : kLegacyPrefixBytes;
107 layout.header_bytes = layout.prefix_bytes + (layout.has_hlod ? kHlodExtensionBytes : 0u) +
108 (layout.has_paged_chunks ? kPagedExtensionBytes : 0u);
109 if (p_file_bytes < layout.header_bytes) {
110 return ERR_FILE_CORRUPT;
111 }
112 r_layout = layout;
113 return OK;
114}
