11// #611: a `Mutex` that remembers which thread currently owns it.
12//
13// GaussianSplatSceneDirector::world_mutex sits on both sides of a lock-order
14// inversion: the main thread takes it and then issues a *blocking* render-thread
15// dispatch (renderer teardown, `initialize()`, `set_max_splats`,
16// `set_gaussian_data`, `set_file_backed_payload_source`), while the render thread
17// itself needs the same mutex inside the director's `*_for_renderer` builders. A
18// blocking dispatch under the lock therefore stalls until the dispatch timeout
19// expires and then silently drops (or rolls back) the operation.
20//
21// The rule "never cross the renderer-contract boundary while holding
22// world_mutex" used to live only in a header comment, with every caller required
23// to get the ordering right independently. This type makes the rule *queryable*
24// at runtime, so the boundary itself can check it instead of being trusted.
25//
26// Every thread is identifiable, including ones Godot's `Thread` never started:
27// `caller_id` is a `thread_local` initialised from `id_counter.increment()`
28// (`core/os/thread.cpp:41`), so the first `get_caller_id()` on any thread mints a
29// unique non-zero id. `Thread::UNASSIGNED_ID` is therefore never *returned*; the
30// check against it below exists only so an unowned mutex — whose `owner_id` is
31// reset to that sentinel — can never spuriously match a caller.
32class ThreadOwnedMutex {
33 mutable Mutex mutex; // Recursive, matching the `Mutex` this replaces.
34 mutable SafeNumeric<uint64_t> owner_id{ Thread::UNASSIGNED_ID };
35 // Only ever touched by the owning thread while `mutex` is held, so a plain
36 // integer is sufficient; nothing outside the critical section reads it.
37 mutable uint32_t recursion_depth = 0;
38
39public:
40 _ALWAYS_INLINE_ void lock() const {
41 mutex.lock();
42 recursion_depth++;
43 owner_id.set(Thread::get_caller_id());
44 }
45
46 _ALWAYS_INLINE_ void unlock() const {
47 if (recursion_depth > 0) {
48 recursion_depth--;
49 }
50 if (recursion_depth == 0) {
51 // Published before the mutex is released so no incoming owner can
52 // observe a stale id.
53 owner_id.set(Thread::UNASSIGNED_ID);
54 }
55 mutex.unlock();
56 }
57
58 _ALWAYS_INLINE_ bool is_held_by_current_thread() const {
59 const Thread::ID caller = Thread::get_caller_id();
60 if (caller == Thread::UNASSIGNED_ID) {
61 return false;
62 }
63 return owner_id.get() == caller;
64 }
65
66 // Recursion depth as seen by the owning thread; 0 for anyone else.
67 _ALWAYS_INLINE_ uint32_t get_owned_recursion_depth() const {
68 return is_held_by_current_thread() ? recursion_depth : 0u;
69 }
70
71 ThreadOwnedMutex() = default;
72 ThreadOwnedMutex(const ThreadOwnedMutex &) = delete;
73 ThreadOwnedMutex &operator=(const ThreadOwnedMutex &) = delete;
74};

76// RAII lock for ThreadOwnedMutex.
77//
78// Godot's `MutexLock` binds a `std::unique_lock` straight onto `MutexImpl`'s
79// private `std::mutex`, bypassing `lock()`/`unlock()` entirely, so it cannot
80// maintain the ownership record. This type must be used instead.
81class ThreadOwnedMutexLock {
82 const ThreadOwnedMutex &mutex;
83
84public:

87 mutex.lock();
88 }
89 ~ThreadOwnedMutexLock() {

91 }
92

97// The renderer-contract boundary check.
98//
99// Returns true when `p_mutex` is held by the calling thread — i.e. when the
100// caller is about to issue a blocking render-thread dispatch from inside the
101// critical section the render thread itself needs. The violation is counted in
102// `r_violation_count` so a caller (or a test) can observe it without parsing
103// logs.
104//
105// A count is evidence of the *ordering* violation, never of a stall: whether the
106// dispatch actually blocks depends on the run (under `--headless` the dispatcher
107// short-circuits and returns immediately).
108//
109// This *reports*; it does not abort. Aborting would change behaviour on the one
110// call path that is still knowingly in violation (`submit_world_submission`'s
111// apply, deferred to the follow-up PR), turning every world submission on a live
112// render thread into a rejection. Reporting is what makes that remaining
113// violation self-describing instead of merely argued — there is no lane in this
114// repo that can reproduce the stall behaviourally (every doctest run is
115// `--headless --test`, under which the dispatcher short-circuits and never
116// blocks).
117_ALWAYS_INLINE_ bool report_lock_held_at_boundary(const ThreadOwnedMutex &p_mutex,
118 SafeNumeric<uint64_t> &r_violation_count) {
119 if (!p_mutex.is_held_by_current_thread()) {
120 return false;
121 }
122 r_violation_count.increment();
123 return true;
124}
