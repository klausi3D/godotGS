10// BV-04a (ADR adr-beyond-vram-worlds.md D-5): a fixed log2 latency histogram whose counters
11// are lock-free atomics in EVERY build flavour, so release-build streaming latencies are
12// measurable. One record() is one atomic add per field (bucket, samples, total) plus one
13// compare-exchange for the max; there is no per-splat work and no allocation.
14//
15// Bucket b holds samples in [2^(b-1), 2^b) microseconds for 1 <= b < BUCKET_COUNT - 1,
16// bucket 0 holds samples of 0 us, and the last bucket is open-ended (>= 2^(BUCKET_COUNT-2) us,
17// about 4.2 s). Counts are cumulative over the owner's lifetime and never reset per frame, so
18// a reader that samples only some frames loses nothing: it diffs two snapshots.
19struct StreamingLatencyHistogram {
20 static constexpr uint32_t BUCKET_COUNT = 24;
21
22 SafeNumeric<uint64_t> buckets[BUCKET_COUNT];
23 SafeNumeric<uint64_t> samples;
24 SafeNumeric<uint64_t> total_usec;
25 SafeNumeric<uint64_t> max_usec;
26
27 static constexpr uint32_t bucket_for(uint64_t p_usec) {
28 uint32_t width = 0;
29 while (p_usec != 0 && width < BUCKET_COUNT - 1) {
30 p_usec >>= 1;
31 width++;
32 }
33 return width;
34 }
35
36 // Exclusive upper bound of bucket p_bucket in microseconds; -1 for the open-ended last one.
37 static constexpr int64_t bucket_upper_usec(uint32_t p_bucket) {
38 return p_bucket >= BUCKET_COUNT - 1 ? int64_t(-1) : (int64_t(1) << p_bucket);
39 }
40
41 _ALWAYS_INLINE_ void record(uint64_t p_usec) {
42 buckets[bucket_for(p_usec)].increment();
43 samples.increment();
44 total_usec.add(p_usec);
45 max_usec.exchange_if_greater(p_usec);
46 }
