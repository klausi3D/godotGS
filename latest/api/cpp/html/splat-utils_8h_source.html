61// Decode a quaternion stored as its first three components in 8-bit fixed point;
62// w is reconstructed from the constraint that the quaternion is unit-length and
63// non-negative. Within-family: `c.flipQ` scales xyz (fused into the decode).
64// Cross-family: coordinateConverter sets `c.flipQ` to identity (so the fused
65// multiply is a no-op) and `c.rotFlipQFunc` rotates the full xyzw afterwards.
66inline void unpackQuaternionFirstThree(
67 float rotation[4], const uint8_t r[3],
68 const CoordinateConverter &c = CoordinateConverter()) {
69 Vec3f xyz = times(
70 plus(
71 times(
72 Vec3f{static_cast<float>(r[0]), static_cast<float>(r[1]), static_cast<float>(r[2])},
73 1.0f / 127.5f),
74 Vec3f{-1, -1, -1}),
75 c.flipQ);
76 std::copy(xyz.data(), xyz.data() + 3, &rotation[0]);
77 rotation[3] = std::sqrt(std::max(0.0f, 1.0f - squaredNorm(xyz)));
78 if (c.rotFlipQFunc) {
79 c.rotFlipQFunc(rotation);
80 }
81}

83// Decode a quaternion stored in the "smallest three" encoding: 2 bits identify
84// the largest component (reconstructed from the unit-length constraint) and the
85// other three are 10-bit signed magnitudes scaled by 1/sqrt(2).
86inline void unpackQuaternionSmallestThree(
87 float rotation[4], const uint8_t r[4],
88 const CoordinateConverter &c = CoordinateConverter()) {
89 uint32_t comp =
90 r[0] + (r[1] << 8) + (r[2] << 16) + (r[3] << 24);
91
92 constexpr uint32_t c_mask = (1u << 9u) - 1u;
93
94 const int i_largest = comp >> 30;
95 float sum_squares = 0;
96 for (int i = 3; i >= 0; --i) {
97 if (i != i_largest) {
98 uint32_t mag = comp & c_mask;
99 uint32_t negbit = (comp >> 9u) & 0x1u;
100 comp = comp >> 10u;
101 rotation[i] = sqrt1_2 * static_cast<float>(mag) / static_cast<float>(c_mask);
102 if (negbit == 1) {
103 rotation[i] = -rotation[i];
104 }
105 sum_squares += rotation[i] * rotation[i];
106 }
107 }
108 rotation[i_largest] = std::sqrt(1.0f - sum_squares);
109
110 // Within-family flip via c.flipQ (identity in the cross-family case, where
111 // c.rotFlipQFunc carries the full flip+rotation on the xyzw quaternion).
112 for (int i = 0; i < 3; i++) {
113 rotation[i] *= c.flipQ[i];
114 }
115 if (c.rotFlipQFunc) {
116 c.rotFlipQFunc(rotation);
117 }
118}
