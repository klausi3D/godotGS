15// PR1b of "world + movable instances in one frame" (docs/architecture/adr-one-route-many-producers.md, sections 4.1 to
16// 4.5): the HLOD route draws the world (row 0) and the visible GaussianSplatNode3D instances (rows 1..N) in one draw list,
17// one depth sort and one byte ledger. This header is the device-free part, tested on the CPU:
18//
19// - the instance region of a bundle: every distinct node asset is packed once, after the world, as whole chunks
20// (atlas records [world_records, +records), chunk tables [node_count, +chunk_count)); instances of one asset share it;
21// - the ledger and the capacities of a bundle with that region (for an empty region they are PR1a's exactly);
22// - the budget rule (ADR 4.5): visible objects reserve their need first, in a deterministic screen-importance order with
23// ties broken by the stable instance key; the world keeps a fixed floor of each performance cap; an object that does
24// not fit the object share is refused whole, with a reason, never drawn partially;
25// - the per-view instance draw records: one DrawnNodeGPU per (admitted row, chunk), with the offset T - camera formed in
26// double (the payload is asset-local, so the node centre c of R s c + T - camera is 0).
27namespace gs_hlod_bundle {

29// The world floor F (ADR 4.5, decision 3): the fraction of each performance cap of a view (the splat cap and the
30// overlap-record cap) the world always keeps, whatever the objects ask for. Objects may reserve at most the rest.
31// One half: the objects of a game frame can never take more than the world keeps, and the world's cut under
32// half its caps is at most a little coarser (the selector refines in priority order, so a halved budget stops the
33// same refinement sequence earlier; it never drops a region). An internal constant on purpose: a project setting
34// would be a settings change (R3, follow-up F3 of the ADR).
35constexpr uint32_t WORLD_FLOOR_NUMERATOR = 1;
36constexpr uint32_t WORLD_FLOOR_DENOMINATOR = 2;

41// One packed node asset. Chunk and record positions are region-local (0-based); the bundle places the region after the
42// world: chunk table index node_count + first_chunk + c, atlas record world_records + first_record + i.
43struct InstanceAsset {
44 uint64_t asset_id = 0;
45 uint32_t first_chunk = 0;
46 uint32_t chunk_count = 0;
47 uint64_t first_record = 0;
48 uint64_t splat_count = 0;
49 uint32_t max_chunk_splats = 0;
50 // Overlap footprint of the asset at scale 1 (the HLOD bake's per-node rule: sum of 36 * s1 * s2 over the splats,
51 // s1, s2 the two largest axes), so objects and world nodes are priced by one record-cost model.

53 Vector3 bounds_center; // Asset-local.

68 int find(uint64_t p_asset_id) const {
69 uint32_t lo = 0, hi = assets.size();
70 while (lo < hi) {
71 const uint32_t mid = lo + (hi - lo) / 2;
72 if (assets[mid].asset_id < p_asset_id) {
73 lo = mid + 1;
74 } else {
75 hi = mid;
76 }
77 }
78 return (lo < assets.size() && assets[lo].asset_id == p_asset_id) ? int(lo) : -1;
79 }

91// Fills the row tallies of `r_region` from `p_rows`. False when a row names an asset the region does not pack.
92inline bool tally_rows(const Rows &p_rows, InstanceRegion &r_region) {
93 r_region.row_chunks = 0;
94 r_region.row_splats = 0;
95 r_region.world_rows = 0;
96 for (uint32_t i = 0; i < p_rows.asset_ids.size(); ++i) {
97 if (p_rows.asset_ids[i] == WORLD_ASSET_ID) {
98 ++r_region.world_rows;
99 continue;
100 }
101 const int index = r_region.find(p_rows.asset_ids[i]);
102 if (index < 0) {
103 return false;
104 }
105 r_region.row_chunks += r_region.assets[index].chunk_count;
106 r_region.row_splats += r_region.assets[index].splat_count;
107 }
108 return true;
109}

111// Visible/draw-list capacity. Canonical (GPU-culled): a slot per (row, chunk) pair, world rows over every node. Relative
112// (CPU-written): the world's node-indexed entries, then one entry per (instance row, chunk).
113inline uint64_t visible_chunk_capacity(uint64_t p_node_count, bool p_relative, const InstanceRegion &p_region) {
114 return p_relative ? p_node_count + p_region.row_chunks : p_node_count * uint64_t(p_region.world_rows) + p_region.row_chunks;
115}

117// nullptr when the rows can be packed with this region into a bundle of `p_node_count` world nodes; otherwise the reason.
118// Row 0 must be the world (the director builds it first; every world draw record names row 0).
119inline const char *rows_refusal(const Rows &p_rows, uint64_t p_node_count, const InstanceRegion &p_region) {
120 const uint32_t n = p_rows.count();
121 if (n == 0) { return "hlod_bundle_rows_empty"; }
122 if (p_rows.grading.size() != n || p_rows.asset_ids.size() != n) { return "hlod_bundle_rows_mismatched"; }
123 if (p_rows.asset_ids[WORLD_ROW] != WORLD_ASSET_ID) { return "hlod_world_row_unavailable"; }
124 for (uint32_t i = 0; i < n; ++i) {
125 if (dense_id_for_asset(p_rows.asset_ids[i], p_region) == UINT32_MAX) { return "hlod_bundle_row_asset_not_packed"; }
126 }
127 InstanceRegion tallied = p_region;
128 if (!tally_rows(p_rows, tallied) || tallied.row_chunks != p_region.row_chunks || tallied.row_splats != p_region.row_splats ||
129 tallied.world_rows != p_region.world_rows) {
130 return "hlod_bundle_rows_region_mismatch";
131 }
132 // Every buffer of the bundle is created with a 32-bit size (hlod_create_buffer).
133 if (uint64_t(n) * sizeof(InstanceDataGPU) > UINT32_MAX ||
134 visible_chunk_capacity(p_node_count, false, p_region) * sizeof(VisibleChunkRefGPU) > UINT32_MAX ||
135 visible_chunk_capacity(p_node_count, true, p_region) * sizeof(DrawnNodeGPU) > UINT32_MAX ||
136 (p_node_count + p_region.chunk_count) * sizeof(ChunkQuantizationGPU) > UINT32_MAX) {
137 return "hlod_bundle_rows_exceed_capacity";
138 }
139 return nullptr;
140}

151// Bytes of one HLOD bundle with an instance region: the PR1a ledger plus the region's atlas records and chunk tables,
152// one AssetMetaGPU per packed node asset and the instance draw/visible slots. For an empty region this is
153// bundle_bytes(..., p_rows) of hlod_bundle_rows.h exactly (world_rows == p_rows).
154inline uint64_t bundle_bytes(uint64_t p_node_count, uint64_t p_records, bool p_relative, bool p_quantized, uint32_t p_capacity,
155 uint32_t p_rows, const InstanceRegion &p_region) {
156 return (p_records + p_region.records) * (p_quantized ? sizeof(PackedGaussianQuantized) : sizeof(PackedGaussian)) +
157 (p_node_count + p_region.chunk_count) * (sizeof(ChunkMetaGPU) + sizeof(AssetChunkIndexGPU) + (p_quantized ? sizeof(ChunkQuantizationGPU) : 0)) +
158 visible_chunk_capacity(p_node_count, p_relative, p_region) * (p_relative ? sizeof(DrawnNodeGPU) : sizeof(VisibleChunkRefGPU)) +
159 (1u + uint64_t(p_region.assets.size())) * sizeof(AssetMetaGPU) + uint64_t(p_rows) * (sizeof(InstanceDataGPU) + sizeof(InstanceGradingGPU)) +
160 uint64_t(p_capacity) * (sizeof(SplatRefGPU) + (p_relative ? sizeof(uint32_t) : 0)) +
161 6 * sizeof(uint32_t) + 2 * sizeof(GaussianSplatting::IndirectDispatchLayout);
162}

164// The world's share of a splat cap: ceil(cap * F). The objects' share is the rest.
165inline uint64_t world_floor_splats(uint64_t p_cap) {
166 return (p_cap * WORLD_FLOOR_NUMERATOR + WORLD_FLOOR_DENOMINATOR - 1u) / WORLD_FLOOR_DENOMINATOR;
167}
168inline uint64_t object_share_splats(uint64_t p_cap) {
169 return p_cap - world_floor_splats(p_cap);
170}
171inline double world_floor_records(double p_cap) {
172 return p_cap * double(WORLD_FLOOR_NUMERATOR) / double(WORLD_FLOOR_DENOMINATOR);
173}
174inline double object_share_records(double p_cap) {
175 return MAX(0.0, p_cap - world_floor_records(p_cap));
176}
177// The objects' share of the overlap records: at most C_view - F (C_view = the configured record cap, or the storage when
178// none is configured), and never more than the allocated storage holds beyond the world's need (capped at its floor),
179// so the world keeps min(world need, storage / 2) and is not degraded at all when the storage holds both.
180inline double object_share_records(double p_view_cap, double p_storage, double p_world_need) {
181 const double view = MAX(p_view_cap, p_storage);
182 return MAX(0.0, MIN(object_share_records(view), p_storage - MIN(p_world_need, world_floor_records(p_storage))));
183}

185// The overlap storage is shared: the route grows it to the world's record need plus the objects' estimate
186// (TileRenderer::ensure_hlod_overlap_capacity). The world's need is tracked apart from the storage, because a storage
187// already grown for the objects is not the world's need: read as such, the same object demand is added on top of it
188// again every frame, and a stationary view regrows the storage (reallocating the overlap buffers and rebuilding the
189// sorter) until the record cap or the ledger headroom stops it.
190struct OverlapWorldNeed {
191 uint64_t world_records = 0; // The world's need the route last sized the storage for.
192 uint64_t storage_records = 0; // The storage capacity right after that sizing (0: never sized).
193};
194// The world's own record need this frame: the tracked need while the storage is still the one the route last sized.
195// Any other storage (the first frame, a measured-overflow growth, a new device) is read whole as the world's: at most one
196// object estimate too much, once, after which the tracked need holds. Never below `p_seed` (the world's seed rule).
197inline uint64_t world_overlap_need(const OverlapWorldNeed &p_tracked, uint64_t p_storage, uint64_t p_seed) {
198 const bool same_storage = p_tracked.storage_records != 0 && p_storage == p_tracked.storage_records;
199 return MAX(same_storage ? p_tracked.world_records : p_storage, p_seed);
200}
201// The absolute storage requirement of a view: the world's need plus the objects' estimate, counted once (32-bit clamp).
202// Unchanged content gives an unchanged requirement, so a storage that already holds it is never grown again.
203inline uint32_t world_plus_objects_overlap_records(uint64_t p_world_need, double p_object_records) {
204 const double objects = std::isfinite(p_object_records) && p_object_records > 0.0 ? std::ceil(MIN(p_object_records, double(UINT32_MAX))) : 0.0;
205 return uint32_t(MIN<uint64_t>(MIN<uint64_t>(p_world_need, UINT32_MAX) + uint64_t(objects), UINT32_MAX));
206}
207// What the route grows the storage to once the requirement above exceeds it: the world's need plus 3/2 of the objects'
208// estimate (the slack grow_hlod_overlap_storage gives a measured overflow), so objects that move and whose estimate
209// rises a little each frame do not reallocate the storage on every rise. The route falls back to the exact requirement
210// when the slack does not fit the headroom. Whether to grow is still decided by the exact requirement.
211inline uint32_t world_plus_objects_overlap_growth(uint64_t p_world_need, double p_object_records) {
212 return world_plus_objects_overlap_records(p_world_need, std::isfinite(p_object_records) ? p_object_records * 1.5 : 0.0);
213}

215// Two caps (PR1b): S = max(1, max_splats) bounds what the WORLD can draw (the world contract clamps it to the world's
216// data count), C = the view's performance cap the objects reserve from (C >= S; renderer get_hlod_view_splat_cap).
217// The world's need W = min(leaf_count, S) is PR1a's world cap.
218inline uint64_t world_need_splats(uint32_t p_leaf_count, int p_max_splats) {
219 return MIN(uint64_t(p_leaf_count), uint64_t(MAX(1, p_max_splats)));
220}
221
222// Splat-ref capacity of the relative bundle: the world's need W plus what the rows can be admitted to (at most the
223// object share of C), never above max(C, W). For an empty region it is PR1a's min(leaf_count, S).
224inline uint32_t relative_capacity(uint32_t p_leaf_count, int p_max_splats, uint64_t p_view_cap, const InstanceRegion &p_region) {
225 const uint64_t world = world_need_splats(p_leaf_count, p_max_splats);
226 const uint64_t view = MAX(MAX(p_view_cap, uint64_t(1)), world);
227 const uint64_t objects = MIN(p_region.row_splats, object_share_splats(view));
228 return uint32_t(MIN(MIN(world + objects, view), uint64_t(UINT32_MAX)));
229}
230// The canonical (GPU-culled whole-tree) bundle holds every leaf and every row's whole asset. UINT32_MAX when it would
231// not fit a 32-bit count.
232inline uint32_t canonical_capacity(uint32_t p_leaf_count, const InstanceRegion &p_region) {
233 const uint64_t total = uint64_t(p_leaf_count) + p_region.row_splats;
234 return total > UINT32_MAX ? UINT32_MAX : uint32_t(total);
235}

262// Admission order: screen importance (descending), then the stable instance key, then the row. A NaN importance sorts
263// as 0, so the order is total and the same view always admits the same objects.
264inline bool admits_before(const ObjectDemand &p_a, const ObjectDemand &p_b) {
265 const double ia = std::isfinite(p_a.importance) ? p_a.importance : 0.0;
266 const double ib = std::isfinite(p_b.importance) ? p_b.importance : 0.0;
267 if (ia != ib) { return ia > ib; }
268 if (p_a.instance_key != p_b.instance_key) { return p_a.instance_key < p_b.instance_key; }
269 return p_a.row < p_b.row;
270}

272// ADR 4.5: objects are admitted in admits_before() order while the cumulative reserve fits the object share of every
273// cap. An object that does not fit is refused whole and the next one is still tried: objects are never dropped as a
274// group because one of them does not fit.
275inline void admit_objects(const ObjectDemand *p_demands, uint32_t p_count, const ObjectShare &p_share, Admission &r_admission) {
276 r_admission = Admission();
277 LocalVector<uint32_t> order;
278 order.resize(p_count);
279 for (uint32_t i = 0; i < p_count; ++i) { order[i] = i; }
280 std::sort(order.ptr(), order.ptr() + p_count, [&](uint32_t a, uint32_t b) { return admits_before(p_demands[a], p_demands[b]); });
281 for (uint32_t k = 0; k < p_count; ++k) {
282 const ObjectDemand &d = p_demands[order[k]];
283 const bool records_ok = std::isfinite(d.records) && d.records >= 0.0 && r_admission.records + d.records <= p_share.records;
284 if (d.splats <= p_share.splats - r_admission.splats && d.chunks <= p_share.nodes - r_admission.nodes && records_ok) {
285 r_admission.splats += d.splats;
286 r_admission.nodes += d.chunks;
287 r_admission.records += d.records;
288 r_admission.admitted.push_back(order[k]);
289 } else {
290 r_admission.refused.push_back(order[k]);
291 }
292 }
293 std::sort(r_admission.admitted.ptr(), r_admission.admitted.ptr() + r_admission.admitted.size(),
294 [&](uint32_t a, uint32_t b) { return p_demands[a].row < p_demands[b].row; });
295}

297// The world cut's caps after the objects reserved theirs (ADR 4.5): splats min(W, C - R), where W = min(leaf_count, S) is
298// the world's need and C the view cap (a streamed atlas caps it further); records: the overlap storage minus the objects'
299// estimate (the storage is grown to hold both, see TileRenderer::ensure_hlod_overlap_capacity). The admission never
300// reserves more than the object share C - F, so the world keeps min(W, F) at least; when W + R <= C nothing is taken
301// from the world at all. Without objects these are the PR1a caps exactly.
302inline void world_view_caps(uint32_t p_leaf_count, int p_max_splats, uint64_t p_view_cap, bool p_streaming, uint64_t p_stream_records,
303 double p_records, const Admission &p_admission, uint64_t &r_splat_budget, double &r_record_budget) {
304 const uint64_t need = world_need_splats(p_leaf_count, p_max_splats);
305 const uint64_t view = MAX(MAX(p_view_cap, uint64_t(1)), need);
306 const uint64_t world = MIN(need, view - MIN(p_admission.splats, view));
307 r_splat_budget = p_streaming ? MIN(world, p_stream_records) : world;
308 r_record_budget = p_records - p_admission.records;
309}

311// One object's view demand: false when its bounding sphere (asset bounds under R s + T) is outside the frustum (the
312// object is not visible in this view: it reserves nothing and is not refused). Otherwise its importance (projected
313// radius in pixels) and its overlap-record estimate, with the selector's cost rule (lod/hlod_runtime_selection.cpp):
314// splats + footprint * f^2 / (d^2 * tile_area), d the distance to the near surface floored at
315// kCostInsideDistanceFraction of the radius.
316inline bool object_view_demand(const InstanceAsset &p_asset, const Vector3 &p_world_center, float p_uniform_scale,
317 const Plane *p_frustum, bool p_frustum_enabled, const Vector3 &p_camera, double p_focal_pixels, double p_tile_area,
318 double &r_importance, double &r_records) {
319 const double scale = std::fabs(double(p_uniform_scale));
320 const double radius = double(p_asset.bounds_radius) * scale;
321 if (p_frustum_enabled && p_frustum) {
322 for (int i = 0; i < 6; ++i) {
323 if (double(p_frustum[i].distance_to(p_world_center)) > radius) { return false; }
324 }
325 }
326 const double dx = double(p_world_center.x) - double(p_camera.x);
327 const double dy = double(p_world_center.y) - double(p_camera.y);
328 const double dz = double(p_world_center.z) - double(p_camera.z);
329 const double dist = std::sqrt(dx * dx + dy * dy + dz * dz);
330 r_importance = radius * p_focal_pixels / MAX(dist, 1.0e-6);
331 const double d = MAX(MAX(dist - radius, 1.0e-6), gs_hlod_runtime::kCostInsideDistanceFraction * radius);
332 const double tile_area = p_tile_area > 0.0 ? p_tile_area : 1.0;
333 r_records = double(p_asset.splat_count) +
334 p_asset.footprint * scale * scale * p_focal_pixels * p_focal_pixels / (MAX(d * d, 1.0e-12) * tile_area);
335 return true;
336}
