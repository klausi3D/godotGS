93 // Reserve a simultaneous-live upper bound BEFORE snapshot/allocation. Frozen default
94 // reduction=4 gives sum(interior counts) <= ceil(input/3) + number of interior nodes:
95 // 4 I <= N + I + 3 J. Every selected candidate has the merged candidate's count.
96 // LocalVector growth is <=1.5x; CowData rounds to <2x. The terms include old+new
97 // owners during growth/probing, not just logical payload sizes. Metadata is disallowed.
98 bool reserve(uint64_t p_count, uint64_t p_high, uint64_t p_old_interior_bytes)
99 {
100 reservation_valid = false;
101 if (!p_count || p_count > input || p_high > 21u || p_count > prepared) {
102 return false;
103 }
104 const uint64_t payload = sizeof(Gaussian) + p_high * sizeof(Vector3);
105 const uint64_t interiors = MIN(prepared - p_count, (p_count + 2u) / 3u + nodes);
106 const uint64_t children = MIN(p_count, uint64_t(kMaxChildren) * kMaxNodeSplats);
107 const uint64_t candidate = MIN(children, uint64_t(kMaxNodeSplats));
108 uint64_t remaining = scratch;
109 auto charge = [&](uint64_t count, uint64_t bytes) { return quality_charge_product(remaining, count, bytes); };
110 if (!charge(3u * p_count, payload) || // source, snapshot, reordered leaves
111 !charge(3u * interiors, payload) || // retained rounded candidates + flattened interior
112 !charge(6u * children + 10u * candidate, payload) || // merge/selection owners and growth
113 !charge(children, 128u) || // merge buckets, keys, members, orders and growth
114 !charge(p_count, 64u) || // permutations, index splits, importance, leaf chunk indices
115 !charge(nodes, 2048u) || // node/build/DFS/BFS tables, child vectors, validation, headers
116 !charge(metric.scratch_peak_bytes, 1u) || // existing old+new metric ledger
117 !charge(p_old_interior_bytes, 1u) || !charge(uint64_t(1) << 20u, 1u)) {
118 return false;
119 }
120 reserved = scratch - remaining;
121 // Strict resident reload overlaps the source and candidate with all/leaf/final
122 // loader copies. For larger caller-supplied caps it can exceed the bake peak.
123 uint64_t reload_remaining = scratch;
124 if (!quality_charge_product(reload_remaining, 5u * p_count + 3u * interiors, payload) ||
125 !quality_charge_product(reload_remaining, nodes, 2048u) ||
126 !quality_charge_product(reload_remaining, p_count, 64u) ||
127 !quality_charge_product(reload_remaining, p_old_interior_bytes, 1u) ||
128 !quality_charge_product(reload_remaining, uint64_t(1) << 20u, 1u)) {
129 return false;
130 }
131 reserved = MAX(reserved, scratch - reload_remaining);
132 admitted_count = p_count;
133 admitted_high = p_high;
134 reservation_valid = true;
135 return true;
136 }
