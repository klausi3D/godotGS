58// #610 S3: InstanceStore owns the per-scenario instance records, the
59// node->slot lookup, the instance/asset generation counters and the
60// asset-retention table. It is the boundary the prune predicate
61// (_world_has_no_instances) and the sibling decomposition slices see:
62// callers touch only these public methods, never the fields -- that is what
63// lets S4/S5/S8 be extracted independently.
64//
65// It takes NO lock of its own: every method is called under the director's
66// single world_mutex, preserving the one ThreadOwnedMutex ownership record
67// the renderer-contract-boundary guard depends on.
68//
69// InstanceRecord::last_lod deliberately stays inside the record and is
70// written in place by the render-thread LOD walk
71// (update_instance_lods_for_renderer). That LOD cache belongs to a later
72// slice (S9), so the store exposes mutable_records() for the walk and hands
73// out const views everywhere else.

99 // Mutable view for the S9-owned render-thread LOD walk, which edits
100 // last_lod / dirty in place. Callers MUST NOT push or remove through
101 // this handle -- membership changes go through append() / remove() so
102 // instance_lookup stays consistent.
103 LocalVector<InstanceRecord> &mutable_records() { return instances; }
104
105 // --- instance mutations (keep instance_lookup consistent) ---
106 void append(const InstanceRecord &p_record) {
107 instance_lookup[p_record.node_id] = instances.size();
108 instances.push_back(p_record);
109 }
110 // Swap-remove the record for p_node_id, fixing up the moved record's
111 // lookup slot. Returns false (and leaves r_asset_id untouched) when the
112 // node is absent; otherwise sets r_asset_id to the removed record's
113 // asset_id so the caller can release the matching asset record.
114 bool remove(ObjectID p_node_id, uint64_t &r_asset_id) {
115 uint32_t *index_ptr = instance_lookup.getptr(p_node_id);
116 if (!index_ptr || *index_ptr >= instances.size()) {
117 return false;
118 }
119 const uint32_t index = *index_ptr;
120 r_asset_id = instances[index].asset_id;
121 const uint32_t last_index = instances.size() - 1;
122 if (index != last_index) {
123 instances[index] = instances[last_index];
124 instance_lookup[instances[index].node_id] = index;
125 }
126 instances.remove_at(last_index);
127 instance_lookup.erase(p_node_id);
128 return true;
129 }
