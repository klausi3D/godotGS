120// Parameters for a sort operation
121struct SortOperationParams {
122 uint32_t element_count = 0;
123 RID keys_buffer;
124 // Values buffer contains indices sorted by key. When instance pipeline is enabled,
125 // values are indices into SplatRefBuffer; otherwise they refer to Gaussian indices.
126 RID values_buffer;
127 // Optional count buffer (IndirectDispatchLayout with element_count at offset 12).
128 // If the sorter supports indirect dispatch, the GPU reads the effective count from
129 // this buffer. Otherwise element_count must carry the CPU-known effective count so
130 // the pipeline can fall back to the direct-count path.
131 RID count_buffer;
132 bool async = true;
