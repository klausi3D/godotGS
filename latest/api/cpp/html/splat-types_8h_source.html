93struct CoordinateConverter {
94 std::array<float, 3> flipP = {1.0f, 1.0f, 1.0f}; // x, y, z flips.
95 std::array<float, 3> flipQ = {1.0f, 1.0f, 1.0f}; // x, y, z flips, w is never flipped.
96 std::array<float, 24> flipSh = // Flips for the 24 spherical harmonics coefficients.
97 {1.0f, 1.0f, 1.0f, 1.0f, 1.0f,
98 1.0f, 1.0f, 1.0f, 1.0f, 1.0f,
99 1.0f, 1.0f, 1.0f, 1.0f, 1.0f,
100 1.0f, 1.0f, 1.0f, 1.0f, 1.0f,
101 1.0f, 1.0f, 1.0f, 1.0f};
102 // Per-band SH transform; for cross-family, rotation and flip are baked in together.
103 // For within-family, these are empty and flipSh above is used instead.
104 std::array<std::function<void(float*)>, SH_MAX_DEGREE> rotFlipShFuncs = {};
105 std::function<void(float*)> rotFlipPFunc = nullptr;
106 std::function<void(float*)> rotFlipQFunc = nullptr;
107};

109constexpr std::array<bool, 3> axesMatch(CoordinateSystem a, CoordinateSystem b) {
110 auto aNum = static_cast<int>(a) - 1;
111 auto bNum = static_cast<int>(b) - 1;
112 if (aNum < 0 || bNum < 0) {
113 return {true, true, true};
114 }
115 return {
116 ((aNum >> 0) & 1) == ((bNum >> 0) & 1),
117 ((aNum >> 1) & 1) == ((bNum >> 1) & 1),
118 ((aNum >> 2) & 1) == ((bNum >> 2) & 1)};
119}

121constexpr bool needRotation(CoordinateSystem a, CoordinateSystem b) {
122 auto aNum = static_cast<int>(a) - 1;
123 auto bNum = static_cast<int>(b) - 1;
124 if (aNum < 0 || bNum < 0) {
125 return false;
126 }
127 return ((aNum >> 3) & 1) != ((bNum >> 3) & 1);
128}

130// SH rotation matrices for R_x(+π/2) and R_x(-π/2). Each entry operates on the 2l+1
131// coefficients of band l (l = 1..4). The minus table is the transpose (= inverse) of the plus
132// table since both are orthogonal matrices.
133inline constexpr std::array<AnalyticRotateShFn, 4> kAnalyticRotatePlusPiHalfAboutXTable = {
134 [](float* p) {
135 const float t0 = p[0], t1 = p[1], t2 = p[2];
136 p[0] = t1;
137 p[1] = -t0;
138 p[2] = t2;
139 },
140 [](float* p) {
141 std::array<float, 5> s{};
142 for (int i = 0; i < 5; ++i) {
143 s[static_cast<size_t>(i)] = p[i];
144 }
145 const float s3 = std::sqrt(3.f);
146 p[0] = s[3];
147 p[1] = -s[1];
148 p[2] = -0.5f * s[2] - (s3 / 2.f) * s[4];
149 p[3] = -s[0];
150 p[4] = -(s3 / 2.f) * s[2] + 0.5f * s[4];
151 },
152 [](float* p) {
153 std::array<float, 7> s{};
154 for (int i = 0; i < 7; ++i) {
155 s[static_cast<size_t>(i)] = p[i];
156 }
157 const float s15 = std::sqrt(15.f);
158 p[0] = -std::sqrt(5.f / 8.f) * s[3] + std::sqrt(3.f / 8.f) * s[5];
159 p[1] = -s[1];
160 p[2] = -std::sqrt(3.f / 8.f) * s[3] - std::sqrt(5.f / 8.f) * s[5];
161 p[3] = std::sqrt(5.f / 8.f) * s[0] + std::sqrt(3.f / 8.f) * s[2];
162 p[4] = -0.25f * s[4] - (s15 / 4.f) * s[6];
163 p[5] = -std::sqrt(3.f / 8.f) * s[0] + std::sqrt(5.f / 8.f) * s[2];
164 p[6] = -(s15 / 4.f) * s[4] + 0.25f * s[6];
165 },
166 [](float* p) {
167 std::array<float, 9> s{};
168 for (int i = 0; i < 9; ++i) {
169 s[static_cast<size_t>(i)] = p[i];
170 }
171 const float s2 = std::sqrt(2.f);
172 const float s5 = std::sqrt(5.f);
173 const float s7 = std::sqrt(7.f);
174 const float s14 = std::sqrt(14.f);
175 const float s35 = std::sqrt(35.f);
176 p[0] = -(s14 / 4.f) * s[5] + (s2 / 4.f) * s[7];
177 p[1] = -0.75f * s[1] + (s7 / 4.f) * s[3];
178 p[2] = -(s2 / 4.f) * s[5] - (s14 / 4.f) * s[7];
179 p[3] = (s7 / 4.f) * s[1] + 0.75f * s[3];
180 p[4] = (3.f / 8.f) * s[4] + (s5 / 4.f) * s[6] + (s35 / 8.f) * s[8];
181 p[5] = (s14 / 4.f) * s[0] + (s2 / 4.f) * s[2];
182 p[6] = (s5 / 4.f) * s[4] + 0.5f * s[6] - (s7 / 4.f) * s[8];
183 p[7] = -(s2 / 4.f) * s[0] + (s14 / 4.f) * s[2];
184 p[8] = (s35 / 8.f) * s[4] - (s7 / 4.f) * s[6] + 0.125f * s[8];
185 },
186};

188inline constexpr std::array<AnalyticRotateShFn, 4> kAnalyticRotateMinusPiHalfAboutXTable = {
189 [](float* p) {
190 const float t0 = p[0], t1 = p[1], t2 = p[2];
191 p[0] = -t1;
192 p[1] = t0;
193 p[2] = t2;
194 },
195 [](float* p) {
196 std::array<float, 5> s{};
197 for (int i = 0; i < 5; ++i) s[static_cast<size_t>(i)] = p[i];
198 const float s3 = std::sqrt(3.f);
199 p[0] = -s[3];
200 p[1] = -s[1];
201 p[2] = -0.5f * s[2] - (s3 / 2.f) * s[4];
202 p[3] = s[0];
203 p[4] = -(s3 / 2.f) * s[2] + 0.5f * s[4];
204 },
205 [](float* p) {
206 std::array<float, 7> s{};
207 for (int i = 0; i < 7; ++i) s[static_cast<size_t>(i)] = p[i];
208 const float s15 = std::sqrt(15.f);
209 p[0] = std::sqrt(5.f / 8.f) * s[3] - std::sqrt(3.f / 8.f) * s[5];
210 p[1] = -s[1];
211 p[2] = std::sqrt(3.f / 8.f) * s[3] + std::sqrt(5.f / 8.f) * s[5];
212 p[3] = -std::sqrt(5.f / 8.f) * s[0] - std::sqrt(3.f / 8.f) * s[2];
213 p[4] = -0.25f * s[4] - (s15 / 4.f) * s[6];
214 p[5] = std::sqrt(3.f / 8.f) * s[0] - std::sqrt(5.f / 8.f) * s[2];
215 p[6] = -(s15 / 4.f) * s[4] + 0.25f * s[6];
216 },
217 [](float* p) {
218 std::array<float, 9> s{};
219 for (int i = 0; i < 9; ++i) s[static_cast<size_t>(i)] = p[i];
220 const float s2 = std::sqrt(2.f);
221 const float s5 = std::sqrt(5.f);
222 const float s7 = std::sqrt(7.f);
223 const float s14 = std::sqrt(14.f);
224 const float s35 = std::sqrt(35.f);
225 p[0] = (s14 / 4.f) * s[5] - (s2 / 4.f) * s[7];
226 p[1] = -0.75f * s[1] + (s7 / 4.f) * s[3];
227 p[2] = (s2 / 4.f) * s[5] + (s14 / 4.f) * s[7];
228 p[3] = (s7 / 4.f) * s[1] + 0.75f * s[3];
229 p[4] = (3.f / 8.f) * s[4] + (s5 / 4.f) * s[6] + (s35 / 8.f) * s[8];
230 p[5] = -(s14 / 4.f) * s[0] - (s2 / 4.f) * s[2];
231 p[6] = (s5 / 4.f) * s[4] + 0.5f * s[6] - (s7 / 4.f) * s[8];
232 p[7] = (s2 / 4.f) * s[0] - (s14 / 4.f) * s[2];
233 p[8] = (s35 / 8.f) * s[4] - (s7 / 4.f) * s[6] + 0.125f * s[8];
234 },
235};

237inline CoordinateConverter coordinateConverter(CoordinateSystem from, CoordinateSystem to, int shDegree) {
238 CoordinateConverter result;
239
240 if (needRotation(from, to)) {
241 // A cross-family conversion decomposes into R_x(±π/2) plus a within-family flip.
242 // Shift `from` into the same family as `to` so the recursive call resolves to a flip only.
243 const bool backward = ((static_cast<int>(from) - 1) >> 3) & 1;
244 const CoordinateSystem innerFrom = backward
245 ? static_cast<CoordinateSystem>(static_cast<int>(from) - 8) // rotated→standard: shift from into standard family
246 : static_cast<CoordinateSystem>(static_cast<int>(from) + 8); // standard→rotated: shift from into rotated family
247 result = coordinateConverter(innerFrom, to, shDegree);
248
249 // Bake the inner flip into each transform function so callers never need to know the order.
250 // Forward (standard→rotated): rotate then flip.
251 // Backward (rotated→standard): flip then rotate.
252 const auto fp = result.flipP;
253 const auto fq = result.flipQ;
254 if (backward) {
255 result.rotFlipPFunc = [fp](float* p) {
256 p[0] *= fp[0]; p[1] *= fp[1]; p[2] *= fp[2];
257 const float y = p[1], z = p[2]; p[1] = z; p[2] = -y; // R_x(-π/2)
258 };
259 result.rotFlipQFunc = [fq](float* p) {
260 p[0] *= fq[0]; p[1] *= fq[1]; p[2] *= fq[2];
261 const float s = std::sqrt(2.f) / 2.f;
262 const float x = p[0], y = p[1], z = p[2], w = p[3];
263 p[0] = s * (-w + x); p[1] = s * (y + z); p[2] = s * (-y + z); p[3] = s * (w + x);
264 };
265 } else {
266 result.rotFlipPFunc = [fp](float* p) {
267 const float y = p[1], z = p[2]; p[1] = -z; p[2] = y; // R_x(+π/2)
268 p[0] *= fp[0]; p[1] *= fp[1]; p[2] *= fp[2];
269 };
270 result.rotFlipQFunc = [fq](float* p) {
271 const float s = std::sqrt(2.f) / 2.f;
272 const float x = p[0], y = p[1], z = p[2], w = p[3];
273 p[0] = s * (w + x); p[1] = s * (y - z); p[2] = s * (y + z); p[3] = s * (w - x);
274 p[0] *= fq[0]; p[1] *= fq[1]; p[2] *= fq[2];
275 };
276 }
277 result.flipP = {1.0f, 1.0f, 1.0f};
278 result.flipQ = {1.0f, 1.0f, 1.0f};
279
280 const AnalyticRotateShFn* shTable = backward
281 ? kAnalyticRotateMinusPiHalfAboutXTable.data()
282 : kAnalyticRotatePlusPiHalfAboutXTable.data();
283 for (int b = 0; b < shDegree && b < SH_MAX_DEGREE; ++b) {
284 const AnalyticRotateShFn rotFn = shTable[b];
285 const size_t bandStart = static_cast<size_t>(b * (b + 2));
286 const size_t bandSize = static_cast<size_t>(2 * b + 3);
287 std::array<float, 9> bandFlip{};
288 for (size_t k = 0; k < bandSize; ++k) bandFlip[k] = result.flipSh[bandStart + k];
289 if (backward) {
290 result.rotFlipShFuncs[static_cast<size_t>(b)] = [rotFn, bandFlip, bandSize](float* p) {
291 for (size_t k = 0; k < bandSize; ++k) p[k] *= bandFlip[k];
292 rotFn(p);
293 };
294 } else {
295 result.rotFlipShFuncs[static_cast<size_t>(b)] = [rotFn, bandFlip, bandSize](float* p) {
296 rotFn(p);
297 for (size_t k = 0; k < bandSize; ++k) p[k] *= bandFlip[k];
298 };
299 }
300 }
301 result.flipSh.fill(1.0f);
302 return result;
303 }
304
305 auto [xMatch, yMatch, zMatch] = axesMatch(from, to);
306 float x = xMatch ? 1.0f : -1.0f;
307 float y = yMatch ? 1.0f : -1.0f;
308 float z = zMatch ? 1.0f : -1.0f;
309
310 result.flipP = {x, y, z};
311 result.flipQ = {y * z, x * z, x * y};
312 result.flipSh = {
313 y, // 0
314 z, // 1
315 x, // 2
316 x * y, // 3
317 y * z, // 4
318 1.0f, // 5
319 x * z, // 6
320 1.0f, // 7
321 y, // 8
322 x * y * z, // 9
323 y, // 10
324 z, // 11
325 x, // 12
326 z, // 13
327 x, // 14
328 // Used https://github.com/nerfstudio-project/gsplat/blob/main/gsplat/cuda/csrc/SphericalHarmonicsCUDA.cu
329 // to compute these values.
330 x * y, // 15
331 y * z, // 16
332 x * y, // 17
333 y * z, // 18
334 1.0f, // 19
335 x * z, // 20
336 1.0f, // 21
337 x * z, // 22
338 y, // 23
339 };
340 return result;
341}

343// A point cloud composed of Gaussians. Each gaussian is represented by:
344// - xyz position
345// - xyz scales (on log scale, compute exp(x) to get scale factor)
346// - xyzw quaternion
347// - alpha (before sigmoid activation, compute sigmoid(a) to get alpha value between 0 and 1)
348// - rgb color (as SH DC component, compute 0.5 + 0.282095 * x to get color value between 0 and 1)
349// - 0 to 71 spherical harmonics coefficients (see comment below)
350struct GaussianCloud {
351 // Total number of points (gaussians) in this splat.
352 int32_t numPoints = 0;
353
354 // Degree of spherical harmonics for this splat.
355 int32_t shDegree = 0;
356
357 // Whether the gaussians should be rendered in antialiased mode (mip splatting)
358 bool antialiased = false;
359
360 // See block comment above for details
361 std::vector<float> positions;
362 std::vector<float> scales;
363 std::vector<float> rotations;
364 std::vector<float> alphas;
365 std::vector<float> colors;
366
367 // Spherical harmonics coefficients. The number of coefficients per point depends on shDegree:
368 // 0 -> 0
369 // 1 -> 9 (3 coeffs x 3 channels)
370 // 2 -> 24 (8 coeffs x 3 channels)
371 // 3 -> 45 (15 coeffs x 3 channels)
372 // 4 -> 72 (24 coeffs x 3 channels)
373 // The color channel is the inner (fastest varying) axis, and the coefficient is the outer
374 // (slower varying) axis, i.e. for degree 1, the order of the 9 values is:
375 // sh1n1_r, sh1n1_g, sh1n1_b, sh10_r, sh10_g, sh10_b, sh1p1_r, sh1p1_g, sh1p1_b
376 std::vector<float> sh;
377
378#ifdef SPZ_BUILD_EXTENSIONS
379 std::vector<SpzExtensionBasePtr> extensions; // List of extensions, if any
380#endif
381
382 // The caller is responsible for freeing the pointers in the returned GaussianCloudData
383 GaussianCloudData data() const {
384 GaussianCloudData data;
385 data.numPoints = numPoints;
386 data.shDegree = shDegree;
387 data.antialiased = antialiased;
388 data.positions = copyFloatBuffer(positions);
389 data.scales = copyFloatBuffer(scales);
390 data.rotations = copyFloatBuffer(rotations);
391 data.alphas = copyFloatBuffer(alphas);
392 data.colors = copyFloatBuffer(colors);
393 data.sh = copyFloatBuffer(sh);
394#ifdef SPZ_BUILD_EXTENSIONS
395 data.extensions = copyExtensions(extensions);
396#else
397 data.extensions = nullptr;
398#endif
399 return data;
400 }

402 // Convert between two coordinate systems, for example from RDF (ply format) to RUB (used by spz).
403 // This is performed in-place.
404 void convertCoordinates(CoordinateSystem from, CoordinateSystem to) {
405 if (numPoints == 0) {
406 // There is nothing to convert.
407 return;
408 }
409 CoordinateConverter c = coordinateConverter(from, to, shDegree);
410 // Positions: call transform (which includes flip for cross-family), else apply flipP.
411 if (c.rotFlipPFunc) {
412 for (size_t i = 0; i < positions.size(); i += 3) {
413 c.rotFlipPFunc(positions.data() + i);
414 }
415 } else {
416 for (size_t i = 0; i < positions.size(); i += 3) {
417 positions[i + 0] *= c.flipP[0];
418 positions[i + 1] *= c.flipP[1];
419 positions[i + 2] *= c.flipP[2];
420 }
421 }
422 // Rotations: same pattern. Within-family only flips x,y,z; w is untouched. Cross-family rotates all four.
423 if (c.rotFlipQFunc) {
424 for (size_t i = 0; i < rotations.size(); i += 4) {
425 c.rotFlipQFunc(rotations.data() + i);
426 }
427 } else {
428 for (size_t i = 0; i < rotations.size(); i += 4) {
429 rotations[i + 0] *= c.flipQ[0];
430 rotations[i + 1] *= c.flipQ[1];
431 rotations[i + 2] *= c.flipQ[2];
432 // Don't modify rotations[i + 3] (w component)
433 }
434 }
435 const size_t numCoeffs = sh.size() / 3;
436 const size_t numCoeffsPerPoint = numCoeffs / numPoints;
437 if (c.rotFlipShFuncs[0]) {
438 // Cross-family: rotation+flip baked into rotFlipShFuncs per band.
439 for (size_t coeffBase = 0; coeffBase < numCoeffs; coeffBase += numCoeffsPerPoint) {
440 for (int band = 0; band < shDegree && band < SH_MAX_DEGREE; ++band) {
441 const size_t bandStart = static_cast<size_t>(band * (band + 2));
442 const size_t bandSize = static_cast<size_t>(2 * band + 3);
443 if (bandStart + bandSize > numCoeffsPerPoint) { break; }
444 std::array<float, 9> tmp{};
445 for (int channel = 0; channel < 3; ++channel) {
446 for (size_t k = 0; k < bandSize; ++k) {
447 tmp[k] = sh[(coeffBase + bandStart + k) * 3 + static_cast<size_t>(channel)];
448 }
449 c.rotFlipShFuncs[static_cast<size_t>(band)](tmp.data());
450 for (size_t k = 0; k < bandSize; ++k) {
451 sh[(coeffBase + bandStart + k) * 3 + static_cast<size_t>(channel)] = tmp[k];
452 }
453 }
454 }
455 }
456 } else {
457 // Within-family: rotate spherical harmonics by inverting coefficients that reference the
458 // y and z axes, for each RGB channel. See spherical_harmonics_kernel_impl.h for spherical
459 // harmonics formulas.
460 for (size_t coeffBase = 0; coeffBase < numCoeffs; coeffBase += numCoeffsPerPoint) {
461 for (size_t j = 0; j < numCoeffsPerPoint; ++j) {
462 const size_t base = (coeffBase + j) * 3;
463 sh[base + 0] *= c.flipSh[j];
464 sh[base + 1] *= c.flipSh[j];
465 sh[base + 2] *= c.flipSh[j];
466 }
467 }
468 }
469 }

475 float medianVolume() const {
476 if (numPoints == 0) {
477 return 0.01f;
478 }
479 // The volume of an ellipsoid is 4/3 * pi * x * y * z, where x, y, and z are the radii on each
480 // axis. Scales are stored on a log scale, and exp(x) * exp(y) * exp(z) = exp(x + y + z). So we
481 // can sort by value = (x + y + z) and compute volume = 4/3 * pi * exp(value) later.
482 std::vector<float> scaleSums;
483 scaleSums.reserve(scales.size() / 3);
484 for (size_t i = 0; i < scales.size(); i += 3) {
485 scaleSums.push_back(scales[i] + scales[i + 1] + scales[i + 2]);
486 }
487 const auto mid = scaleSums.begin() + scaleSums.size() / 2;
488 std::nth_element(scaleSums.begin(), mid, scaleSums.end());
489 return (M_PI * 4 / 3) * exp(*mid);
490 }

525constexpr Vec3f times(const Quat4f &q, const Vec3f &p) {
526 auto [w, x, y, z] = q;
527 auto [vx, vy, vz] = p;
528 auto x2 = x + x;
529 auto y2 = y + y;
530 auto z2 = z + z;
531 auto wx2 = w * x2;
532 auto wy2 = w * y2;
533 auto wz2 = w * z2;
534 auto xx2 = x * x2;
535 auto xy2 = x * y2;
536 auto xz2 = x * z2;
537 auto yy2 = y * y2;
538 auto yz2 = y * z2;
539 auto zz2 = z * z2;
540 return {
541 vx * (1.0f - (yy2 + zz2)) + vy * (xy2 - wz2) + vz * (xz2 + wy2),
542 vx * (xy2 + wz2) + vy * (1.0f - (xx2 + zz2)) + vz * (yz2 - wx2),
543 vx * (xz2 - wy2) + vy * (yz2 + wx2) + vz * (1.0f - (xx2 + yy2))};
544}
