18static inline int _get_int_setting(ProjectSettings *ps, const StringName &name, int fallback) {
19 ERR_FAIL_NULL_V(ps, fallback);
20 if (!ps->has_setting(name)) {
21 return fallback;
22 }
23 Variant value = ps->get_setting_with_override(name);
24 if (value.get_type() == Variant::INT) {
25 return static_cast<int>(value.operator int64_t());
26 }
27 if (value.get_type() == Variant::FLOAT) {
28 return static_cast<int>(value.operator double());
29 }
30 if (value.get_type() == Variant::BOOL) {
31 return value.operator bool() ? 1 : 0;
32 }
33 return fallback;
34}

185// Safe free with source tracking - logs before freeing so we can see where invalid frees come from
186#define GS_SAFE_FREE(device, rid, tag) \
187 do { \
188 if ((rid).is_valid() && (device) != nullptr) { \
189 if (GaussianSplatting::is_debug_resource_logging_enabled()) { \
190 GS_LOG_GPU_MEMORY_DEBUG(vformat("[GS_FREE] %s:%d [%s] freeing RID %llu", __FILE__, __LINE__, tag, (uint64_t)(rid).get_id())); \
191 } \
192 (device)->free(rid); \
193 rid = RID(); \
194 } \
195 } while (0)
196
197// Conditional free - only frees if RID is valid, with tracking
198#define GS_SAFE_FREE_IF_VALID(device, rid, tag) \
199 do { \
200 if ((rid).is_valid()) { \
201 if ((device) != nullptr) { \
202 if (GaussianSplatting::is_debug_resource_logging_enabled()) { \
203 GS_LOG_GPU_MEMORY_DEBUG(vformat("[GS_FREE] %s:%d [%s] freeing RID %llu", __FILE__, __LINE__, tag, (uint64_t)(rid).get_id())); \
204 } \
205 (device)->free(rid); \
206 } else { \
207 if (GaussianSplatting::is_debug_resource_logging_enabled()) { \
208 GS_LOG_GPU_MEMORY_DEBUG(vformat("[GS_FREE] %s:%d [%s] SKIP (null device) RID %llu", __FILE__, __LINE__, tag, (uint64_t)(rid).get_id())); \
209 } \
210 } \
211 rid = RID(); \
212 } \
213 } while (0)
