233 // Scratch texture used to break the R/W hazard in the composite path: destination
234 // is copied here before the compute dispatch, the shader reads from this via a
235 // sampler, and only writes the destination (which is writeonly inside the shader).
236 // `format` is the canonical storage format used to allocate the scratch (SRGB
237 // inputs get swapped to UNORM equivalent to avoid sampler double-decode); the
238 // pool reuses entries across destinations that share the canonical format.
239 // `last_used_id` is bumped on every hit/insert and consumed by the LRU eviction.
240 struct ViewportBlitScratch {
241 RID texture;
242 Size2i extent;
243 RD::DataFormat format = RD::DATA_FORMAT_MAX;
244 RenderingDevice *owner_device = nullptr;
245 uint64_t last_used_id = 0;
246 };
247 // Per-(device, canonical-format) entry cap. Viewport resizes in the editor
248 // create one scratch entry per intermediate extent; without an LRU the pool
249 // accumulates indefinitely until shutdown. 4 is a balance between bounded
250 // memory (~32 MiB at 1080p RGBA8) and absorbing dock-resize trajectories.
251 static constexpr uint32_t VIEWPORT_BLIT_SCRATCH_MAX_PER_KIND = 4;
252 uint64_t viewport_blit_scratch_next_id = 0;
253
254 // Per-cache entry cap shared by `cached_framebuffers` and
255 // `framebuffer_validation_cache`. Both are keyed by
256 // (RenderingDevice*, attachment-RID-set) so multi-device sessions
257 // (multi-window editor, RenderDoc capture, project reload) accumulate one
258 // entry per unique attachment combination, never evicting. The scratch
259 // pool's matching cap is VIEWPORT_BLIT_SCRATCH_MAX_PER_KIND=4 per kind;
260 // these caches are not partitioned by kind so 8 is the global ceiling. On
261 // insert past the cap we evict the entry with the lowest `last_access_id`
262 // via the existing destroy path (see get_cached_framebuffer and
263 // validate_framebuffer_attachments).
264 static constexpr uint32_t MAX_CACHED_FRAMEBUFFER_FORMATS = 8;
265 uint64_t cached_framebuffer_next_id = 0;
266 uint64_t framebuffer_validation_next_id = 0;
