119// Builds exclusive tile ranges and indirect dispatch values from tile counts.
120// `p_out_ranges_words` must have `p_total_tiles * 2` uint32 slots.
121inline bool compute_tile_prefix_cpu(const uint32_t *p_tile_counts, uint32_t p_total_tiles,
122 uint32_t *p_out_ranges_words, uint32_t p_global_sort_capacity, TilePrefixCpuScanResult &r_result) {
123 if (p_tile_counts == nullptr || p_out_ranges_words == nullptr) {
124 return false;
125 }
126
127 uint64_t prefix64 = 0;
128 const uint64_t max_u32 = uint64_t(std::numeric_limits<uint32_t>::max());
129
130 for (uint32_t i = 0; i < p_total_tiles; i++) {
131 const uint32_t pair_index = i * 2u;
132 const uint32_t prefix_u32 = prefix64 > max_u32 ? std::numeric_limits<uint32_t>::max() : uint32_t(prefix64);
133 p_out_ranges_words[pair_index + 0u] = prefix_u32;
134 p_out_ranges_words[pair_index + 1u] = p_tile_counts[i];
135 prefix64 += p_tile_counts[i];
136 }
137
138 r_result.raw_total_saturated = prefix64 > max_u32;
139 r_result.raw_total = r_result.raw_total_saturated ? std::numeric_limits<uint32_t>::max() : uint32_t(prefix64);
140
141 const bool overflow = (p_global_sort_capacity > 0u) && (r_result.raw_total > p_global_sort_capacity);
142 if (overflow) {
143 static uint32_t prefix_overflow_throttle = 0;
144 if (++prefix_overflow_throttle % 300 == 1) {
145 WARN_PRINT(vformat("[TilePrefixScan] Overlap record count %d exceeds global sort capacity %d; "
146 "%d records dropped. Consider increasing max_overlap_records.",
147 r_result.raw_total, p_global_sort_capacity,
148 r_result.raw_total - p_global_sort_capacity));
149 }
150 }
151 const uint32_t clamped_total = overflow ? p_global_sort_capacity : r_result.raw_total;
152 uint64_t groups_x64 = (uint64_t(clamped_total) + uint64_t(kTilePrefixPassLocalSize) - 1u) / uint64_t(kTilePrefixPassLocalSize);
153 if (groups_x64 == 0u) {
154 groups_x64 = 1u;
155 }
156 const uint32_t groups_x = groups_x64 > max_u32 ? std::numeric_limits<uint32_t>::max() : uint32_t(groups_x64);
157
158 r_result.indirect_dispatch.dispatch_x = groups_x;
159 r_result.indirect_dispatch.dispatch_y = 1u;
160 r_result.indirect_dispatch.dispatch_z = 1u;
161 r_result.indirect_dispatch.element_count = clamped_total;
162 r_result.indirect_dispatch.overflow_flag = overflow ? 1u : 0u;
163 r_result.indirect_dispatch.unclamped_total = r_result.raw_total;
164 return true;
165}
