65// Theoretical tree-size validator and each tile's OfflineJob bound. Resident
66// metadata is charged by actual node count, not this conservative bound. Large
67// interiors have > leaf_max descendants, so at any depth there are at most
68// floor(N/(leaf_max+1)). Index splitting can additionally emit a small unary
69// interior over one chunk. There is at most one such interior per index chunk;
70// disjoint index roots have at most 2*ceil(N/leaf_max) chunks in total. Every
71// interior has at most eight children; height is bounded by kMaxHeight.
72inline uint64_t node_bound(uint32_t p_count) {
73 if (p_count <= kMaxNodeSplats) {
74 return 1u;
75 }
76 const uint64_t large = uint64_t(p_count / (kMaxNodeSplats + 1u)) * kMaxHeight;
77 const uint64_t small = 2u * ((uint64_t(p_count) + kMaxNodeSplats - 1u) / kMaxNodeSplats);
78 return 1u + (large + small) * kMaxChildren;
79}
