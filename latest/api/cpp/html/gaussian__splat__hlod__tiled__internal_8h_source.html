56// Same upper bound for metadata admission and each tile's OfflineJob. Large
57// interiors have > leaf_max descendants, so at any depth there are at most
58// floor(N/(leaf_max+1)). Index splitting can additionally emit a small unary
59// interior over one chunk. There is at most one such interior per index chunk;
60// disjoint index roots have at most 2*ceil(N/leaf_max) chunks in total. Every
61// interior has at most eight children; height is bounded by kMaxHeight.
62inline uint64_t node_bound(uint32_t p_count) {
63 if (p_count <= kMaxNodeSplats) {
64 return 1u;
65 }
66 const uint64_t large = uint64_t(p_count / (kMaxNodeSplats + 1u)) * kMaxHeight;
67 const uint64_t small = 2u * ((uint64_t(p_count) + kMaxNodeSplats - 1u) / kMaxNodeSplats);
68 return 1u + (large + small) * kMaxChildren;
69}
