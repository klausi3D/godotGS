39// ---------------------------------------------------------------------------
40// Unsorted global-composite fallback — reject + observability contract (#586)
41// ---------------------------------------------------------------------------
42//
43// The global-composite path can end up rasterizing tiles in UNSORTED order, which
44// is WRONG output: alpha compositing is order-dependent, so an unsorted draw is
45// not "slightly off", it is incorrect.
46//
47// #586: the PERMANENT class of that failure — no sorter exists at all
48// (TileGlobalSortResources::sorter is invalid) — no longer rasterizes.
49// `unsorted_composite_must_reject_frame()` below marks it fatal for the frame,
50// and the choke point in tile_renderer.cpp turns that into a publish REJECT:
51// render() returns an invalid RID and the caller presents nothing for the
52// frame. The TRANSIENT reasons (sorter exists, one dispatch failed) keep the
53// counter + throttled WARN and are still presented; see the rationale on
54// `unsorted_composite_must_reject_frame`.
55//
56// DESIGN RULE (this is the point of the refactor): the "is this frame producing
57// unsorted output?" decision is derived from exactly TWO pure inputs computed once
58// per frame — `global_composite_has_translucent_work()` and a
59// `GlobalSortAttemptOutcome` — and classified at ONE choke point by
60// `classify_unsorted_composite()`. Instrumentation must NEVER be re-derived at the
61// individual failure sites: an earlier revision did that and the counter's copy of
62// the work predicate silently omitted the indirect-work term, so GPU-driven frames
63// (CPU splat_count == 0, real work described by the instance indirect-dispatch
64// buffer) rasterized unsorted WITHOUT being counted. An under-reporting counter is
65// worse than no counter — it manufactures confidence that the wrong-output path is
66// rare. Adding a new sort-failure mode must mean adding a `GlobalSortAttemptOutcome`
67// value, which this switch then forces you to classify.
68
69// Why the sort did not produce a correctly-ordered result for this frame.
70enum class UnsortedCompositeReason : uint8_t {
71 NONE = 0, // Output is correctly sorted (or there was nothing to composite).
72 SORTER_UNAVAILABLE = 1, // No sorter could be built at all (probe false, creation failed, or no indirect entry point).
73 SORT_DISPATCH_FAILED = 2, // Sorter exists; the sync fallback dispatch returned an error.
74 ASYNC_SORT_NOT_SUBMITTED = 3, // Async submit returned timeline=0 and sync fallback is disallowed.
75 // NOT produced by classify_unsorted_composite(): the frame is abandoned BEFORE a sort
76 // outcome can exist, at the pre-binning resource check in tile_renderer.cpp. It is
77 // still a REJECT of a frame that carried translucent work (render() returns an invalid
78 // RID), and it is the exit a VRAM-pressure sorter failure takes: disable_sorter() frees
79 // the key/value buffers and the same pressure then fails their reallocation. Recorded
80 // so that reject telemetry can attribute it instead of counting nothing (#586).
81 GLOBAL_SORT_RESOURCES_UNAVAILABLE = 4,
82};

100// What actually happened at the sort-dispatch site this frame.
101enum class GlobalSortAttemptOutcome : uint8_t {
102 NOT_ATTEMPTED = 0, // No sort dispatched (sorter invalid, or no work).
103 SUBMITTED = 1, // Async indirect sort submitted (timeline != 0) -> ordered output.
104 SYNC_FALLBACK_OK = 2, // Async returned 0; sync fallback ran and succeeded -> ordered output.
105 SYNC_FALLBACK_FAILED = 3, // Async returned 0; sync fallback ran and failed -> UNSORTED.
106 NOT_SUBMITTED = 4, // Async returned 0 and sync fallback disallowed -> UNSORTED.
107};

109// THE single definition of "this frame has translucent content the global-composite
110// path will rasterize". The sort gate and the unsorted-output counter MUST both be
111// derived from this one call — never from a hand-copied expression.
112//
113// In sync-readback mode the CPU has a fresh overlap-record count, so that is
114// authoritative. In async (GPU-driven) mode the CPU count is stale by design, and
115// work may be described ONLY by the instance indirect-dispatch buffer, with
116// CPU-side splat_count reading 0. Dropping that `p_has_instance_indirect` term is
117// exactly the under-reporting bug this function exists to prevent.
118static inline bool global_composite_has_translucent_work(
119 bool p_allow_sync_readback,
120 uint32_t p_overlap_record_count,
121 uint32_t p_splat_count,
122 bool p_has_instance_indirect) {
123 if (p_allow_sync_readback) {
124 return p_overlap_record_count > 0;
125 }
126 return p_splat_count > 0 || p_has_instance_indirect;
127}

129// The single choke point: map (work present, sort outcome) -> wrong-output reason.
130// If there is no translucent work, nothing is composited, so no frame can be wrong.
131static inline UnsortedCompositeReason classify_unsorted_composite(
132 bool p_has_translucent_work,
133 GlobalSortAttemptOutcome p_outcome) {
134 if (!p_has_translucent_work) {
135 return UnsortedCompositeReason::NONE;
136 }
137 switch (p_outcome) {
138 case GlobalSortAttemptOutcome::NOT_ATTEMPTED:
139 // Work exists but no sort was dispatched -> the sorter was unavailable.
140 return UnsortedCompositeReason::SORTER_UNAVAILABLE;
141 case GlobalSortAttemptOutcome::SUBMITTED:
142 case GlobalSortAttemptOutcome::SYNC_FALLBACK_OK:
143 return UnsortedCompositeReason::NONE;
144 case GlobalSortAttemptOutcome::SYNC_FALLBACK_FAILED:
145 return UnsortedCompositeReason::SORT_DISPATCH_FAILED;
146 case GlobalSortAttemptOutcome::NOT_SUBMITTED:
147 return UnsortedCompositeReason::ASYNC_SORT_NOT_SUBMITTED;
148 }
149 return UnsortedCompositeReason::NONE;
150}

152// #586 FIX: which unsorted-composite reasons must REJECT the frame (nothing is
153// published) instead of presenting it. This is the whole correctness decision,
154// isolated as one pure function so it is testable without a GPU, and so adding an
155// UnsortedCompositeReason forces an explicit fatal / non-fatal choice here: the
156// switch has no default label (a new enumerator is a compiler warning), and the
157// fall-through after it FAILS CLOSED — an enumerator this function does not know
158// is rejected, never presented.
159//
160// SORTER_UNAVAILABLE is FATAL for the frame that observes it. It is the class where
161// no sorter exists at all, so EVERY frame that reaches the composite in this state
162// would order translucent splats by atomic-append order. Presenting nothing is
163// honest; presenting wrong alpha ordering that looks like a normal render is not.
164// This is a PER-FRAME verdict: whether the state persists is decided by
165// TileGlobalSortResources::ensure_resources (at this revision it latches until
166// reset_state(); making it retry is the follow-up PR for #586).
167//
168// GLOBAL_SORT_RESOURCES_UNAVAILABLE is recorded at an exit that has ALREADY
169// abandoned the frame, so "must reject" is a statement of fact about it; it is
170// listed as fatal so the predicate stays total and a future caller that did have
171// a choice could not present a frame that was never binned.
172//
173// The two TRANSIENT reasons are deliberately NOT fatal:
174// * SORT_DISPATCH_FAILED — the sorter EXISTS and the device is capable; the
175// sync fallback dispatch returned an error for one frame.
176// * ASYNC_SORT_NOT_SUBMITTED — async submit returned timeline=0 and the sync
177// fallback is disallowed by the readback policy.
178// Both are per-frame and recover on the next frame on working hardware; rejecting
179// them would turn a one-frame ordering artefact into a visible hitch. They keep the
180// counter + throttled WARN (observable, not silent).
181static inline bool unsorted_composite_must_reject_frame(UnsortedCompositeReason p_reason) {
182 switch (p_reason) {
183 case UnsortedCompositeReason::SORTER_UNAVAILABLE:
184 case UnsortedCompositeReason::GLOBAL_SORT_RESOURCES_UNAVAILABLE:
185 return true;
186 case UnsortedCompositeReason::NONE:
187 case UnsortedCompositeReason::SORT_DISPATCH_FAILED:
188 case UnsortedCompositeReason::ASYNC_SORT_NOT_SUBMITTED:
189 return false;
190 }
191 return true; // Fail closed: an unclassified reason is never presented.
192}

194// Warning throttle for the unsorted global-composite fallback.
195//
196// Every frame classified as wrong-output bumps a persistent counter
197// (TilePerformanceMetrics::unsorted_composite_frames, surfaced via
198// get_binning_debug_counters()); this predicate rate-limits the paired WARN so the
199// degradation is visible in production without one log line per frame. It fires on
200// the first degraded frame and then once per interval — NOT one-shot-forever, so a
201// persistent degradation keeps re-surfacing in the log. The reject counter
202// (TilePerformanceMetrics::global_composite_rejected_frames) uses the same throttle.
203//
204// Argument is the post-increment counter value (first degraded frame == 1).
205static constexpr uint64_t UNSORTED_COMPOSITE_WARN_INTERVAL_FRAMES = 300;
206
207static inline bool should_warn_unsorted_composite(uint64_t p_unsorted_composite_frames) {
208 if (p_unsorted_composite_frames == 0) {
209 return false;
210 }
211 return p_unsorted_composite_frames == 1 ||
212 (p_unsorted_composite_frames % UNSORTED_COMPOSITE_WARN_INTERVAL_FRAMES) == 0;
213}

215// Sort fallback policy. The instance domain has no safe unsorted fallback, so
216// the only options are reuse-previous or fail. The global domain keeps the CPU
217// sort as the correctness-preserving fallback when the GPU sort fails.
218static inline SortFallbackPolicyDecision build_sort_fallback_policy(
219 SortFallbackScenario p_scenario, bool p_instance_pipeline_active, bool /*p_strict_global_sort*/ = false) {
220 SortFallbackPolicyDecision decision;
221 auto push_action = [&](SortFallbackAction p_action) {
222 if (decision.action_count < 4) {
223 decision.actions[decision.action_count++] = p_action;
224 }
225 };
226
227 switch (p_scenario) {
228 case SortFallbackScenario::FORCE_CPU_OVERRIDE: {
229 if (p_instance_pipeline_active) {
230 push_action(SortFallbackAction::REUSE_PREVIOUS_SORT);
231 push_action(SortFallbackAction::FAIL);
232 } else {
233 decision.cpu_sort_forced = true;
234 push_action(SortFallbackAction::RUN_CPU_SORT);
235 push_action(SortFallbackAction::FAIL);
236 }
237 } break;
238 case SortFallbackScenario::SORTER_UNAVAILABLE:
239 case SortFallbackScenario::GPU_SORT_FAILED: {
240 push_action(SortFallbackAction::REUSE_PREVIOUS_SORT);
241 if (!p_instance_pipeline_active) {
242 push_action(SortFallbackAction::RUN_CPU_SORT);
243 }
244 push_action(SortFallbackAction::FAIL);
245 } break;
246 }
247
248 return decision;
249}

251// ---------------------------------------------------------------------------
252// Sorter-init retry policy (GPU-003, refs #922; tile sorter refs #586)
253// ---------------------------------------------------------------------------
254//
255// !! SHARED BY TWO SORTERS. The constants and predicates in this block are
256// consumed by BOTH
257// * RenderSortingOrchestrator::refresh_gpu_sorter (instance/depth sorter, GPU-003), and
258// * TileGlobalSortResources::ensure_resources (tile-overlap sorter, #586 PR 2).
259// Editing a constant or a predicate here moves both retry schedules at once.
260// That is deliberate -- two sorters with two policies for the same failure
261// class is where divergence bugs live -- but it must be impossible to miss
262// when reading this header, hence this paragraph.
263//
264// Repeated GPU-sorter creation failures (typically transient VRAM pressure at
265// startup, or device contention) previously tripped a ONE-WAY latch: after 5
266// failures refresh_gpu_sorter() early-returned forever, so GPU sorting stayed
267// dead for the whole session even after memory freed -- and the instance route
268// has no CPU fallback (see build_sort_fallback_policy above), so splats froze
269// at the last sorted order or stopped rendering entirely. The tile-overlap
270// sorter had the same shape with a worse consequence: its disable_sorter()
271// latched sorter_available=false until renderer teardown, and every
272// translucent global-composite frame was composited unsorted (before #976) or
273// rejected (after #976) for the rest of the session.
274//
275// The replacement policy: capped exponential backoff. Below the cap the
276// backoff doubles per consecutive failure (60, 120, 240, 480, 960 frames); at
277// SORTER_INIT_DEGRADED_FAILURE_THRESHOLD failures the subsystem counts as
278// DEGRADED and the capped interval becomes a periodic recovery probe. There is
279// deliberately NO failure count that stops retrying: the cap bounds the retry
280// RATE (which is all the old latch's OOM-thrash concern needs), never the
281// retry COUNT. Degradation is additionally surfaced through
282// record_rendering_error (diagnostics) on the instance route, not only a log
283// line -- see RenderSortingOrchestrator::refresh_gpu_sorter. The tile sorter
284// has no diagnostics sink of its own; it surfaces the state through the
285// global_composite_rejected_frames / global_sort_sorter_init_failures counters
286// and the same throttled log lines.
287//
288// What this policy deliberately does NOT do (recorded here, not only in a PR
289// thread, per the repository's "disproportionate, not unfixable" rule): it does
290// not classify a creation failure as transient vs deterministic and retry only
291// the transient class, and it does not lift a latch on a configuration change
292// by signature. That refinement (the shape PR #852 built, and the Codex
293// finding "Stop retrying deterministic sorter creation failures") is FEASIBLE
294// and is deferred to v1.0 -- it was the highest-defect-density component of
295// #852 (5 of its 16 review findings), and the public-alpha bar blocks on the
296// user-visible defect this policy fixes, not on the precision of which
297// failures get re-probed. The model under which the deferral is acceptable:
298// every attempt is rate-bounded by the cap; the predictable deterministic
299// failures (capability probe, workgroup limit, buffer-size limit) are caught by
300// allocation-free preflights BEFORE create_sorter() is called, so a doomed
301// attempt of that class costs a few device-limit queries; only a transient
302// allocation failure (where retrying is the point) or a deterministic
303// shader/pipeline build failure (which no preflight can predict) reaches the
304// compile path, measured at 6-14 ms warm / ~120 ms cold per attempt on an RTX
305// 3090 -- once per 1800 frames at the cap, on a renderer that is presenting
306// nothing anyway. A config change is picked up by the next scheduled probe,
307// because every attempt re-reads the live configuration.
308
309static constexpr uint64_t SORTER_INIT_BASE_BACKOFF_FRAMES = 60;
310static constexpr uint64_t SORTER_INIT_MAX_BACKOFF_FRAMES = 1800; // ~30 s at 60 fps
311static constexpr uint32_t SORTER_INIT_DEGRADED_FAILURE_THRESHOLD = 5;
312
313// Backoff (in frames) imposed after the p_failure_count-th consecutive
314// sorter-init failure. 0 failures -> no backoff.
315static inline uint64_t sorter_init_backoff_frames(uint32_t p_failure_count) {
316 if (p_failure_count == 0) {
317 return 0;
318 }
319 // 60 << 5 already exceeds the cap, so bounding the doubling exponent at 5
320 // keeps the shift well-defined for arbitrarily large failure counts.
321 const uint32_t doublings = (p_failure_count - 1) < 5 ? (p_failure_count - 1) : 5;
322 const uint64_t backoff = SORTER_INIT_BASE_BACKOFF_FRAMES << doublings;
323 return backoff < SORTER_INIT_MAX_BACKOFF_FRAMES ? backoff : SORTER_INIT_MAX_BACKOFF_FRAMES;
324}
325
326// Whether refresh_gpu_sorter() should attempt sorter creation this frame.
327// INVARIANT (the fix for GPU-003): for EVERY failure count, this returns true
328// once SORTER_INIT_MAX_BACKOFF_FRAMES frames have elapsed since the last
329// failure -- there is no permanently-disabled state.
330static inline bool should_attempt_sorter_init(uint32_t p_failure_count, uint64_t p_frames_since_last_failure) {
331 if (p_failure_count == 0) {
332 return true;
333 }
334 return p_frames_since_last_failure >= sorter_init_backoff_frames(p_failure_count);
335}

337// Degraded = the old policy's "permanently disabled" threshold. Used only for
338// escalated observability (ERROR log + RenderingError), never to stop retrying.
339static inline bool sorter_init_degraded(uint32_t p_failure_count) {
340 return p_failure_count >= SORTER_INIT_DEGRADED_FAILURE_THRESHOLD;
341}
