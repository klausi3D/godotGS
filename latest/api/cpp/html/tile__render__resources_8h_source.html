21// Growth/shrink policy for an only-grow GPU scratch buffer. The buffer grows
22// immediately to demand (plus optional slack); it shrinks back only after demand
23// stays at or below `shrink_trigger_percent` of the current capacity for
24// `shrink_hysteresis_frames` consecutive frames. The wide hysteresis is what
25// prevents a single demand spike from causing grow/shrink cycling (which stalls
26// the driver and transiently double-allocates). See tile_compute_buffer_resize_plan.
27struct TileBufferShrinkPolicy {
28 uint32_t growth_slack_percent = 0u; // extra headroom added on growth, as a % of demand
29 uint32_t min_growth_slack_bytes = 0u; // absolute floor for the growth headroom
30 uint32_t shrink_trigger_percent = 50u; // shrink only if required <= capacity * pct/100
31 uint32_t shrink_hysteresis_frames = 120u; // frames demand must stay low before shrinking
32};

52// Shared opt-in policy for the large demand-driven scratch buffers (the global
53// projection buffer + the global sort key/value buffers; the separate instance/
54// depth sort buffers are a future extension). Both grow with per-frame demand and
55// oscillate by ~15x across a single camera sweep, so the
56// hysteresis is wide (4s at 60 fps) and resets on any demand recovery — a shrink is
57// bounded to views that hold low demand for the full window (a slow sweep staying below
58// the trigger for >240 frames can still shrink then re-grow — bounded, not impossible). A 50%
59// trigger means "demand dropped at least one power-of-two octave" (the projection
60// byte size is power-of-two rounded). Growth slack is 0 for both: the projection
61// rounding already provides headroom, and the sorter picks its own capacity.
62constexpr uint32_t TILE_SCRATCH_SHRINK_TRIGGER_PERCENT = 50u;
63constexpr uint32_t TILE_SCRATCH_SHRINK_HYSTERESIS_FRAMES = 240u;
