83class TiledAutoTree {
84public:
85 // Pass 1 counts centres and chooses an adaptive frontier; pass 2 admits and
86 // bakes each tile with its forced lattice root and frozen MERGE_ONLY defaults;
87 // pass 3 stitches tile-root payloads and assembles canonical BFS records and
88 // interior payloads (leaves remain DFS/importance ordered, as in bake_world).
89 // Recheck the assembled tree before publishing. File reopen/decode/W13 belongs
90 // to the D2-3 writer integration, not this in-memory API.
91 // Output must equal in-core bake_world byte for byte, including global leaf
92 // source indices, independent of worker count and chunk enumeration order.
93 // No bound script API, GSQ I/O, Tier A job or cache integration here.
94 // The initial histogram is serial; overfull fine cells are refined by bounded
95 // rereads. Coincident/finest-cell overflow uses the in-core index-split shape
96 // with reserved Morton/global-index scratch, never an oversized spatial tile.
97 //
98 // Synchronous, all-or-nothing: on ANY error reset r_result, drain workers and
99 // release reservations. r_report may retain diagnostics, never payloads.
100 // RAM -> ERR_OUT_OF_MEMORY; deadline -> ERR_TIMEOUT; cancel -> ERR_SKIP;
101 // malformed source/params -> ERR_INVALID_DATA; another offline job -> ERR_BUSY;
102 // success -> OK. Check limits
103 // throughout all three passes and immediately before publication.
104 static Error build(const TiledBakeSource &p_source, const TiledBakeParams &p_params,
105 BakeResult &r_result, TiledBakeReport &r_report, String *r_error = nullptr);
106};
