115class TiledAutoTree {
116public:
117 // Bounded production path. Completed tile payloads leave the coordinator
118 // immediately; only topology and sink-owned frontier scratch survive a batch.
119 static Error build_streamed(const TiledBakeSource &p_source, const TiledBakeParams &p_params,
120 TiledBakeSink &p_sink, TiledBakeReport &r_report, String *r_error = nullptr);
121 // Pass 1 counts centres and chooses an adaptive frontier; pass 2 admits and
122 // bakes each tile with its forced lattice root and frozen MERGE_ONLY defaults,
123 // stitching completed ancestors immediately; pass 3 assembles canonical BFS records.
124 // This compatibility API is an in-memory sink over build_streamed and therefore
125 // explicitly reserves its retained output (leaves remain DFS/importance ordered).
126 // Recheck the assembled tree before publishing. File reopen/decode/W13 belongs
127 // to the D2-3 writer integration, not this in-memory API.
128 // Output must equal in-core bake_world byte for byte, including global leaf
129 // source indices, independent of worker count and chunk enumeration order.
130 // No bound script API, GSQ I/O, Tier A job or cache integration here.
131 // The initial histogram is serial; overfull fine cells are refined by bounded
132 // rereads. Coincident/finest-cell overflow uses the in-core index-split shape
133 // with reserved Morton/global-index scratch, never an oversized spatial tile.
134 //
135 // Synchronous, all-or-nothing: on ANY error reset r_result, drain workers and
136 // release reservations. r_report may retain diagnostics, never payloads.
137 // RAM -> ERR_OUT_OF_MEMORY; deadline -> ERR_TIMEOUT; cancel -> ERR_SKIP;
138 // malformed source/params -> ERR_INVALID_DATA; another offline job -> ERR_BUSY;
139 // success -> OK. Check limits
140 // throughout all three passes and immediately before publication.
141 static Error build(const TiledBakeSource &p_source, const TiledBakeParams &p_params,
142 BakeResult &r_result, TiledBakeReport &r_report, String *r_error = nullptr);
143};
