6// Scene-depth linearization pair for converting a raw NDC depth sample to positive
7// view-space z: perspective uses view_z = mul / (add - raw); orthographic consumers
8// use the z_near/z_far pair directly. ONE derivation shared by the whole-pixel
9// composite (viewport_blit) and the per-splat raster clip (compositing slice D) so
10// both sides convert scene depth with bit-identical math — a divergence would make
11// the raster clip and the composite guard disagree at silhouettes.
12struct GSSceneDepthLinearize {
13 float mul = 0.0f;
14 float add = 1.0f;
15};

19 GSSceneDepthLinearize out;
20 out.mul = p_z_near;

31 }
32 return out;
33}
